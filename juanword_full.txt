package juanword.ui;

import java.awt.AlphaComposite;
import java.awt.Color;
import java.awt.Composite;
import java.awt.Font;
import java.awt.Graphics2D;
import java.awt.RenderingHints;

public final class FloatingText {
    private final String text;
    private final Color color;
    private float x;
    private float y;
    private final float vy;
    private int lifeMs;
    private final int maxLifeMs;

    public FloatingText(String text, int x, int y, Color color) {
        this.text = text != null ? text : "";
        this.color = color != null ? color : Color.WHITE;
        this.x = x;
        this.y = y;
        this.vy = -0.05f;
        this.lifeMs = 900;
        this.maxLifeMs = 900;
    }

    public void update(long dtMs) {
        this.y += vy * dtMs;
        this.lifeMs -= dtMs;
    }

    public boolean isExpired() {
        return lifeMs <= 0;
    }

    public void paint(Graphics2D g2) {
        if (isExpired() || text.isEmpty()) return;
        float alpha = Math.clamp((float) lifeMs / maxLifeMs, 0.0f, 1.0f);

        Composite oldComp = g2.getComposite();
        Object oldHint = g2.getRenderingHint(RenderingHints.KEY_TEXT_ANTIALIASING);
        g2.setRenderingHint(RenderingHints.KEY_TEXT_ANTIALIASING, RenderingHints.VALUE_TEXT_ANTIALIAS_ON);
        g2.setComposite(AlphaComposite.getInstance(AlphaComposite.SRC_OVER, alpha));

        Font font = new Font(Font.SANS_SERIF, Font.BOLD, 22);
        g2.setFont(font);

        // Shadow
        g2.setColor(new Color(0, 0, 0, (int) (alpha * 180)));
        g2.drawString(text, x + 2, y + 2);

        // Foreground
        g2.setColor(new Color(color.getRed(), color.getGreen(), color.getBlue(), (int) (alpha * 255)));
        g2.drawString(text, x, y);

        g2.setComposite(oldComp);
        g2.setRenderingHint(RenderingHints.KEY_TEXT_ANTIALIASING, oldHint != null ? oldHint : RenderingHints.VALUE_TEXT_ANTIALIAS_DEFAULT);
    }
}

package juanword.ui;

import java.awt.AlphaComposite;
import java.awt.Color;
import java.awt.Graphics2D;
import java.awt.geom.AffineTransform;
import java.awt.image.BufferedImage;
import juanword.model.AttackType;

public final class EnemyAnimator {
    private float x;
    private float targetX;
    private long entranceElapsedMs = 600;
    private int flashRemainingMs;
    private float shakeAmplitude;
    private float shakePhase;
    private AttackType attackType;
    private long attackElapsedMs;
    private long attackDurationMs;
    private boolean impactReached;
    private long idleElapsedMs;

    public void startEntrance(int targetX) {
        this.targetX = targetX;
        this.x = 960.0f;
        this.entranceElapsedMs = 0;
    }

    public void startAttack(AttackType type) {
        startAttack(type, 360);
    }

    public void startAttack(AttackType type, long durationMs) {
        this.attackType = (type != null) ? type : AttackType.PUNCH;
        this.attackDurationMs = Math.max(100, durationMs);
        this.attackElapsedMs = 0;
        this.impactReached = false;
    }

    public long getAttackDurationMs() {
        return attackDurationMs;
    }

    public long getAttackElapsedMs() {
        return attackElapsedMs;
    }

    public boolean isAttacking() {
        return attackType != null;
    }

    public boolean consumeImpact() {
        if (impactReached) {
            impactReached = false;
            return true;
        }
        return false;
    }

    public void triggerHit() {
        this.flashRemainingMs = 120;
        this.shakeAmplitude = 14.0f;
        this.shakePhase = 0.0f;
    }

    public void update(long dtMs) {
        if (entranceElapsedMs < 600) {
            entranceElapsedMs += dtMs;
            float t = Math.min(1.0f, (float) entranceElapsedMs / 600.0f);
            float ease = 1.0f - (float) Math.pow(1.0f - t, 3); // cubic ease-out
            this.x = 960.0f + (targetX - 960.0f) * ease;
        } else {
            this.x = targetX;
        }

        if (attackType != null) {
            long prev = attackElapsedMs;
            attackElapsedMs += dtMs;
            if (prev < attackDurationMs / 2 && attackElapsedMs >= attackDurationMs / 2) {
                impactReached = true;
            }
            if (attackElapsedMs >= attackDurationMs) {
                attackType = null;
            }
        } else {
            idleElapsedMs += dtMs;
        }

        if (flashRemainingMs > 0) {
            flashRemainingMs = (int) Math.max(0, flashRemainingMs - dtMs);
        }

        if (shakeAmplitude > 0) {
            shakePhase += 0.04f * dtMs;
            shakeAmplitude = Math.max(0.0f, shakeAmplitude - 0.04f * dtMs);
        }
    }

    public BufferedImage getActiveSprite(BufferedImage idle, BufferedImage attack) {
        if (attackType == null || attack == null) return idle;
        float t = (float) attackElapsedMs / attackDurationMs;
        if (t >= 0.30f && t < 0.78f) {
            return attack;
        }
        return idle;
    }

    public float getDrawOffsetX() {
        float lungeX = 0f;
        if (attackType != null) {
            float t = (float) attackElapsedMs / attackDurationMs;
            if (t < 0.30f) {
                float p = t / 0.30f;
                // Pull back to the right (away from player on the left)
                lungeX = (float) (12.0 * Math.sin(p * Math.PI * 0.5));
            } else if (t < 0.78f) {
                float p = (t - 0.30f) / 0.48f;
                // Lunge forward to the left (towards player on the left)
                lungeX = (float) (-38.0 * Math.sin(p * Math.PI));
            } else {
                float p = (t - 0.78f) / 0.22f;
                lungeX = (float) (-12.0 * (1.0 - p));
            }
        }
        float shakeOffset = (shakeAmplitude > 0) ? (float) (Math.sin(shakePhase) * shakeAmplitude) : 0f;
        return lungeX + shakeOffset;
    }

    public float getDrawOffsetY() {
        if (attackType == null) {
            return (float) (Math.sin(idleElapsedMs * 0.0035) * 2.5);
        }
        return 0f;
    }

    public float getShadowScale() {
        if (attackType == null) {
            return 1.0f - 0.04f * (float) Math.sin(idleElapsedMs * 0.0035);
        }
        float t = (float) attackElapsedMs / attackDurationMs;
        if (t >= 0.30f && t < 0.78f) {
            float p = (t - 0.30f) / 0.48f;
            return 1.0f + 0.12f * (float) Math.sin(p * Math.PI);
        }
        return 1.0f;
    }

    public long getIdleElapsedMs() {
        return idleElapsedMs;
    }

    public boolean isEntering() {
        return entranceElapsedMs < 600;
    }

    public void completeEntrance() {
        this.entranceElapsedMs = 600;
        this.x = targetX;
    }

    public void paint(Graphics2D g2, BufferedImage sprite, int y) {
        if (sprite == null) return;
        paint(g2, sprite, (int) x, y, sprite.getWidth(), sprite.getHeight());
    }

    public void paint(Graphics2D g2, BufferedImage sprite, int targetX, int targetY, int targetW, int targetH) {
        paint(g2, sprite, targetX, targetY, targetW, targetH, 0.90);
    }

    public void paint(Graphics2D g2, BufferedImage sprite, int targetX, int targetY, int targetW, int targetH, double footRatio) {
        if (sprite == null) return;

        AffineTransform oldXf = g2.getTransform();
        float offsetX = getDrawOffsetX();
        float offsetY = getDrawOffsetY();

        double rotation = 0.0;
        double scaleX = 1.0;
        double scaleY = 1.0;

        if (attackType != null) {
            float t = (float) attackElapsedMs / attackDurationMs;
            if (t < 0.30f) {
                // Wind-up: pull back and lean back (clockwise tilt)
                float p = t / 0.30f;
                rotation = 0.14 * p;
                scaleX = 0.98;
                scaleY = 1.02;
            } else if (t < 0.78f) {
                // Strike: lean forward into strike (counter-clockwise tilt)
                float p = (t - 0.30f) / 0.48f;
                rotation = 0.14 - 0.30 * Math.sin(p * Math.PI);
                scaleX = 1.08;
                scaleY = 0.96;
            } else {
                // Recover
                float p = (t - 0.78f) / 0.22f;
                rotation = -0.12 * (1.0 - p);
                scaleX = 1.0 + 0.03 * (1.0 - p);
            }
        } else {
            // Idle breathing subtle scale
            scaleY = 1.0 + 0.015 * Math.sin(idleElapsedMs * 0.0035);
            scaleX = 1.0 - 0.008 * Math.sin(idleElapsedMs * 0.0035);
        }

        int baseX = isEntering() ? (int) x : targetX;
        int drawX = (int) (baseX + offsetX);
        int drawY = (int) (targetY + offsetY);

        // Pivot around feet / lower center for grounded, natural rotation
        double pivotX = drawX + targetW / 2.0;
        double pivotRatio = (footRatio > 0.0 && footRatio <= 1.0) ? footRatio : 0.90;
        double pivotY = drawY + targetH * pivotRatio;

        g2.translate(pivotX, pivotY);
        if (rotation != 0.0) g2.rotate(rotation);
        if (scaleX != 1.0 || scaleY != 1.0) g2.scale(scaleX, scaleY);
        g2.translate(-targetW / 2.0, -targetH * pivotRatio);

        g2.drawImage(sprite, 0, 0, targetW, targetH, null);

        if (flashRemainingMs > 0) {
            BufferedImage flashMask = new BufferedImage(Math.max(1, targetW), Math.max(1, targetH), BufferedImage.TYPE_INT_ARGB);
            Graphics2D fg = flashMask.createGraphics();
            fg.drawImage(sprite, 0, 0, targetW, targetH, null);
            fg.setComposite(AlphaComposite.SrcIn);
            fg.setColor(new Color(255, 255, 255, 200));
            fg.fillRect(0, 0, targetW, targetH);
            fg.dispose();
            g2.drawImage(flashMask, 0, 0, null);
        }

        g2.setTransform(oldXf);
    }
}

    private void recalculateLayout() {
        int w = getWidth() > 0 ? getWidth() : 960;
        int h = getHeight() > 0 ? getHeight() : 540;

        // Scale factors relative to 960x540
        double scaleX = w / 960.0;
        double scaleY = h / 540.0;
        double S = Math.min(scaleX, scaleY);

        // Tile grid dimensions
        int tileSize = (int) Math.round(50 * S);
        int tileGap = (int) Math.round(6 * S);
        int totalGridW = 4 * tileSize + 3 * tileGap;
        int gridX = (w - totalGridW) / 2;
        int gridY = (int) Math.round(296 * scaleY);

        for (int r = 0; r < 4; r++) {
            for (int c = 0; c < 4; c++) {
                tileBounds[r][c] = new Rectangle(
                    gridX + c * (tileSize + tileGap),
                    gridY + r * (tileSize + tileGap),
                    tileSize, tileSize
                );
            }

        int btnW = (int) Math.round(112 * scaleX);
        int btnH = (int) Math.round(34 * scaleY);

        clearBtnBounds.setBounds(gridX - btnW - (int) (20 * scaleX), gridY + (int) (98 * scaleY), btnW, btnH);

        scrambleBtnBounds.setBounds(gridX + totalGridW + (int) (20 * scaleX), gridY + (int) (54 * scaleY), btnW, btnH);

        backToMapBounds.setBounds((int) (24 * scaleX), (int) (14 * scaleY), (int) (145 * scaleX), (int) (34 * scaleY));

        // Arena card bounds & attack button

        int centerX = cardX + juanW + (int) (10 * scaleX);
        int centerW = cardW - juanW - enemyW - (int) (20 * scaleX);
        int actY = cardY + (int) (172 * scaleY);
        int actH = (int) (44 * scaleY);

    private void gameTick(ActionEvent e) {
        long now = System.currentTimeMillis();
        long dt = (lastTickTime == 0) ? 16L : (now - lastTickTime);
        lastTickTime = now;
        long dtMs = Math.clamp(dt, 1L, 100L);

        if (paused) {
            repaint();
            return;
        }

        playerAnimator.update(dtMs);
        enemyAnimator.update(dtMs);

        for (AttackEffect ae : attackEffects) ae.update(dtMs);
        attackEffects.removeIf(AttackEffect::isFinished);

        for (FloatingText ft : floatingTexts) ft.update(dtMs);
        floatingTexts.removeIf(FloatingText::isExpired);

        if (uiShakeFrames > 0) {
            uiShakeFrames = Math.max(0, uiShakeFrames - Math.max(1, (int) Math.round(dtMs / 16.0)));
        }

        if (playerAnimator.consumeImpact()) {
            if (pendingImpactAction != null) {
                Runnable act = pendingImpactAction;
                pendingImpactAction = null;
                act.run();
            }
        }
        if (enemyAnimator.consumeImpact()) {
            if (pendingImpactAction != null) {
                Runnable act = pendingImpactAction;
                pendingImpactAction = null;
                act.run();
            }
        }

        // Count turn timer
        tickAccumulatorMs += dtMs;
        while (tickAccumulatorMs >= 1000) {
            tickAccumulatorMs -= 1000;
            if (session != null && session.getDifficulty().isTimed() && !inputLocked) {
                int sec = session.getTurnSeconds() - 1;
                session.setTurnSeconds(sec);
                if (sec <= 0) {
                    onTurnTimerExpired();
                }
            }
        }

        outcome = evaluateOutcome();
        if (outcome == BattleOutcome.VICTORY) {
            onVictory();
            return;
        } else if (outcome == BattleOutcome.DEFEAT) {
            onDefeat();
            return;
        }

        repaint();
    }

    private void onMouseMoved(MouseEvent e) {
        Point p = e.getPoint();
        boolean hover = backToMapBounds.contains(p) || attackBtnBounds.contains(p)
            || backTileBtnBounds.contains(p) || clearBtnBounds.contains(p) || scrambleBtnBounds.contains(p);

        if (!hover) {
            for (int r = 0; r < 4; r++) {
                for (int c = 0; c < 4; c++) {
                    if (tileBounds[r][c] != null && tileBounds[r][c].contains(p)) {
                        hover = true;
                        break;
                    }
                }
            }
        }
        setCursor(hover ? Cursor.getPredefinedCursor(Cursor.HAND_CURSOR) : Cursor.getDefaultCursor());
    }

    private void onMouseClicked(MouseEvent e) {
        if (inputLocked || paused) return;
        Point p = e.getPoint();

        if (backToMapBounds.contains(p)) {
            saveMidBattle();
            loop.stop();
            host.returnToMap();
            return;
        }
        if (attackBtnBounds.contains(p)) {
            onSubmit();
            return;
        }
        if (backTileBtnBounds.contains(p)) {
            if (!selectedPath.isEmpty()) {
                selectedPath.remove(selectedPath.size() - 1);
                repaint();
            }
            return;
        }
        if (clearBtnBounds.contains(p)) {
            onClear();
            return;
        }
        if (scrambleBtnBounds.contains(p)) {
            BattleContext ctx = new BattleContext(session, "", (txt, col) -> spawnFloatingText(txt, new Point(getWidth() / 2, 340), col));
            ctx.scrambleGrid();
            selectedPath.clear();

            // Clicking Bag-o inflicts self damage
            int selfDamage = 5;
            ctx.damagePlayer(selfDamage);
            playerAnimator.triggerHit();
            uiShakeFrames = 6;
            spawnFloatingText("-" + selfDamage + " HP", new Point(140, 220), Color.RED);

            dialogueText = "Gibag-o ang tanang letra! Naangol ka ug " + selfDamage + " HP sa Bag-o.";
            if (session.getPlayerHp() <= 0) {
                onDefeat();
            }
            repaint();
            return;
        }

        for (int r = 0; r < 4; r++) {
            for (int c = 0; c < 4; c++) {
                if (tileBounds[r][c] != null && tileBounds[r][c].contains(p)) {
                    onTileClicked(r, c);
                    return;
                }
            }
        }
    }

    private void onTileClicked(int row, int col) {
        int existingIndex = -1;
        for (int i = 0; i < selectedPath.size(); i++) {
            int[] pos = selectedPath.get(i);
            if (pos[0] == row && pos[1] == col) {
                existingIndex = i;
                break;
            }
        }
        if (existingIndex >= 0) {
            while (selectedPath.size() > existingIndex) {
                selectedPath.remove(selectedPath.size() - 1);
            }
        } else {
            selectedPath.add(new int[]{row, col});
        }
        repaint();
    }

        submitWord(word);
    }

    private String getSelectedWord() {
        StringBuilder sb = new StringBuilder();
        TileComponent[][] grid = session.getGrid();
        for (int[] pos : selectedPath) {
            sb.append(grid[pos[0]][pos[1]].getLetter());
        }
        return sb.toString();
    }

    public void submitWord(String word) {
        if (DataManager.getInstance().isValidCebuanoWord(word)) {
            List<TileComponent> tiles = new ArrayList<>();
            TileComponent[][] grid = session.getGrid();
            for (int[] pos : selectedPath) {
                tiles.add(grid[pos[0]][pos[1]]);
            }
            resolveValidWord(tiles);
        } else {
            handleInvalidWord();
        }
    }

    public int computeDamage(List<TileComponent> tiles) {
        if (tiles == null || tiles.isEmpty()) return 0;
        int sum = 0;
        for (TileComponent t : tiles) {
            sum += t.getPoints();
        }
        int wordLen = tiles.size();
        double mult = (session != null && session.getDifficulty() != null)
            ? session.getDifficulty().getLengthMultiplier(wordLen)
            : 1.0;
        double dmg = sum * mult;

        if (session != null) {
            boolean isExtreme = session.getDifficulty() == DifficultySetting.EXTREME || session.getMapIndex() == 4;
            boolean isFinalBoss = session.getEnemy() != null && "GUDO".equalsIgnoreCase(session.getEnemy().getEnemyId());
            boolean isBoss = session.getEnemy() != null && session.getEnemy().isBoss();

            if (isExtreme) {
                // Ghost increase for Extreme: scale damage up to match 1.5x enemy HP pools
                double ghostMult = 2.0;
                int ghostFlat = 10 + Math.max(0, (wordLen - 2) * 4);
                dmg = dmg * ghostMult + ghostFlat;

                if (isFinalBoss) {
                    // Extra ghost increase specifically for the final boss (Gudo)
                    dmg = dmg * 1.35 + 12;
                } else if (isBoss) {
                    // Ghost increase for Extreme boss (e.g. Porter)
                    dmg = dmg * 1.30 + 10;
                }
            } else if (isFinalBoss) {
                // If Gudo is fought outside Extreme, still grant the final boss ghost boost
                dmg = dmg * 1.50 + 20;
            }
        }

        return Math.max(1, (int) Math.round(dmg));
    }

    private void resolveValidWord(List<TileComponent> tiles) {
        int dmg = computeDamage(tiles);
        String word = getSelectedWord();
        dialogueText = "Giatake nimo si " + session.getEnemy().getName() + " gamit ang pulong '" + word + "' (" + dmg + " DMG)!";
        playPlayerAttack(() -> onPlayerImpact(dmg));

        BattleContext ctx = new BattleContext(session, word, (txt, col) -> spawnFloatingText(txt, getPlayerCenter(), col));
        for (TileComponent t : tiles) {
            t.onWordSubmitted(ctx);
        }

        refillConsumedTiles(selectedPath);
        selectedPath.clear();
    }

    private void playPlayerAttack(Runnable onImpact) {
        inputLocked = true;
        this.pendingImpactAction = onImpact;
        boolean isGudoPlayer = session != null && session.getPlayer() != null &&
            session.getPlayer().getPlayerName() != null &&
            session.getPlayer().getPlayerName().equalsIgnoreCase("Gudo");
        long duration = isGudoPlayer ? 800L : 450L;
        playerAnimator.startAttack(AttackType.SLASH, duration);
    }

    private void onPlayerImpact(int damage) {
        enemyAnimator.triggerHit();
        Point ep = getEnemyCenter();
        attackEffects.add(new AttackEffect(AttackType.SLASH, ep.x, ep.y));
        session.getEnemy().takeDamage(damage);
        spawnFloatingText("-" + damage, ep, new Color(255, 140, 0));

        if (session.getEnemy().isDefeated()) {
            outcome = BattleOutcome.VICTORY;
        } else {
            enemyTurn();
        }
    }

    private void playEnemyAttack() {
        inputLocked = true;
        AttackType type = session.getEnemy().getAttackType();
        this.pendingImpactAction = this::onEnemyImpact;
        boolean isGudo = session != null && session.getEnemy() != null &&
            "GUDO".equalsIgnoreCase(session.getEnemy().getEnemyId());
        long duration = isGudo ? 800L : 450L;
        enemyAnimator.startAttack(type, duration);
    }

    private void onEnemyImpact() {
        playerAnimator.triggerHit();
        Point pp = getPlayerCenter();
        attackEffects.add(new AttackEffect(session.getEnemy().getAttackType(), pp.x, pp.y));
        dialogueText = session.getEnemy().getName() + " mibalos ug atake (" + session.getEnemy().getAttack() + " ATK)!";
        BattleContext ctx = new BattleContext(session, "", (txt, col) -> spawnFloatingText(txt, getEnemyCenter(), col));
        session.getEnemy().onEnemyTurn(ctx);

        if (session.getPlayerHp() <= 0) {
            outcome = BattleOutcome.DEFEAT;
        } else {
            inputLocked = false;
            session.setTurnSeconds(session.getDifficulty().getTurnTimeSeconds());
        }
    }

    private void handleInvalidWord() {
        uiShakeFrames = 15;
        dialogueText = "Sayop nga pulong! Dili kini anaa sa diksyonaryo sa Bisaya.";
        spawnFloatingText("Sayop!", new Point(getWidth() / 2, 280), Color.RED);

        int counterDmg = session.getDifficulty().getInvalidCounterDamage();
        if (counterDmg > 0) {
            session.setPlayerHp(session.getPlayerHp() - counterDmg);
            spawnFloatingText("-" + counterDmg + " (Sayop)", getPlayerCenter(), Color.RED);
            if (session.getPlayerHp() <= 0) {
                outcome = BattleOutcome.DEFEAT;
            }
        }
        selectedPath.clear();
        repaint();
    }

    private void onTurnTimerExpired() {
        selectedPath.clear();
        dialogueText = "Nahurot ang oras! Turno na sa kontra.";
        spawnFloatingText("Nahurot ang Oras!", new Point(getWidth() / 2, 280), Color.YELLOW);
        playEnemyAttack();
    }

    private void enemyTurn() {
        Timer timer = new Timer(400, ev -> {
            ((Timer) ev.getSource()).stop();
            playEnemyAttack();
        });
        timer.setRepeats(false);
        timer.start();
    }

    private void refillConsumedTiles(List<int[]> used) {
        TileGridGenerator.refill(session.getGrid(), used);
    }

    private BattleOutcome evaluateOutcome() {
        if (session != null && session.getEnemy().isDefeated()) {
            return BattleOutcome.VICTORY;
        }
        if (session != null && session.getPlayerHp() <= 0) {
            return BattleOutcome.DEFEAT;
        }
        return BattleOutcome.ONGOING;
    }

    public void onVictory() {
        loop.stop();
        int scoreAward = 100 * session.getStageIndex();
        session.getPlayer().addScore(scoreAward);
        session.getPlayer().recordDefeat(session.getEnemy().getEnemyId());

        // In Mt. Apo, beating Boss Porter (stage 3) unlocks Gudo and proceeds immediately to the Gudo boss
        if (session.getMapIndex() == 4 && session.getStageIndex() == 3) {
            DataManager.getInstance().clearActiveBattle(session.getPlayer());
            DataManager.getInstance().savePlayers();
            BattleSession nextSession = new BattleSessionBuilder()
                .withPlayer(session.getPlayer())
                .onMap(4)
                .atStage(4)
                .withSavedHealth(session.getPlayerHp(), null)
                .build();
            begin(nextSession);
            return;
        }

        if (session.isBossStage()) {
            if (session.getMapIndex() == session.getPlayer().getUnlockedMap() && session.getMapIndex() < 4) {
                session.getPlayer().unlockMap(session.getMapIndex() + 1);
            }
            DataManager.getInstance().clearActiveBattle(session.getPlayer());
            DataManager.getInstance().savePlayers();

            if (session.isFinalBoss()) {
                JOptionPane.showMessageDialog(this, "NADAUG KA SA TANANG MAPA UG NALUPIG NIMO SI GUDO!\nCongratulations!", "KADAUGAN!", JOptionPane.INFORMATION_MESSAGE);
            } else {
                JOptionPane.showMessageDialog(this, "NADAUG ANG BOSS!\nNaabli ang sunod nga mapa!", "DAUG!", JOptionPane.INFORMATION_MESSAGE);
            }
            host.returnToMap();
        } else {
            DataManager.getInstance().clearActiveBattle(session.getPlayer());
            DataManager.getInstance().savePlayers();
            // Automatically advance to the next enemy with zero confirmation prompt
            BattleSession nextSession = new BattleSessionBuilder()
                .withPlayer(session.getPlayer())
                .onMap(session.getMapIndex())
                .atStage(session.getStageIndex() + 1)
                .withSavedHealth(session.getPlayerHp(), null)
                .build();
            begin(nextSession);
        }
    }

    private void onDefeat() {
        loop.stop();
        DataManager.getInstance().clearActiveBattle(session.getPlayer());
        DataManager.getInstance().savePlayers();
        JOptionPane.showMessageDialog(this, "NAPILDE KA!\nSulayi pag-usab.", "Pilde", JOptionPane.ERROR_MESSAGE);
        host.returnToMap();
    }

        this.paused = false;
        this.lastTickTime = System.currentTimeMillis();
    }

    public int getSelectedDamagePreview() {
        if (selectedPath == null || selectedPath.isEmpty() || session == null) return 0;
        TileComponent[][] grid = session.getGrid();
        if (grid == null) return 0;
        List<TileComponent> tiles = new ArrayList<>();
        for (int[] pos : selectedPath) {
            if (pos[0] >= 0 && pos[0] < 4 && pos[1] >= 0 && pos[1] < 4) {
                tiles.add(grid[pos[0]][pos[1]]);
            }
        }
        return computeDamage(tiles);
    }

    }

    @Override
    protected void paintComponent(Graphics g) {
        super.paintComponent(g);
        Graphics2D g2 = (Graphics2D) g;
        g2.setRenderingHint(RenderingHints.KEY_ANTIALIASING, RenderingHints.VALUE_ANTIALIAS_ON);
        g2.setRenderingHint(RenderingHints.KEY_TEXT_ANTIALIASING, RenderingHints.VALUE_TEXT_ANTIALIAS_ON);
        g2.setRenderingHint(RenderingHints.KEY_INTERPOLATION, RenderingHints.VALUE_INTERPOLATION_BILINEAR);

        int w = getWidth();
        int h = getHeight();
        recalculateLayout();

        // Screen shake
        if (uiShakeFrames > 0) {
            int sx = (int) (Math.sin(uiShakeFrames * 1.5) * 6);
            g2.translate(sx, 0);
        }

        // 1-3. Cached Composite Background (Sky + Landmark + Scrim)
        renderCachedBackground(g2, w, h);

        if (session == null) {
            drawIndigenousFrame(g2, w, h);
            return;
        }

        // 4. TOP HEADER BAR
        drawTopHeaderBar(g2, w, h);

        // 5. COMBAT ARENA CARD (Juan, Enemy, Dialogue Log, Word Rack & Attack Action)
        drawArenaCard(g2, w, h);

        // 6. BOTTOM LETTER TILES & CONTROLS
        drawBottomLetterGridAndButtons(g2, w, h);

        // 7. ATTACK EFFECTS & FLOATING TEXTS
        for (AttackEffect ae : attackEffects) ae.paint(g2);
        for (FloatingText ft : floatingTexts) ft.paint(g2);

        // 8. LAYER 4: INDIGENOUS FRAME BORDER (Strictly overlays border on all resolutions)
        drawIndigenousFrame(g2, w, h);

        // 9. PAUSE OVERLAY
        if (paused) drawPauseOverlay(g2, w, h);
    }

    public Landmark(int mapIndex, String name, String spriteFile, Rectangle bounds) {
        this.mapIndex = mapIndex;
        this.name = name != null ? name : "Landmark " + mapIndex;
        this.bounds = bounds != null ? new Rectangle(bounds) : new Rectangle(0, 0, 100, 100);
        BufferedImage raw = ImageLoader.load(spriteFile);
        if (raw != null) {
            if (mapIndex == 1) {
                int cw = Math.min(648, raw.getWidth());
                int ch = Math.min(512, raw.getHeight());
                this.sprite = raw.getSubimage(0, 0, cw, ch);
            } else if (mapIndex == 2) {
                int subY = Math.min(300, raw.getHeight() - 1);
                int subH = Math.min(1620, raw.getHeight() - subY);
                this.sprite = raw.getSubimage(0, subY, raw.getWidth(), subH);
            } else if (mapIndex == 3) {
                int subY = Math.min(201, raw.getHeight() - 1);
                int subH = Math.min(865, raw.getHeight() - subY);
                int subW = Math.min(1612, raw.getWidth());
                this.sprite = raw.getSubimage(0, subY, subW, subH);
            } else {
                this.sprite = raw;
            }
            this.silhouette = ImageLoader.createSilhouette(this.sprite, new Color(14, 18, 28, 252));
        } else {
            this.sprite = null;
            this.silhouette = null;
        }
    }

    public boolean contains(Point p) {
        if (bounds == null || p == null) return false;
        if (mapIndex == 4) {
            int top = (int) (bounds.height * 0.08);
            int bottom = (int) (bounds.height * 0.72);
            if (p.y < top || p.y > bottom || p.x < 0 || p.x > bounds.width) {
                return false;
            }
            if (sprite != null && bounds.width > 0 && bounds.height > 0) {
                int imgX = (int) ((double) p.x / bounds.width * sprite.getWidth());
                int imgY = (int) ((double) p.y / bounds.height * sprite.getHeight());
                if (imgX >= 0 && imgX < sprite.getWidth() && imgY >= 0 && imgY < sprite.getHeight()) {
                    int pixel = sprite.getRGB(imgX, imgY);
                    int alpha = (pixel >> 24) & 0xff;
                    return alpha > 30;
                }
            }
            return true;
        }
        return bounds.contains(p);
    }

    public void setLocked(boolean locked) {
        this.locked = locked;
        if (this.locked && this.silhouette == null && this.sprite != null) {

public final class AttackEffect {
    private final AttackType type;
    private final int centerX;
    private final int centerY;
    private long elapsedMs;
    private final long durationMs;
    private boolean finished;

    public AttackEffect(AttackType type, int centerX, int centerY) {
        this.type = type != null ? type : AttackType.SLASH;
        this.centerX = centerX;
        this.centerY = centerY;
        this.elapsedMs = 0;
        this.durationMs = 300;
        this.finished = false;
    }

    public void update(long dtMs) {
        elapsedMs += dtMs;
        if (elapsedMs >= durationMs) {
            finished = true;
        }
    }

    public boolean isFinished() {
        return finished;
    }

    public void paint(Graphics2D g2) {
        if (finished) return;
        float progress = Math.min(1.0f, (float) elapsedMs / durationMs);
        float alpha = Math.max(0.0f, 1.0f - progress);

        Composite oldComp = g2.getComposite();
        Object oldHint = g2.getRenderingHint(RenderingHints.KEY_ANTIALIASING);
        g2.setRenderingHint(RenderingHints.KEY_ANTIALIASING, RenderingHints.VALUE_ANTIALIAS_ON);

        if (type == AttackType.SLASH) {
            int len = (int) (60 + progress * 60);
            int x1 = centerX - len / 2;
            int y1 = centerY - len / 2;
            int x2 = centerX + len / 2;
            int y2 = centerY + len / 2;

            g2.drawLine(x1, y1, x2, y2);
            g2.setColor(new Color(100, 200, 255, (int) (alpha * 180)));
            g2.drawLine(x1 - 4, y1, x2 - 4, y2);
        } else {
            // PUNCH burst / ring

package juanword.ui;

import java.awt.AlphaComposite;
import java.awt.Color;
import java.awt.Graphics2D;
import java.awt.geom.AffineTransform;
import java.awt.image.BufferedImage;
import juanword.model.AttackType;

public final class PlayerAnimator {
    private final int anchorX;
    private final int anchorY;
    private AttackType attackType;
    private long attackElapsedMs;
    private long attackDurationMs;
    private boolean impactReached;
    private int flashRemainingMs;
    private float shakeAmplitude;
    private float shakePhase;
    private long idleElapsedMs;

    public PlayerAnimator(int anchorX, int anchorY) {
        this.anchorX = anchorX;
        this.anchorY = anchorY;
    }

    public void startAttack(AttackType type) {
        startAttack(type, 360);
    }

    public void startAttack(AttackType type, long durationMs) {
        this.attackType = (type != null) ? type : AttackType.SLASH;
        this.attackDurationMs = Math.max(100, durationMs);
        this.attackElapsedMs = 0;
        this.impactReached = false;
    }

    public long getAttackDurationMs() {
        return attackDurationMs;
    }

    public long getAttackElapsedMs() {
        return attackElapsedMs;
    }

    public boolean isAttacking() {
        return attackType != null;
    }

    public BufferedImage getActiveSprite(BufferedImage idle, BufferedImage preAttack, BufferedImage attack) {
        if (attackType == null) return idle;
        float t = (float) attackElapsedMs / attackDurationMs;
        if (t < 0.30f) {
            return (preAttack != null) ? preAttack : idle;
        } else if (t < 0.78f) {
            return (attack != null) ? attack : idle;
        } else {
            return idle;
        }
    }

    public boolean consumeImpact() {
        if (impactReached) {
            impactReached = false;
            return true;
        }
        return false;
    }

    public void triggerHit() {
        this.flashRemainingMs = 120;
        this.shakeAmplitude = 14.0f;
        this.shakePhase = 0.0f;
    }

    public void update(long dtMs) {
        if (attackType != null) {
            long prev = attackElapsedMs;
            attackElapsedMs += dtMs;
            if (prev < attackDurationMs / 2 && attackElapsedMs >= attackDurationMs / 2) {
                impactReached = true;
            }
            if (attackElapsedMs >= attackDurationMs) {
                attackType = null;
            }
        } else {
            idleElapsedMs += dtMs;
        }

        if (flashRemainingMs > 0) {
            flashRemainingMs = (int) Math.max(0, flashRemainingMs - dtMs);
        }

        if (shakeAmplitude > 0) {
            shakePhase += 0.04f * dtMs;
            shakeAmplitude = Math.max(0.0f, shakeAmplitude - 0.04f * dtMs);
        }
    }

    public float getDrawOffsetX() {
        float lungeX = 0f;
        if (attackType != null) {
            float t = (float) attackElapsedMs / attackDurationMs;
            if (t < 0.30f) {
                float p = t / 0.30f;
                lungeX = (float) (-12.0 * Math.sin(p * Math.PI * 0.5));
            } else if (t < 0.78f) {
                float p = (t - 0.30f) / 0.48f;
                lungeX = (float) (38.0 * Math.sin(p * Math.PI));
            } else {
                float p = (t - 0.78f) / 0.22f;
                lungeX = (float) (12.0 * (1.0 - p));
            }
        }
        float shakeOffset = (shakeAmplitude > 0) ? (float) (Math.sin(shakePhase) * shakeAmplitude) : 0f;
        return lungeX + shakeOffset;
    }

    public float getDrawOffsetY() {
        if (attackType == null) {
            return (float) (Math.sin(idleElapsedMs * 0.0035) * 2.5);
        }
        return 0f;
    }

    public float getShadowScale() {
        if (attackType == null) {
            return 1.0f - 0.04f * (float) Math.sin(idleElapsedMs * 0.0035);
        }
        float t = (float) attackElapsedMs / attackDurationMs;
        if (t >= 0.30f && t < 0.78f) {
            float p = (t - 0.30f) / 0.48f;
            return 1.0f + 0.12f * (float) Math.sin(p * Math.PI);
        }
        return 1.0f;
    }

    public long getIdleElapsedMs() {
        return idleElapsedMs;
    }

    public void paint(Graphics2D g2, BufferedImage sprite) {
        if (sprite == null) return;
        paint(g2, sprite, anchorX, anchorY, sprite.getWidth(), sprite.getHeight());
    }

    public void paint(Graphics2D g2, BufferedImage sprite, int targetX, int targetY, int targetW, int targetH) {
        paint(g2, sprite, targetX, targetY, targetW, targetH, 0.90);
    }

    public void paint(Graphics2D g2, BufferedImage sprite, int targetX, int targetY, int targetW, int targetH, double footRatio) {
        if (sprite == null) return;

        AffineTransform oldXf = g2.getTransform();
        float offsetX = getDrawOffsetX();
        float offsetY = getDrawOffsetY();

        double rotation = 0.0;
        double scaleX = 1.0;
        double scaleY = 1.0;

        if (attackType != null) {
            float t = (float) attackElapsedMs / attackDurationMs;
            if (t < 0.30f) {
                // Wind-up: pull back and lean back slightly
                float p = t / 0.30f;
                rotation = -0.14 * p;
                scaleX = 0.98;
                scaleY = 1.02;
            } else if (t < 0.78f) {
                // Strike: lean forward into slash
                float p = (t - 0.30f) / 0.48f;
                rotation = -0.14 + 0.30 * Math.sin(p * Math.PI);
                scaleX = 1.08;
                scaleY = 0.96;
            } else {
                // Recover back to idle
                float p = (t - 0.78f) / 0.22f;
                rotation = 0.12 * (1.0 - p);
                scaleX = 1.0 + 0.03 * (1.0 - p);
            }
        } else {
            // Idle breathing subtle scale
            scaleY = 1.0 + 0.015 * Math.sin(idleElapsedMs * 0.0035);
            scaleX = 1.0 - 0.008 * Math.sin(idleElapsedMs * 0.0035);
        }

        int drawX = (int) (targetX + offsetX);
        int drawY = (int) (targetY + offsetY);

        // Pivot around feet / lower center for grounded, natural rotation
        double pivotX = drawX + targetW / 2.0;
        double pivotRatio = (footRatio > 0.0 && footRatio <= 1.0) ? footRatio : 0.90;
        double pivotY = drawY + targetH * pivotRatio;

        g2.translate(pivotX, pivotY);
        if (rotation != 0.0) g2.rotate(rotation);
        if (scaleX != 1.0 || scaleY != 1.0) g2.scale(scaleX, scaleY);
        g2.translate(-targetW / 2.0, -targetH * pivotRatio);

        g2.drawImage(sprite, 0, 0, targetW, targetH, null);

        if (flashRemainingMs > 0) {
            BufferedImage flashMask = new BufferedImage(Math.max(1, targetW), Math.max(1, targetH), BufferedImage.TYPE_INT_ARGB);
            Graphics2D fg = flashMask.createGraphics();
            fg.drawImage(sprite, 0, 0, targetW, targetH, null);
            fg.setComposite(AlphaComposite.SrcIn);
            fg.setColor(new Color(255, 255, 255, 200));
            fg.fillRect(0, 0, targetW, targetH);
            fg.dispose();
            g2.drawImage(flashMask, 0, 0, null);
        }

        g2.setTransform(oldXf);
    }
}

    private void onContinue() {
        Player active = host.getActivePlayer();
        if (active != null) {
            if (active.hasActiveBattle()) {
                host.resumeBattle(active);
            } else {
                host.showScreen(Screen.MAP_SELECT);
            }
        } else if (!profiles.isEmpty()) {

            if (p.hasActiveBattle()) {
                host.resumeBattle(p);
            } else {

            }

        }
    }

    public void recalculateLayout() {
        int w = getWidth() > 0 ? getWidth() : 960;
        int h = getHeight() > 0 ? getHeight() : 540;
        if (w == lastW && h == lastH) return;
        lastW = w;
        lastH = h;

        double scaleX = w / 960.0;
        double scaleY = h / 540.0;
        double S = Math.min(scaleX, scaleY);

        // 1. Responsive Player Card Bounds (top-left inset within indigenous frame)
        int cardX = (int) Math.round(36 * scaleX);
        int cardY = (int) Math.max(24 * scaleY, 20 + 10 * scaleY);
        int cardW = (int) Math.round(230 * S);
        int cardH = (int) Math.round(48 * S);
        playerCardBounds.setBounds(cardX, cardY, cardW, cardH);

        // 2. Responsive Menu Buttons
        // Buttons scale with S and are centered horizontally
        int btnW = (int) Math.round(196 * S);
        int btnH = (int) Math.round(48 * S);
        int gap = (int) Math.round(6 * S);
        int totalBtnH = (btnH * 4) + (gap * 3);

        int bottomMargin = (int) Math.round(30 * scaleY);
        int startY = h - bottomMargin - totalBtnH;
        int btnX = (w - btnW) / 2;

        continueBtn.setBounds(btnX, startY, btnW, btnH);
        newGameBtn.setBounds(btnX, startY + (btnH + gap), btnW, btnH);
        settingBtn.setBounds(btnX, startY + (btnH + gap) * 2, btnW, btnH);
        exitBtn.setBounds(btnX, startY + (btnH + gap) * 3, btnW, btnH);

        // 3. Title Logo Bounds
        // Title logo must be placed in the space above startY without overlapping buttons
        if (titleLogo != null && titleLogo.getWidth() > 0) {
            double aspect = (double) titleLogo.getHeight() / titleLogo.getWidth();
            int topMargin = (int) Math.round(26 * scaleY);
            int availableH = startY - topMargin - (int) Math.round(14 * scaleY);

            int maxTitleW = (int) (w * 0.54);
            int titleW = maxTitleW;
            int titleH = (int) Math.round(titleW * aspect);

            if (titleH > availableH && availableH > 50) {
                titleH = availableH;
                titleW = (int) Math.round(titleH / aspect);
            }

            int titleX = (w - titleW) / 2;
            int titleY = topMargin + Math.max(0, (availableH - titleH) / 2);
            titleBounds.setBounds(titleX, titleY, titleW, titleH);
        }
    }

    private void drawMenuButton(Graphics2D g2, JButton btn, BufferedImage sprite) {
        if (sprite == null || btn == null || !btn.isVisible()) {
            return;
        }
        Rectangle b = btn.getBounds();
        ButtonModel model = btn.getModel();
        boolean pressed = model.isPressed() && model.isArmed() && btn.isEnabled();
        boolean hovered = model.isRollover() && btn.isEnabled();
        boolean enabled = btn.isEnabled();

        int drawX = b.x;
        int drawY = b.y + (pressed ? 2 : 0);
        int w = b.width;
        int h = b.height;

        Graphics2D bg = (Graphics2D) g2.create();
        if (!enabled) {
            bg.setComposite(AlphaComposite.getInstance(AlphaComposite.SRC_OVER, 0.38f));
            bg.drawImage(sprite, drawX, drawY, w, h, null);
        } else if (hovered) {
            bg.drawImage(sprite, drawX, drawY, w, h, null);
            bg.setComposite(AlphaComposite.getInstance(AlphaComposite.SRC_ATOP, 0.14f));
            bg.setColor(Color.WHITE);
            int corner = Math.max(12, (int) Math.round(24 * (h / 48.0)));
            bg.fillRoundRect(drawX, drawY, w, h, corner, corner);
        } else {
            bg.drawImage(sprite, drawX, drawY, w, h, null);
        }
        bg.dispose();
    }

        if (isTransitioning) return;
        mousePoint = e.getPoint();
        Landmark previouslyHovered = hovered;
        hovered = null;

        // Foreground buildings (0, 1, 2) have priority over background Mount Apo (3)
        for (int i = 0; i < Math.min(3, landmarks.size()); i++) {
            Landmark lm = landmarks.get(i);
            if (lm.contains(mousePoint)) {
                hovered = lm;
                break;
            }
        }
        if (hovered == null && landmarks.size() > 3) {
            Landmark apo = landmarks.get(3);
            if (apo.contains(mousePoint)) {
                hovered = apo;
            }
        }

        boolean overBack = backButtonBounds.contains(mousePoint);
        boolean overSettings = settingsButtonBounds.contains(mousePoint);
        boolean overFightGudo = fightGudoButtonBounds.width > 0 && fightGudoButtonBounds.contains(mousePoint);
        if (overBack || overSettings || overFightGudo) {
            hovered = null;
        }

        setCursor((hovered != null || overBack || overSettings || overFightGudo) ? Cursor.getPredefinedCursor(Cursor.HAND_CURSOR) : Cursor.getDefaultCursor());

        if (hovered != previouslyHovered) {
            repaint();
        }
    }

    private void onMouseClicked(MouseEvent e) {
        if (isTransitioning) return;
        if (fightGudoButtonBounds.width > 0 && fightGudoButtonBounds.contains(e.getPoint())) {
            launchBattle(4, 4);
            return;
        }
        if (backButtonBounds.contains(e.getPoint())) {
            stop();
            host.showScreen(Screen.START);
            return;
        }
        if (settingsButtonBounds.contains(e.getPoint())) {
            stop();
            host.showScreen(Screen.START);
            return;
        }

        Landmark clicked = null;
        // Foreground buildings (0, 1, 2) have priority over background Mount Apo (3)
        for (int i = 0; i < Math.min(3, landmarks.size()); i++) {
            Landmark lm = landmarks.get(i);
            if (lm.contains(e.getPoint())) {
                clicked = lm;
                break;
            }
        }
        if (clicked == null && landmarks.size() > 3) {
            Landmark apo = landmarks.get(3);
            if (apo.contains(e.getPoint())) {
                clicked = apo;
            }
        }

        if (clicked != null) {
            Player p = host.getActivePlayer();
            int unlocked = (p != null) ? p.getUnlockedMap() : 1;
            if (clicked.getMapIndex() <= unlocked) {
                launchBattle(clicked.getMapIndex());
            } else {
                String prevLandmark = switch (clicked.getMapIndex()) {
                    case 2 -> "Ateneo Blue Building (Sayon)";
                    case 3 -> "Davao Museum (Sakto)";
                    case 4 -> "San Pedro Cathedral (Lisod)";
                    default -> "naunang mapa";
                };
                String diffName = DifficultySetting.forMap(clicked.getMapIndex()).getDisplayName();
                String msg = "🔒 NAKA-LOCK PA KINI NGA MAPA!\n\n"
                    + "Kinahanglan nimo pildihon una ang " + prevLandmark + "\n"
                    + "aron maablihan ang bag-ong mapa ug ang predefined difficulty niini (" + diffName + ").\n\n"
                    + "Padayon sa pagduwa ug pagkat-on, Juan!";
                JOptionPane.showMessageDialog(this, msg, "Nakasera ang Mapa", JOptionPane.WARNING_MESSAGE);
            }
        }
    }

    private void launchBattle(int mapIndex) {
        launchBattle(mapIndex, 1);
    }

    private void launchBattle(int mapIndex, int stageIndex) {
        stop();
        Player p = host.getActivePlayer();
        BattleSession session = new BattleSessionBuilder()

            .onMap(mapIndex)

    private void tick(ActionEvent e) {
        if (isTransitioning) {
            transitionProgress += 0.035f;
            if (transitionProgress >= 1.0f) {
                transitionProgress = 1.0f;
                isTransitioning = false;
            }
        } else {
            for (Landmark lm : landmarks) {
                lm.updateScale(lm == hovered);
            }
        }
        repaint();
    }

            g2.translate(slideOffset, 0);
        } else {
            for (int i = 0; i < Math.min(3, landmarks.size()); i++) landmarks.get(i).paint(g2);

package juanword.ui;

import java.awt.AlphaComposite;
import java.awt.Color;
import java.awt.Font;
import java.awt.Graphics2D;
import java.awt.RenderingHints;
import java.awt.image.BufferedImage;
import java.io.File;
import java.io.InputStream;
import java.nio.file.Path;
import java.nio.file.Paths;
import java.util.Map;
import java.util.concurrent.ConcurrentHashMap;
import javax.imageio.ImageIO;

public final class ImageLoader {
    private static final Map<String, BufferedImage> CACHE = new ConcurrentHashMap<>();

    private ImageLoader() {}

    public static BufferedImage load(String path) {
        if (path == null || path.isBlank()) {
            return createPlaceholder("Image");
        }
        BufferedImage cached = CACHE.get(path);
        if (cached != null) {
            return cached;
        }

        String[] searchPaths = new String[] {
            path,
            "assets/" + path,
            "assets/sprites/" + path,
            "assets/maps/" + path,
            "assets/ui/" + path,
            "assets/letters/" + path,
            new File(path).getName(),
            "assets/sprites/" + new File(path).getName(),
            "assets/maps/" + new File(path).getName(),
            "assets/ui/" + new File(path).getName(),
            "assets/letters/" + new File(path).getName()
        };

        for (String p : searchPaths) {
            File f = new File(p);
            if (f.exists() && f.isFile()) {
                try {
                    BufferedImage img = ImageIO.read(f);
                    if (img != null) {
                        CACHE.put(path, img);
                        return img;
                    }
                } catch (Exception ignored) {}
            }
        }

        // Try classpath resource
        String resPath = path.startsWith("/") ? path : "/" + path;
        try (InputStream in = ImageLoader.class.getResourceAsStream(resPath)) {
            if (in != null) {
                BufferedImage img = ImageIO.read(in);
                if (img != null) {
                    CACHE.put(path, img);
                    return img;
                }
            }
        } catch (Exception ignored) {}

        String altRes = "/assets/" + (path.startsWith("/") ? path.substring(1) : path);
        try (InputStream in = ImageLoader.class.getResourceAsStream(altRes)) {
            if (in != null) {
                BufferedImage img = ImageIO.read(in);
                if (img != null) {
                    CACHE.put(path, img);
                    return img;
                }
            }
        } catch (Exception ignored) {}

        BufferedImage placeholder = createPlaceholder(new File(path).getName());
        CACHE.put(path, placeholder);
        return placeholder;
    }

    public static BufferedImage loadLetterTile(char letter) {
        char lower = Character.toLowerCase(letter);
        char upper = Character.toUpperCase(letter);
        BufferedImage img = load("letters/" + lower + ".png");
        if (img == null || isPlaceholder(img)) {
            img = load("letters/" + upper + ".png");
        }
        if (img == null || isPlaceholder(img)) {
            int num = (upper - 'A') + 1;
            img = load("letters/" + num + ".png");
        }
        return (img != null && !isPlaceholder(img)) ? img : null;
    }

    public static boolean isPlaceholder(BufferedImage img) {
        return img != null && img.getWidth() == 120 && img.getHeight() == 120;
    }

    public static BufferedImage createSilhouette(BufferedImage src, Color tint) {
        if (src == null) return null;
        BufferedImage silhouette = new BufferedImage(src.getWidth(), src.getHeight(), BufferedImage.TYPE_INT_ARGB);
        Graphics2D g2 = silhouette.createGraphics();
        g2.drawImage(src, 0, 0, null);
        g2.setComposite(AlphaComposite.SrcIn);
        g2.setColor(tint != null ? tint : new Color(18, 20, 28, 225));
        g2.fillRect(0, 0, src.getWidth(), src.getHeight());
        g2.dispose();
        return silhouette;
    }

    public static BufferedImage scale(BufferedImage src, int w, int h) {
        if (src == null || w <= 0 || h <= 0) return src;
        BufferedImage scaled = new BufferedImage(w, h, BufferedImage.TYPE_INT_ARGB);
        Graphics2D g2 = scaled.createGraphics();
        g2.setRenderingHint(RenderingHints.KEY_INTERPOLATION, RenderingHints.VALUE_INTERPOLATION_BILINEAR);
        g2.setRenderingHint(RenderingHints.KEY_RENDERING, RenderingHints.VALUE_RENDER_QUALITY);
        g2.drawImage(src, 0, 0, w, h, null);
        g2.dispose();
        return scaled;
    }

    private static BufferedImage createPlaceholder(String label) {
        int w = 120;
        int h = 120;
        BufferedImage img = new BufferedImage(w, h, BufferedImage.TYPE_INT_ARGB);
        Graphics2D g2 = img.createGraphics();
        g2.setRenderingHint(RenderingHints.KEY_ANTIALIASING, RenderingHints.VALUE_ANTIALIAS_ON);
        g2.setColor(new Color(40, 50, 70, 200));
        g2.fillRoundRect(4, 4, w - 8, h - 8, 16, 16);
        g2.setColor(new Color(200, 210, 230));
        g2.drawRoundRect(4, 4, w - 8, h - 8, 16, 16);
        g2.setFont(new Font(Font.SANS_SERIF, Font.BOLD, 11));
        g2.drawString(label != null && label.length() > 14 ? label.substring(0, 14) : (label != null ? label : "?"), 10, h / 2);
        g2.dispose();
        return img;
    }
}

package juanword.tile;

import java.util.List;
import java.util.Random;
import juanword.data.DataManager;

public final class TileGridGenerator {
    private static final Random RNG = new Random();

    private TileGridGenerator() {}

    public static TileComponent[][] generateFresh4x4() {
        TileComponent[][] grid = new TileComponent[4][4];
        for (int r = 0; r < 4; r++) {
            for (int c = 0; c < 4; c++) {
                grid[r][c] = generateTile();
            }
        }
        return grid;
    }

    public static TileComponent generateTile() {
        char letter = DataManager.getInstance().drawRandomLetter();
        TileComponent base = new StandardTileComponent(letter);
        int roll = RNG.nextInt(100);
        if (roll < 78) {
            return base;
        } else if (roll < 90) {
            return new GoldTileDecorator(base);
        } else {
            return new HealingTileDecorator(base);
        }
    }

    public static TileComponent applyModifier(TileComponent base, String code) {
        if (base == null) return null;
        if (code == null) return base;
        String trimmed = code.trim().toUpperCase();
        if ("GLD".equals(trimmed)) {
            return new GoldTileDecorator(base);
        } else if ("HEL".equals(trimmed)) {
            return new HealingTileDecorator(base);
        }
        return base;
    }

    public static void refill(TileComponent[][] grid, List<int[]> positions) {
        if (grid == null || positions == null) return;
        for (int[] pos : positions) {
            if (pos != null && pos.length >= 2) {
                int r = pos[0];
                int c = pos[1];
                if (r >= 0 && r < grid.length && c >= 0 && c < grid[r].length) {
                    grid[r][c] = generateTile();
                }
            }
        }
    }

package juanword.enemy;

import juanword.data.DataManager;
import juanword.data.EnemySpec;
import juanword.model.AttackType;
import juanword.model.BattleContext;
import juanword.model.DifficultySetting;

public abstract class Enemy {
    protected String name;
    protected String enemyId;
    protected DifficultySetting difficulty;
    protected int maxHp;
    protected int currentHp;
    protected int attack;
    protected boolean boss;
    protected String spritePath;
    protected AttackType attackType = AttackType.PUNCH;
    protected int turnCount = 0;

    protected Enemy(String name, DifficultySetting diff) {
        this.name = name;
        this.enemyId = name.toUpperCase().replace(' ', '_');
        this.difficulty = diff != null ? diff : DifficultySetting.SAYON;

        EnemySpec spec = DataManager.getInstance().getEnemySpecById(this.enemyId);
        double scale = this.difficulty.getStatScale();
        if (spec != null) {
            this.maxHp = Math.max(1, (int) Math.round(spec.baseHp() * scale)); //easy 1.0x, medium 1.15x, hard 1.30x, extreme 1.50x
            this.attack = Math.max(1, (int) Math.round(spec.baseAttack() * scale));
            this.boss = spec.boss();
            this.spritePath = spec.spritePath();
        } else {
            this.maxHp = Math.max(1, (int) Math.round(50 * scale));
            this.attack = Math.max(1, (int) Math.round(8 * scale));
            this.boss = false;
            this.spritePath = "assets/sprites/" + this.enemyId.toLowerCase() + ".png";
        }
        this.currentHp = this.maxHp;
    }

    public String getName() { return name; }
    public String getEnemyId() { return enemyId; }
    public String getSpritePath() { return spritePath; }
    public String getAttackSpritePath() {
        if ("GUDO".equalsIgnoreCase(enemyId)) {
            return "assets/sprites/gudoattack.png";
        }
        return null;
    }
    public int getMaxHp() { return maxHp; }
    public int getCurrentHp() { return currentHp; }
    public int getAttack() { return attack; }
    public boolean isBoss() { return boss; }
    public AttackType getAttackType() { return attackType; }
    public int getTurnCount() { return turnCount; }

    public void setCurrentHp(int hp) {
        this.currentHp = Math.clamp(hp, 0, maxHp);
    }

    public void takeDamage(int amount) {
        if (amount > 0) {
            this.currentHp = Math.max(0, this.currentHp - amount);
        }
    }

    public void heal(int amount) {
        if (amount > 0) {
            this.currentHp = Math.min(this.maxHp, this.currentHp + amount);
        }
    }

    public boolean isDefeated() {
        return currentHp <= 0;
    }

    public double getHpFraction() {
        if (maxHp <= 0) return 0.0;
        return (double) currentHp / maxHp;
    }

    public void onEnemyTurn(BattleContext ctx) {
        turnCount++;
        if (ctx != null) {
            ctx.damagePlayer(attack);
        }
    }
}

        return this;

            throw new NullPointerException("player profile required");
        }
        Enemy enemy = EnemyFactory.createEnemy(mapIndex, stageIndex, difficulty);
        if (savedEnemyHp != null) {
            enemy.setCurrentHp(Math.min(savedEnemyHp, enemy.getMaxHp()));
        }

        int hp = (savedPlayerHp != null) ? savedPlayerHp : player.getMaxHp();
        int timer = (savedTurnSeconds >= 0) ? savedTurnSeconds : difficulty.getTurnTimeSeconds();
        return new BattleSession(player, mapIndex, stageIndex, difficulty, enemy, grid, hp, timer);
    }

                        if (!rawWord.equals("WORD") && !rawWord.isEmpty()) {
                            String clean = rawWord.replaceAll("[^A-Z]", "");

    private void loadLetterSpecs() {
        letterSpecs.clear();
        weightedLetterPool.clear();
        Path letterFile = dataDir.resolve("letter_values.csv");
        boolean loaded = false;
        if (Files.exists(letterFile)) {
            try {
                List<String[]> rows = CsvUtil.readAll(letterFile);
                for (String[] row : rows) {
                    if (row.length >= 3 && !row[0].equalsIgnoreCase("letter")) {
                        String letStr = row[0].trim().toUpperCase();
                        if (!letStr.isEmpty()) {
                            char letter = letStr.charAt(0);
                            int points = Integer.parseInt(row[1].trim());
                            int weight = Integer.parseInt(row[2].trim());
                            letterSpecs.put(letter, new LetterSpec(letter, points, weight));
                        }
                    }
                }
                if (!letterSpecs.isEmpty()) {
                    buildWeightedPool();
                    loaded = true;
                }
            } catch (Exception ignored) {}
        }

        if (!loaded) {
            String fallback = "AAAAAAIIIIIUUUUBDGGHKKLLLMNNNPRSSTTWY";
            for (char c : fallback.toCharArray()) {
                weightedLetterPool.add(c);
                letterSpecs.putIfAbsent(c, new LetterSpec(c, 1, 1));

                if (sb.length() > 0) sb.append(';');
                if (r < grid.length && c < grid[r].length && grid[r][c] != null) {
                    sb.append(grid[r][c].getLetter()).append(':').append(grid[r][c].getModifierCode());
                } else {

    public TileComponent[][] deserializeGrid(String text) {
        TileComponent[][] grid = new TileComponent[4][4];
        if (text == null || text.isBlank()) {
            return TileGridGenerator.generateFresh4x4();
        }
        String[] tokens = text.split(";");
        int idx = 0;
        for (int r = 0; r < 4; r++) {
            for (int c = 0; c < 4; c++) {
                if (idx < tokens.length) {
                    grid[r][c] = deserializeTile(tokens[idx++]);
                } else {
                    grid[r][c] = TileGridGenerator.generateTile();
                }
            }
        }
        return grid;
    }

    public TileComponent deserializeTile(String token) {
        if (token == null || token.isBlank()) {
            return TileGridGenerator.generateTile();
        }
        String[] parts = token.split(":");
        char letter = parts[0].trim().isEmpty() ? 'A' : parts[0].trim().toUpperCase().charAt(0);
        TileComponent base = new StandardTileComponent(letter);
        if (parts.length > 1) {
            return TileGridGenerator.applyModifier(base, parts[1].trim());
        }
        return base;
    }

        return rows;
    }

    public static String[] parseLine(String line) {
        List<String> tokens = new ArrayList<>();
        StringBuilder sb = new StringBuilder();
        boolean inQuotes = false;
        for (int i = 0; i < line.length(); i++) {
            char c = line.charAt(i);
            if (c == '"') {
                if (inQuotes && i + 1 < line.length() && line.charAt(i + 1) == '"') {
                    sb.append('"');
                    i++;
                } else {
                    inQuotes = !inQuotes;
                }
            } else if (c == ',' && !inQuotes) {
                tokens.add(sb.toString().trim());
                sb.setLength(0);
            } else {
                sb.append(c);
            }
        }
        tokens.add(sb.toString().trim());
        return tokens.toArray(new String[0]);
    }

    public static String escape(String value) {
        if (value == null) {
            return "";
        }
        if (value.contains(",") || value.contains("\"") || value.contains("\n") || value.contains("\r")) {
            return "\"" + value.replace("\"", "\"\"") + "\"";
        }
        return value;
    }

    public static void writeAtomic(Path file, List<String[]> rows) throws IOException {
        Path parent = file.getParent();
        if (parent != null && !Files.exists(parent)) {
            Files.createDirectories(parent);
        }
        Path tmp = file.resolveSibling(file.getFileName().toString() + ".tmp");
        try (BufferedWriter writer = Files.newBufferedWriter(tmp, UTF8)) {
            for (String[] row : rows) {
                StringBuilder sb = new StringBuilder();
                for (int i = 0; i < row.length; i++) {
                    if (i > 0) sb.append(',');
                    sb.append(escape(row[i]));
                }
                writer.write(sb.toString());
                writer.newLine();
            }
        }
        try {
            Files.move(tmp, file, StandardCopyOption.REPLACE_EXISTING, StandardCopyOption.ATOMIC_MOVE);
        } catch (AtomicMoveNotSupportedException e) {
            Files.move(tmp, file, StandardCopyOption.REPLACE_EXISTING);
        }
    }
