    public AlviolaEnemy() {

    }

    @Override
    public void render(Graphics2D g2, int x, int y, int w, int h, boolean isHurt, int lungeOffset) {
        // Body: Solid Cerulean (flashes Crimson Red when hurt)
        g2.setColor(isHurt ? GamePalette.CRIMSON_RED : GamePalette.CERULEAN);
        g2.fillRoundRect(x, y, w, h, 12, 12);

        // Steel plate cross-brace X in Dusk Navy
        g2.setColor(GamePalette.DUSK_NAVY);
        g2.setStroke(new BasicStroke(2.5f));
        g2.drawLine(x + 6, y + 6, x + w - 6, y + h - 6);
        g2.drawLine(x + w - 6, y + 6, x + 6, y + h - 6);

        // Steel Corner Reinforcements: Midnight Navy
        g2.setColor(GamePalette.MIDNIGHT_NAVY);
        g2.setStroke(new BasicStroke(3.5f));
        g2.drawRoundRect(x, y, w, h, 12, 12);

        // Heavy Steel Hex Bolts: Cream
        g2.setColor(GamePalette.CREAM);
        g2.fillOval(x + 4, y + 4, 6, 6);
        g2.fillOval(x + w - 10, y + 4, 6, 6);
        g2.fillOval(x + 4, y + h - 10, 6, 6);
        g2.fillOval(x + w - 10, y + h - 10, 6, 6);

        // Expressive Eyes: Sharp, focused angular eyes
        if (isHurt) {
            g2.setColor(GamePalette.CREAM);
            g2.setFont(new Font("Arial", Font.BOLD, 16));
            g2.drawString(">  <", x + (w / 2) - 14, y + 26);
        } else {
            g2.setColor(GamePalette.CREAM);
            g2.fillOval(x + 26, y + 15, 15, 13);
            g2.fillOval(x + 68, y + 15, 15, 13);
            g2.setColor(GamePalette.MIDNIGHT_NAVY);
            g2.fillOval(x + 31, y + 17, 7, 7);
            g2.fillOval(x + 71, y + 17, 7, 7);
            // Fierce diagonal steel brows
            g2.setColor(GamePalette.MIDNIGHT_NAVY);
            g2.setStroke(new BasicStroke(2.5f));
            g2.drawLine(x + 24, y + 11, x + 42, y + 16);
            g2.drawLine(x + 85, y + 11, x + 67, y + 16);
        }

        // Nameplate Plaque: ALVIOLA
        int plateW = 82;
        int plateH = 22;
        int plateX = x + (w - plateW) / 2;
        int plateY = y + 46;

        g2.setColor(GamePalette.DUSK_NAVY);
        g2.fillRoundRect(plateX, plateY, plateW, plateH, 6, 6);
        g2.setColor(GamePalette.SKY_BLUE);
        g2.setStroke(new BasicStroke(1.5f));
        g2.drawRoundRect(plateX, plateY, plateW, plateH, 6, 6);

        g2.setColor(GamePalette.CREAM);
        g2.setFont(new Font("Arial", Font.BOLD, 12));
        String nameStr = "ALVIOLA";
        int tw = g2.getFontMetrics().stringWidth(nameStr);
        g2.drawString(nameStr, plateX + (plateW - tw) / 2, plateY + 16);
    }
}

    public enum DifficultySetting {
        SAYON("Sayon", GamePalette.FOREST_GREEN, 50, 6, 1.0, 1.0, false, 0),
        SAKTO("Sakto", GamePalette.PUMPKIN_ORANGE, 70, 10, 1.5, 1.5, false, 12),
        LISOD("Lisod", GamePalette.CRIMSON_RED, 110, 15, 2.5, 2.5, true, 25),
        EXTREME("Extreme", new Color(160, 32, 240), 160, 22, 4.0, 4.0, true, 20);

        public final String label;
        public final Color badgeColor;
        public final int enemyHp;
        public final int enemyAtk;
        public final double rewardMultiplier;
        public final double scoreMultiplier;
        public final boolean hasTimer;
        public final int turnLimitOrSeconds;

        DifficultySetting(String label, Color badgeColor, int enemyHp, int enemyAtk,
                          double rewardMultiplier, double scoreMultiplier,
                          boolean hasTimer, int turnLimitOrSeconds) {
            this.label = label;
            this.badgeColor = badgeColor;
            this.enemyHp = enemyHp;
            this.enemyAtk = enemyAtk;
            this.rewardMultiplier = rewardMultiplier;
            this.scoreMultiplier = scoreMultiplier;
            this.hasTimer = hasTimer;
            this.turnLimitOrSeconds = turnLimitOrSeconds;
        }

        public static DifficultySetting fromString(String name) {
            if ("Extreme".equalsIgnoreCase(name) || "Kinalisoran".equalsIgnoreCase(name)) return EXTREME;
            if ("Lisod".equalsIgnoreCase(name) || "Hard".equalsIgnoreCase(name)) return LISOD;
            if ("Sakto".equalsIgnoreCase(name) || "Medium".equalsIgnoreCase(name)) return SAKTO;
            return SAYON;
        }

        /**
         * DIAL 1: Word length damage multiplier.
         * Harder difficulties make shorter letter words worth less!
         */
        public double getLengthMultiplier(int length) {
            switch (this) {
                case EXTREME:
                    if (length <= 2) return 0.10;
                    if (length == 3) return 0.25;
                    if (length == 4) return 0.70;
                    return 1.25;
                case LISOD:
                    // Hard: Shorter words are heavily penalized!
                    if (length <= 2) return 0.20;
                    if (length == 3) return 0.35; // 3-letter words only 35% damage!
                    if (length == 4) return 0.85; // 4-letter words 85% damage
                    if (length == 5) return 1.50; // 5-letter words 150%
                    if (length == 6) return 2.20; // 6-letter words 220%
                    return 3.00;                 // 7+ letter words 300%!

                case SAKTO:
                    // Medium: Shorter words are moderately weakened
                    if (length <= 2) return 0.40;
                    if (length == 3) return 0.65; // 3-letter words 65% damage
                    if (length == 4) return 1.00; // Baseline
                    if (length == 5) return 1.45;
                    if (length == 6) return 1.90;
                    return 2.40;

                case SAYON:
                default:
                    // Easy: Friendly scaling, 3-letter words deal full 100% damage
                    if (length <= 2) return 0.80;
                    if (length == 3) return 1.00;
                    if (length == 4) return 1.20;
                    if (length == 5) return 1.45;
                    return 1.80;
            }
        }
    }

    private void preloadLetterTileImages() {
        if (app == null) return;
        for (char c = 'A'; c <= 'Z'; c++) {
            Image raw = app.loadGameImage("letters/" + c + ".png");
            if (raw == null) raw = app.loadGameImage("letters/" + Character.toLowerCase(c) + ".png");
            if (raw == null) raw = app.loadGameImage("letters/" + ((c - 'A') + 1) + ".png");
            if (raw != null) {
                letterTileImages.put(c, toBufferedImage(raw));
            }
        }
    }

    public BufferedImage getLetterTileImage(char letter) {
        char upper = Character.toUpperCase(letter);
        BufferedImage img = letterTileImages.get(upper);
        if (img == null && app != null) {
            Image raw = app.loadGameImage("letters/" + upper + ".png");
            if (raw == null) raw = app.loadGameImage("letters/" + Character.toLowerCase(upper) + ".png");
            if (raw == null) raw = app.loadGameImage("letters/" + ((upper - 'A') + 1) + ".png");
            if (raw != null) {
                img = toBufferedImage(raw);
                letterTileImages.put(upper, img);
            }
        }
        return img;
    }

    private BufferedImage toBufferedImage(Image img) {
        if (img == null) return null;
        if (img instanceof BufferedImage) return (BufferedImage) img;
        int w = img.getWidth(null);
        int h = img.getHeight(null);
        if (w <= 0 || h <= 0) return null;
        BufferedImage bImg = new BufferedImage(w, h, BufferedImage.TYPE_INT_ARGB);
        Graphics2D g2 = bImg.createGraphics();
        g2.drawImage(img, 0, 0, null);
        g2.dispose();
        return bImg;
    }

    private void loadLetterData() {
        letterPoints = dm.loadLetterPoints();
        Map<Character, Integer> weights = dm.loadLetterWeights();
        weightedPool.clear();
        for (Map.Entry<Character, Integer> entry : weights.entrySet()) {
            char ch = entry.getKey();
            int w = entry.getValue();
            for (int i = 0; i < w; i++) {
                weightedPool.add(ch);
            }
        }
        if (weightedPool.isEmpty()) {
            String fallback = "AAAAAAIIIIIUUUUBDGGHKKLLLMNNNPRSSTTWY";
            for (char c : fallback.toCharArray()) {
                weightedPool.add(Character.toLowerCase(c));
            }
        }
    }

    private char getRandomLetter() {
        return weightedPool.get(rng.nextInt(weightedPool.size()));
    }

    // --- SETUP BATTLE FOR MAP & DIFFICULTY ---
    // PATTERN 3: BUILDER - Primary entry point accepting BattleSession constructed via BattleSessionBuilder
    public void setupBattle(BattleSession session) {
        if (session == null) {
            session = new BattleSessionBuilder().build();
        }
        this.currentPlayer = session.getPlayer();
        this.mapName = session.getMapName();
        this.stageNum = session.getStageNum();
        this.difficulty = session.getDifficulty();

        // Configure active enemy using Factory Method
        configureEnemyForStage(stageNum, difficulty);

        // Restore enemy HP if resuming an in-progress battle
        if (session.isResume() && currentPlayer != null && session.getSavedEnemyHealth() > 0 && session.getSavedEnemyHealth() <= enemyMaxHp) {
            this.enemyCurHp = session.getSavedEnemyHealth();
            this.dizonCurHp = enemyCurHp;
            if (currentEnemy != null) currentEnemy.setCurHp(enemyCurHp);
        }

        // Player HP
        this.juanMaxHp = (currentPlayer != null && currentPlayer.maxHealth > 0) ? currentPlayer.maxHealth : 100;
        if (currentPlayer != null && currentPlayer.currentHealth > 0) {
            this.juanCurHp = Math.min(currentPlayer.currentHealth, juanMaxHp);
        } else {
            this.juanCurHp = juanMaxHp;
        }

        // Dial 3: Pacing / Timers (timer runs only after title card is dismissed)
        if (turnTimer != null) turnTimer.stop();
        if (difficulty.hasTimer) {
            timerSeconds = difficulty.turnLimitOrSeconds; // 25s
            turnTimer = new javax.swing.Timer(1000, e -> onTimerTick());
        } else {
            turnsLeft = difficulty.turnLimitOrSeconds; // 12 turns for Sakto, 0 (infinite) for Sayon
        }

        updateHeaderInfo();
        updateCombatBars();
        updateJuanInfo();
        clearWordSelection();

        // Restore saved letter tiles if resuming, otherwise deal new random grid
        boolean tilesRestored = false;
        if (session.isResume() && session.getSavedTiles() != null && !session.getSavedTiles().isEmpty()) {
            tilesRestored = restoreTiles(session.getSavedTiles());
        }
        if (!tilesRestored) {
            dealNewGrid();
        }

        saveGameState(); // Persist current health & tiles

        if (session.isResume()) {
            battleLogLabel.setText("<html><center>▶ Gipadayon ang dula!<br>Ang imong kinabuhi (" + juanCurHp + " HP) ug mga letra napreserbar.</center></html>");
        } else {
            battleLogLabel.setText("Nagsugod na ang gubat! Pagpili ug mga letra gikan sa ubos para makatake.");
        }

        if (dizonBox != null) {
            dizonBox.setEnemy(currentEnemy);
            dizonBox.setHurt(false);
        }

        // Prominent Round Title Card for every round including Round 1!
        int roundNum = (currentPlayer != null && currentPlayer.currentRound > 0) ? currentPlayer.currentRound : 1;
        showRoundTitleCard(mapName, stageNum, roundNum, enemyName, enemySubtitle);
    }

    public void setupBattle(Player player, String map, int stage, String diffName) {
        setupBattle(player, map, stage, diffName, false);
    }

    public void setupBattle(Player player, String map, int stage, String diffName, boolean isResume) {
        BattleSession session = new BattleSessionBuilder()
                .player(player)
                .mapName(map)
                .stageNum(stage)
                .difficulty(diffName)
                .asResume(isResume)
                .savedTiles((player != null && player.savedTiles != null) ? player.savedTiles : "")
                .savedEnemyHealth((player != null) ? player.savedEnemyHealth : -1)
                .build();
        setupBattle(session);
    }

    public void resumeBattle(Player player) {
        String map = (player != null && player.savedMap != null && !player.savedMap.isEmpty())
                ? player.savedMap : "San Pedro Cathedral";
        int stage = (player != null && player.currentStage > 0) ? player.currentStage : 1;
        String diff = (player != null && player.savedDifficulty != null && !player.savedDifficulty.isEmpty())
                ? player.savedDifficulty : "Sakto";
        setupBattle(player, map, stage, diff, true);
    }

    public void showRoundTitleCard(String map, int stage, int round, String enemy, String subtitle) {
        if (titleCardOverlay != null && layeredPane != null) {
            layeredPane.moveToFront(titleCardOverlay);
            titleCardOverlay.showCard(map, stage, round, enemy, subtitle, () -> {
                // Callback when title card is dismissed:
                if (difficulty.hasTimer && turnTimer != null && !turnTimer.isRunning()) {
                    turnTimer.start();
                }
            });
        }
    }

    // PATTERN 4: DECORATOR - Helper to roll bonus decorated tiles (Gold & Healing)
    private TileComponent createRandomTileComponent(char c, int basePts) {
        TileComponent comp = new StandardTileComponent(c, basePts);
        int roll = rng.nextInt(100);
        if (roll < 12) {
            // 12% chance: Gold Tile Decorator (2x points)
            comp = new GoldTileDecorator(comp);
        } else if (roll < 22) {
            // 10% chance: Healing Tile Decorator (+8 HP to Juan on submit)
            comp = new HealingTileDecorator(comp);
        }
        return comp;
    }

    public String serializeTiles() {
        StringBuilder sb = new StringBuilder();
        for (int i = 0; i < 16; i++) {
            if (tiles[i] != null) {
                if (sb.length() > 0) sb.append(";");
                TileComponent comp = tiles[i].getTileComponent();
                String effect = (comp != null) ? comp.getEffectName() : "STANDARD";
                sb.append(Character.toUpperCase(tiles[i].getLetter()))
                  .append(":").append(tiles[i].getPoints())
                  .append(":").append(effect);
            }
        }
        return sb.toString();
    }

    public boolean restoreTiles(String serialized) {
        if (serialized == null || serialized.trim().isEmpty()) return false;
        String clean = serialized.trim();
        String[] parts = clean.split(";");
        if (parts.length == 16) {
            for (int i = 0; i < 16; i++) {
                String part = parts[i].trim();
                if (part.contains(":")) {
                    String[] lp = part.split(":");
                    char c = lp[0].charAt(0);
                    int pts = 1;
                    try { pts = Integer.parseInt(lp[1].trim()); } catch (Exception ignored) {}
                    String effect = (lp.length >= 3) ? lp[2].trim().toUpperCase() : "STANDARD";

                    TileComponent comp;
                    if ("GOLD".equals(effect)) {
                        int base = (pts > 1 && pts % 2 == 0) ? pts / 2 : letterPoints.getOrDefault(Character.toLowerCase(c), 1);
                        comp = new GoldTileDecorator(new StandardTileComponent(c, base));
                    } else if ("HEAL".equals(effect)) {
                        comp = new HealingTileDecorator(new StandardTileComponent(c, pts));
                    } else {
                        comp = new StandardTileComponent(c, pts);
                    }
                    tiles[i].setTileComponent(comp);
                } else if (!part.isEmpty()) {
                    char c = part.charAt(0);
                    int pts = letterPoints.getOrDefault(Character.toLowerCase(c), 1);
                    tiles[i].setTileComponent(new StandardTileComponent(c, pts));
                }
                tiles[i].setSelectedTile(false);
            }
            return true;
        } else if (clean.length() == 16 && !clean.contains(";")) {
            for (int i = 0; i < 16; i++) {
                char c = clean.charAt(i);
                int pts = letterPoints.getOrDefault(Character.toLowerCase(c), 1);
                tiles[i].setTileComponent(new StandardTileComponent(c, pts));
                tiles[i].setSelectedTile(false);
            }
            return true;
        }
        return false;
    }

    private void dealNewGrid() {
        for (int i = 0; i < 16; i++) {
            char c = getRandomLetter();
            int pts = letterPoints.getOrDefault(c, 1);
            tiles[i].setTileComponent(createRandomTileComponent(c, pts));
            tiles[i].setSelectedTile(false);
        }
        saveGameState();
    }

    private void onTileClicked(LetterTileButton tile) {
        if (isEnemyTurnActive) return;
        if (!tile.isSelectedTile()) {
            tile.setSelectedTile(true);
            selectedTiles.push(tile);

            // Pop up into middle screen Bookworm word rack!
            PoppedLetterTile pTile = new PoppedLetterTile(tile.getTileComponent(), tile);
            int popTileSz = (int) Math.round(48 * currentResponsiveScale);
            pTile.setPreferredSize(new Dimension(popTileSz, popTileSz));
            poppedTiles.add(pTile);
            if (poppedTilesContainer != null) {
                poppedTilesContainer.add(pTile);
                poppedTilesContainer.revalidate();
                poppedTilesContainer.repaint();
            }
        } else if (!selectedTiles.isEmpty() && selectedTiles.peek() == tile) {
            // Unselect top tile
            selectedTiles.pop();
            tile.setSelectedTile(false);
            if (!poppedTiles.isEmpty()) {
                PoppedLetterTile removed = poppedTiles.remove(poppedTiles.size() - 1);
                if (poppedTilesContainer != null) {
                    poppedTilesContainer.remove(removed);
                    poppedTilesContainer.revalidate();
                    poppedTilesContainer.repaint();
                }
            }
        }
        updateWordTray();
    }

    private void onPoppedTileClicked(PoppedLetterTile clicked) {
        if (isEnemyTurnActive) return;
        int targetIdx = poppedTiles.indexOf(clicked);
        if (targetIdx < 0) return;

        // Deselect and pop all tiles from the end of the word down to targetIdx
        while (poppedTiles.size() > targetIdx && !selectedTiles.isEmpty()) {
            PoppedLetterTile removedTile = poppedTiles.remove(poppedTiles.size() - 1);
            if (poppedTilesContainer != null) {
                poppedTilesContainer.remove(removedTile);
            }
            LetterTileButton topButton = selectedTiles.pop();
            topButton.setSelectedTile(false);
        }

        if (poppedTilesContainer != null) {
            poppedTilesContainer.revalidate();
            poppedTilesContainer.repaint();
        }
        updateWordTray();
    }

    private void backspaceTile() {
        if (isEnemyTurnActive) return;
        if (!selectedTiles.isEmpty()) {
            LetterTileButton top = selectedTiles.pop();
            top.setSelectedTile(false);
            if (!poppedTiles.isEmpty()) {
                PoppedLetterTile removed = poppedTiles.remove(poppedTiles.size() - 1);
                if (poppedTilesContainer != null) {
                    poppedTilesContainer.remove(removed);
                    poppedTilesContainer.revalidate();
                    poppedTilesContainer.repaint();
                }
            }
            updateWordTray();
        }
    }

    private void clearWordSelection() {
        if (isEnemyTurnActive) return;
        while (!selectedTiles.isEmpty()) {
            selectedTiles.pop().setSelectedTile(false);
        }
        poppedTiles.clear();
        if (poppedTilesContainer != null) {
            poppedTilesContainer.removeAll();
            poppedTilesContainer.revalidate();
            poppedTilesContainer.repaint();
        }
        updateWordTray();
    }

    private void shuffleGrid() {
        if (isEnemyTurnActive) return;
        clearWordSelection();
        dealNewGrid();
        battleLogLabel.setText("<html><center>🔄 Gibag-o ang mga letra sa board!<br>Mo-atake si " + enemyName + " samtang nag-shuffle ka!</center></html>");
        isEnemyTurnActive = true;
        if (attackButton != null) attackButton.setEnabled(false);
        javax.swing.Timer turnDelay = new javax.swing.Timer(300, e -> {
            ((javax.swing.Timer) e.getSource()).stop();
            enemyTurn();
        });
        turnDelay.setRepeats(false);
        turnDelay.start();
    }

    private String getSelectedWord() {
        StringBuilder sb = new StringBuilder();
        // Stack holds in reverse order of selection, so iterate backwards
        List<LetterTileButton> list = new ArrayList<>(selectedTiles);
        Collections.reverse(list);
        for (LetterTileButton b : list) {
            sb.append(b.getLetter());
        }
        return sb.toString().toLowerCase();
    }

    private void updateWordTray() {
        String word = getSelectedWord();
        boolean valid = !word.isEmpty() && dm.isValidWord(word);

        if (wordRack != null) {
            wordRack.setValidWord(valid);
            wordRack.repaint();
        }

        if (word.isEmpty()) {
            damagePreviewLabel.setText("Pili ug mga letra sa ubos para makaporma ug pulong");
            damagePreviewLabel.setForeground(GamePalette.SKY);
            if (dizonHealthBar != null) dizonHealthBar.setPreviewDamage(0, false);
            if (attackButton != null) attackButton.repaint();
            return;
        }

        int basePts = calculateBasePoints(word);
        double mult = difficulty.getLengthMultiplier(word.length());
        int damage = (int) Math.round(basePts * mult);
        if (damage < 1) damage = 1;

        if (valid) {
            damagePreviewLabel.setText("✓ Balido nga pulong! Andam na mo-atake.");
            damagePreviewLabel.setForeground(GamePalette.SPICY_SWEETCORN);
            if (dizonHealthBar != null) dizonHealthBar.setPreviewDamage(damage, true);
        } else {
            damagePreviewLabel.setText("✕ Dili pa balido nga pulong");
            damagePreviewLabel.setForeground(GamePalette.PUMPKIN_ORANGE);
            if (dizonHealthBar != null) dizonHealthBar.setPreviewDamage(0, false);
        }

        if (attackButton != null) attackButton.repaint();
    }

    private int calculateSelectedTilesPoints() {
        if (selectedTiles.isEmpty()) return 0;
        int sum = 0;
        for (LetterTileButton b : selectedTiles) {
            sum += b.getPoints();
        }
        return sum;
    }

    private int calculateBasePoints(String word) {
        if (!selectedTiles.isEmpty()) {
            return calculateSelectedTilesPoints();
        }
        int sum = 0;
        for (char c : word.toLowerCase().toCharArray()) {
            sum += letterPoints.getOrDefault(c, 1);
        }
        return sum;
    }

    // --- ATTACK EXECUTION ---
    private void performAttack() {
        if (isEnemyTurnActive) return;
        String word = getSelectedWord();
        if (word.length() < 2) {
            battleLogLabel.setText("<html><center>⚠️ Mubo ra kaayo ang pulong!<br>Pili ug labing menos 2 o 3 ka letra.</center></html>");
            return;
        }

        boolean valid = dm.isValidWord(word);
        if (!valid) {
            battleLogLabel.setText("<html><center>⚠️ Ang \"" + word.toUpperCase() + "\" dili makit-an sa Cebuano dictionary!</center></html>");
            if (difficulty == DifficultySetting.LISOD) {
                // On Lisod, fumble penalty applies with enemy counterattack animation!
                battleLogLabel.setText("<html><center>⚠️ Sayop nga pulong!<br>Nakahigayon si " + enemyName + " ug niigo!</center></html>");
                if (dizonBox != null) {
                    isEnemyTurnActive = true;
                    if (attackButton != null) attackButton.setEnabled(false);
                    dizonBox.triggerAttack(
                        () -> {
                            if (juanAvatarPanel != null) juanAvatarPanel.triggerHit();
                            applyDamageToJuan(8);
                        },
                        () -> {
                            isEnemyTurnActive = false;
                            if (attackButton != null) attackButton.setEnabled(true);
                        }
                    );
                } else {
                    applyDamageToJuan(8);
                }
            }
            return;
        }

        // PATTERN 4: DECORATOR - Trigger decorator callbacks on submitted tiles (e.g. Healing, Score bonus)
        for (LetterTileButton usedTile : selectedTiles) {
            TileComponent comp = usedTile.getTileComponent();
            if (comp != null) {
                comp.onWordSubmitted(currentPlayer, this);
            }
        }

        // Valid word attack!
        isEnemyTurnActive = true;
        if (attackButton != null) attackButton.setEnabled(false);

        int basePts = calculateBasePoints(word);
        double mult = difficulty.getLengthMultiplier(word.length());
        int damage = (int) Math.round(basePts * mult);
        if (damage < 1) damage = 1;

        final String dialNote = (word.length() >= 5) ? " (Kusog nga atake!)" : "";
        final int finalDamage = damage;
        final String submittedWord = word.toUpperCase();

        // Refill used tiles (Bookworm falling tile effect) with random decorators
        for (LetterTileButton usedTile : selectedTiles) {
            char newChar = getRandomLetter();
            int newPts = letterPoints.getOrDefault(newChar, 1);
            usedTile.setTileComponent(createRandomTileComponent(newChar, newPts));
            usedTile.setSelectedTile(false);
        }
        selectedTiles.clear();
        poppedTiles.clear();
        if (poppedTilesContainer != null) {
            poppedTilesContainer.removeAll();
            poppedTilesContainer.revalidate();
            poppedTilesContainer.repaint();
        }
        updateWordTray();

        battleLogLabel.setText("<html><center>⚔️ Giandam ni Juan ang atake gamit ang \"" + submittedWord + "\"!</center></html>");

        Runnable onStrikeImpact = () -> {
            // Hit active enemy at strike apex
            enemyCurHp = Math.max(0, enemyCurHp - finalDamage);
            dizonCurHp = enemyCurHp;
            if (currentEnemy != null) {
                currentEnemy.takeDamage(finalDamage);
            }
            updateCombatBars();
            if (dizonBox != null) dizonBox.triggerHit();
            saveGameState(); // Persist refilled letter tiles & enemy HP immediately
            battleLogLabel.setText("<html><center>🎯 Gisumiter ang \"" + submittedWord + "\"! Naigo si " + enemyName + "!" + dialNote + "</center></html>");
        };

        Runnable onAttackComplete = () -> {
            // Check Enemy Defeated / Victory
            if (enemyCurHp <= 0) {
                if (turnTimer != null) turnTimer.stop();
                battleLogLabel.setText("<html><center>💥 <b>GRABENG IGO!</b> Napildi si <b>" + enemyName + "</b>!</center></html>");

                if (dizonBox != null) {
                    dizonBox.triggerDeath(() -> {
                        handleEnemyDefeated();
                    });
                } else {
                    handleEnemyDefeated();
                }
                return;
            }

            // Enemy turn with delayed animation!
            // Pause 350ms so player can register the exchange before enemy counterattacks
            javax.swing.Timer turnDelay = new javax.swing.Timer(350, e -> {
                ((javax.swing.Timer) e.getSource()).stop();
                enemyTurn();
            });
            turnDelay.setRepeats(false);
            turnDelay.start();
        };

        if (juanAvatarPanel != null) {
            juanAvatarPanel.triggerAttack(onStrikeImpact, onAttackComplete);
        } else {
            onStrikeImpact.run();
            onAttackComplete.run();
        }
    }

    private void enemyTurn() {
        if (enemyCurHp <= 0) {
            isEnemyTurnActive = false;
            if (attackButton != null) attackButton.setEnabled(true);
            return;
        }

        // Reset timer if on Lisod
        if (difficulty.hasTimer) {
            timerSeconds = difficulty.turnLimitOrSeconds;
            pacingInfoLabel.setText("⏱️ Timer: " + timerSeconds + "s");
        } else if (difficulty == DifficultySetting.SAKTO) {
            turnsLeft--;
            pacingInfoLabel.setText("Turns Left: " + turnsLeft);
            if (turnsLeft <= 0) {
                battleLogLabel.setText("<html><center>⌛ Nahurot na ang mga turno!<br>Nasuko si " + enemyName + " ug niigo ug grabeng atake!</center></html>");
                if (dizonBox != null) {
                    dizonBox.triggerAttack(
                        () -> {
                            if (juanAvatarPanel != null) juanAvatarPanel.triggerHit();
                            applyDamageToJuan(999);
                        },
                        () -> {
                            isEnemyTurnActive = false;
                        }
                    );
                } else {
                    applyDamageToJuan(999);
                }
                return;
            }
        }

        // Announce enemy attack
        battleLogLabel.setText("<html><center>⚔️ Mo-atake na si " + enemyName + "...</center></html>");

        // Enemy strikes Juan with full lunge animation!
        if (dizonBox != null) {
            dizonBox.triggerAttack(
                () -> {
                    // On impact frame: Juan recoils and takes damage!
                    if (juanAvatarPanel != null) {
                        juanAvatarPanel.triggerHit();
                    }
                    applyDamageToJuan(enemyAtk);
                },
                () -> {
                    // On animation completion: Unlock player turn
                    isEnemyTurnActive = false;
                    if (attackButton != null) {
                        attackButton.setEnabled(true);
                    }
                }
            );
        } else {
            if (juanAvatarPanel != null) juanAvatarPanel.triggerHit();
            applyDamageToJuan(enemyAtk);
            isEnemyTurnActive = false;
            if (attackButton != null) attackButton.setEnabled(true);
        }
    }

    // PATTERN 4: DECORATOR - Healing Tile effect
    public void healJuan(int amount) {
        if (amount <= 0 || juanCurHp >= juanMaxHp) return;
        int oldHp = juanCurHp;
        juanCurHp = Math.min(juanMaxHp, juanCurHp + amount);
        int restored = juanCurHp - oldHp;
        updateCombatBars();
        saveHealth();
        battleLogLabel.setText("<html><center>💚 Naayo si Juan ug +" + restored + " HP gikan sa Healing Tile!</center></html>");
    }

    private void applyDamageToJuan(int dmg) {
        juanCurHp = Math.max(0, juanCurHp - dmg);
        updateCombatBars();
        saveHealth();
        if (juanCurHp <= 0) {
            // Short delay so player sees the impact before defeat dialog appears
            javax.swing.Timer defeatTimer = new javax.swing.Timer(350, e -> {
                ((javax.swing.Timer) e.getSource()).stop();
                handleDefeat();
            });
            defeatTimer.setRepeats(false);
            defeatTimer.start();
        } else {
            battleLogLabel.setText("<html><center>💥 Niigo si " + enemyName + " ug " + dmg + " damage kang Juan!</center></html>");
        }
    }

    // Active Timer Tick (for Lisod)
    private void onTimerTick() {
        if (enemyCurHp <= 0 || juanCurHp <= 0 || isEnemyTurnActive) return;
        timerSeconds--;
        pacingInfoLabel.setText("⏱️ Timer: " + timerSeconds + "s");
        if (timerSeconds <= 5) {
            pacingInfoLabel.setForeground(GamePalette.PUMPKIN_ORANGE);
        } else {
            pacingInfoLabel.setForeground(GamePalette.SKY);
        }

        if (timerSeconds <= 0) {
            timerSeconds = difficulty.turnLimitOrSeconds;
            battleLogLabel.setText("<html><center>⏰ Nahurot ang oras!<br>Mo-atake na si " + enemyName + "!</center></html>");
            if (dizonBox != null) {
                isEnemyTurnActive = true;
                if (attackButton != null) attackButton.setEnabled(false);
                dizonBox.triggerAttack(
                    () -> {
                        if (juanAvatarPanel != null) juanAvatarPanel.triggerHit();
                        applyDamageToJuan(enemyAtk);
                    },
                    () -> {
                        isEnemyTurnActive = false;
                        if (attackButton != null) attackButton.setEnabled(true);
                    }
                );
            } else {
                applyDamageToJuan(enemyAtk);
            }
        }
    }

    // --- ENEMY DEFEATED TRANSITION & NEXT ENEMY ENTRANCE ---
    private void handleEnemyDefeated() {
        if (turnTimer != null) turnTimer.stop();

        // Calculate rewards according to stage & difficulty
        int coinsEarned = (int) Math.round(20 * stageNum * difficulty.rewardMultiplier);
        int xpEarned = (int) Math.round(40 * stageNum * difficulty.rewardMultiplier);
        int scoreEarned = (int) Math.round((enemyMaxHp * 10) * difficulty.scoreMultiplier);

        if (currentPlayer != null) {
            currentPlayer.coins += coinsEarned;
            currentPlayer.experience += xpEarned;
            currentPlayer.score += scoreEarned;
            currentPlayer.currentRound++;
            currentPlayer.currentHealth = juanCurHp;

            // Unlock next stage / boss if cleared for this specific difficulty
            String diffLabel = (difficulty != null && difficulty.label != null) ? difficulty.label : "Sakto";
            if (stageNum == 1) {
                currentPlayer.recordStageUnlock(diffLabel, 2);
            } else if (stageNum == 2) {
                currentPlayer.recordStageUnlock(diffLabel, 3); // Unlocks Boss Rola for this difficulty!
            } else if (stageNum == 3) {
                currentPlayer.recordStageUnlock(diffLabel, 4); // Cleared Boss Rola for this difficulty!
            }

            currentPlayer.lastSaved = LocalDateTime.now().format(DateTimeFormatter.ofPattern("yyyy-MM-dd HH:mm"));
            dm.savePlayer(currentPlayer);
            dm.addHighScore(currentPlayer.playerName, currentPlayer.score, stageNum);
            dm.recordCreatureDefeated(currentPlayer.playerId, "E00" + stageNum);
        }

        updateJuanInfo();

        if (stageNum == 1) {
            // Stage 1 cleared: Advance to Stage 2 (Alviola) with slide-in entrance from right!
            advanceToNextEnemy(2, "ALVIOLA", "The Steel Box", 15);
        } else if (stageNum == 2) {
            // Stage 2 cleared: Advance to Stage 3 (Rola Boss) with slide-in entrance from right!
            advanceToNextEnemy(3, "ROLA", "THE BOSS", 22);
        } else {
            // Stage 3 Boss cleared: Map victory!
            handleVictory();
        }
    }

    private void advanceToNextEnemy(int nextStage, String nextEnemyName, String nextEnemyTitle, int nextAtk) {
        advanceToNextEnemy(nextStage, nextEnemyName, nextEnemyTitle, nextAtk, null);
    }

    private void advanceToNextEnemy(int nextStage, String nextEnemyName, String nextEnemyTitle, int nextAtk, Runnable onComplete) {
        this.stageNum = nextStage;
        if (currentPlayer != null) {
            currentPlayer.currentStage = nextStage;
            currentPlayer.currentRound = 1;
            currentPlayer.savedDifficulty = difficulty.label;
            currentPlayer.recordStageUnlock(difficulty.label, nextStage);
            dm.savePlayer(currentPlayer);
        }

        // Factory Method creates next enemy
        this.currentEnemy = EnemyFactory.createEnemy(nextStage, difficulty);
        this.enemyName = (currentEnemy != null) ? currentEnemy.getName() : nextEnemyName;
        this.enemyMaxHp = (currentEnemy != null) ? currentEnemy.getMaxHp() : 95;
        this.enemyCurHp = enemyMaxHp;
        this.enemyAtk = (currentEnemy != null) ? currentEnemy.getAtk() : nextAtk;
        this.dizonMaxHp = enemyMaxHp;
        this.dizonCurHp = enemyCurHp;

        // Update UI headers
        if (enemyTitleLabel != null) {
            enemyTitleLabel.setText("STAGE " + nextStage + ": " + enemyName + " (" + nextEnemyTitle + ")");
            if (nextStage == 3) {
                enemyTitleLabel.setForeground(GamePalette.HONEY_GOLD);
            } else {
                enemyTitleLabel.setForeground(GamePalette.SPICY_SWEETCORN);
            }
        }
        if (enemyIntentLabel != null) {
            enemyIntentLabel.setText("ATK: " + enemyAtk + " kada turno");
        }
        if (locationTitleLabel != null) {
            locationTitleLabel.setText((mapName != null ? mapName.toUpperCase() : "MAPA") + " - STAGE " + nextStage);
        }

        // Reset & animate enemy health bar
        if (dizonHealthBar != null) {
            dizonHealthBar.setHp(0, enemyMaxHp);
        }

        // Announce new enemy arrival in battle log
        battleLogLabel.setText("<html><center>⚠️ <b>BAG-ONG KAAWAY!</b> Niduol si <b>" + enemyName + "</b> gikan sa tuo!</center></html>");

        // Animate new enemy entering from the right side of the screen!
        if (dizonBox != null) {
            dizonBox.triggerEntrance(currentEnemy, () -> {
                // Enemy arrived and settled!
                animateHealthBarFill(enemyMaxHp);

                // Reset turns / timer
                if (difficulty.hasTimer) {
                    timerSeconds = difficulty.turnLimitOrSeconds;
                    pacingInfoLabel.setText("⏱️ Timer: " + timerSeconds + "s");
                    if (turnTimer != null) turnTimer.start();
                } else if (difficulty == DifficultySetting.SAKTO) {
                    turnsLeft = difficulty.turnLimitOrSeconds;
                    pacingInfoLabel.setText("Turns Left: " + turnsLeft);
                }

                battleLogLabel.setText("<html><center>⚔️ Andam na si <b>" + enemyName + "</b>! Imong turno, Juan!</center></html>");

                isEnemyTurnActive = false;
                if (attackButton != null) attackButton.setEnabled(true);
                saveGameState();
                if (onComplete != null) onComplete.run();
            });
        } else {
            if (dizonHealthBar != null) dizonHealthBar.setHp(enemyMaxHp, enemyMaxHp);
            isEnemyTurnActive = false;
            if (attackButton != null) attackButton.setEnabled(true);
            saveGameState();
            if (onComplete != null) onComplete.run();
        }
    }

    private void animateHealthBarFill(int targetHp) {
        if (dizonHealthBar == null) return;
        final int steps = 12;
        final int[] step = {0};
        javax.swing.Timer fillTimer = new javax.swing.Timer(25, null);
        fillTimer.addActionListener(e -> {
            step[0]++;
            int currentVal = (int) Math.round((double) targetHp * step[0] / steps);
            dizonHealthBar.setHp(currentVal, targetHp);
            if (step[0] >= steps) {
                fillTimer.stop();
                dizonHealthBar.setHp(targetHp, targetHp);
            }
        });
        fillTimer.start();
    }

    private void handleDefeat() {
        if (turnTimer != null) turnTimer.stop();

        String defMsg = "💀 NAPILDI SI JUAN! (DEFEAT)\n\n"
                + "Gipildi ka ni " + enemyName + " sa Stage " + stageNum + " (" + mapName + ").\n"
                + "Gusto ba nimo sulayan pag-usab (Retry)?";

        int opt = JOptionPane.showConfirmDialog(
            this,
            defMsg,
            "Napildi",
            JOptionPane.YES_NO_OPTION,
            JOptionPane.ERROR_MESSAGE
        );

        if (opt == JOptionPane.YES_OPTION) {
            // Restore full health on retry
            if (currentPlayer != null) {
                currentPlayer.currentHealth = juanMaxHp;
                dm.savePlayer(currentPlayer);
            }
            setupBattle(currentPlayer, mapName, stageNum, difficulty.label);
        } else {
            // Revive with full health when returning after defeat
            if (currentPlayer != null) {
                currentPlayer.currentHealth = juanMaxHp;
                dm.savePlayer(currentPlayer);
            }
            exitBattle();
        }
    }

    // --- POPPED-UP LETTER TILE (Bookworm Style) ---
    private class PoppedLetterTile extends JComponent {
        private final char letter;
        private final int points;
        private final TileComponent tileComponent;
        private final LetterTileButton sourceTile;
        private float scale = 1.45f;
        private int offsetY = -8;
        private javax.swing.Timer popTimer;
        private boolean isHovered = false;

        PoppedLetterTile(char letter, int points, LetterTileButton sourceTile) {
            this(sourceTile != null ? sourceTile.getTileComponent() : new StandardTileComponent(letter, points), sourceTile);
        }

        PoppedLetterTile(TileComponent component, LetterTileButton sourceTile) {
            this.tileComponent = (component != null) ? component : new StandardTileComponent('a', 1);
            this.letter = Character.toUpperCase(this.tileComponent.getLetter());
            this.points = this.tileComponent.getPoints();
            this.sourceTile = sourceTile;
            setPreferredSize(new Dimension(48, 48));
            setMinimumSize(new Dimension(48, 48));
            setOpaque(false);
            if (app.customCursor != null) setCursor(app.customCursor);
            setToolTipText("I-klik aron tangtangon kini nga letra (Click to remove)");

            // Pop-up animation timer: 60fps, 10 frames (~140ms)
            final long startTime = System.currentTimeMillis();
            final int duration = 140; // ms
            popTimer = new javax.swing.Timer(14, e -> {
                long elapsed = System.currentTimeMillis() - startTime;
                float t = Math.min(1.0f, (float) elapsed / duration);
                // Elastic bounce curve: starts at 1.45, dips to ~0.94, settles at 1.0
                scale = 1.0f + 0.45f * (1.0f - t) * (float) Math.cos(t * Math.PI * 1.5);
                offsetY = (int) (-8 * (1.0f - t));
                repaint();
                if (t >= 1.0f) {
                    scale = 1.0f;
                    offsetY = 0;
                    ((javax.swing.Timer) e.getSource()).stop();
                    repaint();
                }
            });
            popTimer.start();

            addMouseListener(new MouseAdapter() {
                @Override
                public void mouseEntered(MouseEvent e) {
                    isHovered = true;
                    repaint();
                }

                @Override
                public void mouseExited(MouseEvent e) {
                    isHovered = false;
                    repaint();
                }

                @Override
                public void mouseClicked(MouseEvent e) {
                    onPoppedTileClicked(PoppedLetterTile.this);
                }
            });
        }

        @Override
        protected void paintComponent(Graphics g) {
            Graphics2D g2 = (Graphics2D) g.create();
            g2.setRenderingHint(RenderingHints.KEY_ANTIALIASING, RenderingHints.VALUE_ANTIALIAS_ON);
            g2.setRenderingHint(RenderingHints.KEY_INTERPOLATION, RenderingHints.VALUE_INTERPOLATION_BILINEAR);

            int compW = getWidth();
            int compH = getHeight();
            int tileW = Math.max(32, Math.min(compW - 4, compH - 4));
            int tileH = tileW;

            // Apply scale and bounce offset centered on component
            g2.translate(compW / 2.0, compH / 2.0 + offsetY);
            g2.scale(scale, scale);
            g2.translate(-tileW / 2.0, -tileH / 2.0);

            int x = 0;
            int y = 0;

            // Drop shadow
            g2.setColor(new Color(0, 0, 0, 45));
            g2.fillRoundRect(x + 2, y + 4, tileW, tileH, 10, 10);

            // Tile Background
            String currentWord = getSelectedWord();
            boolean isValid = !currentWord.isEmpty() && dm.isValidWord(currentWord);

            BufferedImage tileImg = getLetterTileImage(letter);
            if (tileImg != null) {
                // Outer glow if valid word or hovered
                if (isValid) {
                    g2.setColor(new Color(245, 180, 50, 180));
                    g2.setStroke(new BasicStroke(3.0f));
                    g2.drawRoundRect(x - 1, y - 1, tileW + 2, tileH + 2, 10, 10);
                } else if (isHovered) {
                    g2.setColor(new Color(255, 100, 100, 180));
                    g2.setStroke(new BasicStroke(2.5f));
                    g2.drawRoundRect(x - 1, y - 1, tileW + 2, tileH + 2, 10, 10);
                }

                g2.drawImage(tileImg, x, y, tileW, tileH, null);

                if (isValid) {
                    Graphics2D gTint = (Graphics2D) g2.create();
                    gTint.setComposite(AlphaComposite.getInstance(AlphaComposite.SRC_ATOP, 0.18f));
                    gTint.setColor(new Color(255, 200, 50));
                    gTint.fillRoundRect(x, y, tileW, tileH, 8, 8);
                    gTint.dispose();

                    g2.setColor(new Color(240, 150, 30));
                    g2.setStroke(new BasicStroke(2.0f));
                    g2.drawRoundRect(x, y, tileW, tileH, 8, 8);
                } else if (isHovered) {
                    g2.setColor(new Color(255, 90, 90));
                    g2.setStroke(new BasicStroke(2.0f));
                    g2.drawRoundRect(x, y, tileW, tileH, 8, 8);
                }
            } else {
                // Procedural fallback
                if (isValid) {
                    g2.setColor(isHovered ? GamePalette.SPICY_SWEETCORN : GamePalette.HONEY_GOLD);
                } else {
                    g2.setColor(isHovered ? Color.WHITE : GamePalette.CREAM);
                }
                g2.fillRoundRect(x, y, tileW, tileH, 10, 10);

                if (isValid) {
                    g2.setColor(isHovered ? GamePalette.SPICY_SWEETCORN : GamePalette.PUMPKIN_ORANGE);
                    g2.setStroke(new BasicStroke(isHovered ? 2.2f : 1.8f));
                } else {
                    g2.setColor(isHovered ? GamePalette.PUMPKIN_ORANGE : GamePalette.SKY_BLUE);
                    g2.setStroke(new BasicStroke(isHovered ? 2.0f : 1.4f));
                }
                g2.drawRoundRect(x, y, tileW, tileH, 10, 10);

                // Main Letter (Centered)
                g2.setFont(new Font("Arial", Font.BOLD, 22));
                String str = String.valueOf(letter);
                FontMetrics fm = g2.getFontMetrics();
                int lx = x + (tileW - fm.stringWidth(str)) / 2;
                int ly = y + (tileH + fm.getAscent() - fm.getDescent()) / 2 - 1;
                g2.setColor(GamePalette.MIDNIGHT_NAVY);
                g2.drawString(str, lx, ly);

                // Points Value (Bottom-Right Subscript for procedural fallback)
                g2.setFont(new Font("Arial", Font.BOLD, 10));
                String ptStr = String.valueOf(points);
                FontMetrics pfm = g2.getFontMetrics();
                int pw = pfm.stringWidth(ptStr);
                int px = x + tileW - pw - 4;
                int py = y + tileH - 4;

                g2.setColor(new Color(255, 255, 255, 190));
                g2.fillRoundRect(px - 2, py - pfm.getAscent() + 1, pw + 4, pfm.getAscent() + 2, 4, 4);
                g2.setColor(new Color(210, 195, 175));
                g2.setStroke(new BasicStroke(1.0f));
                g2.drawRoundRect(px - 2, py - pfm.getAscent() + 1, pw + 4, pfm.getAscent() + 2, 4, 4);

                g2.setColor(isValid ? new Color(200, 80, 20) : GamePalette.MIDNIGHT_NAVY);
                g2.drawString(ptStr, px, py);
            }

            // PATTERN 4: DECORATOR - Render decorator badge on popped tile
            if (tileComponent != null) {
                tileComponent.renderDecoratorBadge(g2, x, y, tileW, tileH);
            }

            // Hover indicator: small "✕" on top right if hovered
            if (isHovered) {
                g2.setFont(new Font("Arial", Font.BOLD, 10));
                g2.setColor(GamePalette.PUMPKIN_ORANGE);
                g2.drawString("✕", x + tileW - 10, y + 11);
            }

            g2.dispose();
        }
    }

    // --- ENEMY HEALTH BAR (No numbers, highlights word damage chunk) ---
    private static class EnemyHealthBar extends JPanel {
        private int maxHp = 70;
        private int curHp = 70;
        private int previewDamage = 0;
        private boolean isValidWord = false;
        private float pulsePhase = 0f;
        private float pulseAlpha = 1.0f;
        private final javax.swing.Timer pulseTimer;

        EnemyHealthBar() {
            setOpaque(false);
            setPreferredSize(new Dimension(150, 22));
            setMinimumSize(new Dimension(130, 22));

            // Pulse timer for highlighted hit chunk
            pulseTimer = new javax.swing.Timer(35, e -> {
                if (isValidWord && previewDamage > 0) {
                    pulsePhase += 0.15f;
                    pulseAlpha = 0.82f + 0.18f * (float) Math.sin(pulsePhase);
                    repaint();
                } else if (pulseAlpha != 1.0f) {
                    pulseAlpha = 1.0f;
                    repaint();
                }
            });
            pulseTimer.start();
        }

        void setHp(int cur, int max) {
            this.curHp = Math.max(0, cur);
            this.maxHp = Math.max(1, max);
            repaint();
        }

        void setPreviewDamage(int dmg, boolean valid) {
            this.previewDamage = valid ? Math.max(0, dmg) : 0;
            this.isValidWord = valid;
            repaint();
        }

        @Override
        protected void paintComponent(Graphics g) {
            super.paintComponent(g);
            Graphics2D g2 = (Graphics2D) g.create();
            g2.setRenderingHint(RenderingHints.KEY_ANTIALIASING, RenderingHints.VALUE_ANTIALIAS_ON);

            int w = getWidth() - 4;
            int h = getHeight() - 4;
            int x = 2;
            int y = 2;
            int arc = 8;

            // 1. Drop shadow
            g2.setColor(new Color(0, 0, 0, 35));
            g2.fillRoundRect(x + 1, y + 2, w, h, arc, arc);

            // 2. Empty/Depleted Bar Background (Solid Midnight Navy)
            g2.setColor(GamePalette.MIDNIGHT_NAVY);
            g2.fillRoundRect(x, y, w, h, arc, arc);

            // Calculate health widths
            double curRatio = Math.max(0.0, Math.min(1.0, (double) curHp / maxHp));
            int curWidth = (int) Math.round(w * curRatio);

            if (curWidth > 0) {
                int dmgWidth = 0;
                int remainingWidth = curWidth;

                if (isValidWord && previewDamage > 0) {
                    double dmgRatio = Math.min(curRatio, (double) previewDamage / maxHp);
                    dmgWidth = (int) Math.round(w * dmgRatio);
                    remainingWidth = Math.max(0, curWidth - dmgWidth);
                }

                Shape clipBackup = g2.getClip();
                // Clip to health bar shape so rounded corners remain crisp
                g2.clip(new java.awt.geom.RoundRectangle2D.Float(x, y, w, h, arc, arc));

                // 3. Remaining Health (Solid Pumpkin Orange)
                if (remainingWidth > 0) {
                    g2.setColor(GamePalette.PUMPKIN_ORANGE);
                    g2.fillRect(x, y, remainingWidth, h);
                }

                // 4. Highlighted Hit Chunk (Solid with pulse alpha)
                if (dmgWidth > 0) {
                    int dmgX = x + remainingWidth;
                    boolean isLethal = previewDamage >= curHp;

                    int alphaInt = Math.max(0, Math.min(255, (int) (255 * pulseAlpha)));

                    if (isLethal) {
                        // Solid lethal Honey Gold with pulse alpha
                        Color lethalCol = new Color(GamePalette.HONEY_GOLD.getRed(),
                                                    GamePalette.HONEY_GOLD.getGreen(),
                                                    GamePalette.HONEY_GOLD.getBlue(),
                                                    alphaInt);
                        g2.setColor(lethalCol);
                        g2.fillRect(dmgX, y, dmgWidth, h);

                        // Subtle KO text in Midnight Navy
                        g2.setFont(new Font("Arial", Font.BOLD, 10));
                        String ko = "☠️ KO";
                        FontMetrics fm = g2.getFontMetrics();
                        int kx = dmgX + (dmgWidth - fm.stringWidth(ko)) / 2;
                        g2.setColor(GamePalette.MIDNIGHT_NAVY);
                        g2.drawString(ko, kx, y + (h / 2) + 3);
                    } else {
                        // Solid highlighted Spicy Sweetcorn hit chunk with pulse alpha
                        Color hitCol = new Color(GamePalette.SPICY_SWEETCORN.getRed(),
                                                 GamePalette.SPICY_SWEETCORN.getGreen(),
                                                 GamePalette.SPICY_SWEETCORN.getBlue(),
                                                 alphaInt);
                        g2.setColor(hitCol);
                        g2.fillRect(dmgX, y, dmgWidth, h);

                        // Vertical divider notch separating remaining health from damage chunk
                        g2.setColor(GamePalette.CREAM);
                        g2.setStroke(new BasicStroke(1.5f));
                        g2.drawLine(dmgX, y, dmgX, y + h);
                    }
                }

                g2.setClip(clipBackup);
            }

            // 5. Outer Metallic Frame (Glows Spicy Sweetcorn if damage chunk is active, else Dusk Navy)
            if (isValidWord && previewDamage > 0) {
                g2.setColor(GamePalette.SPICY_SWEETCORN);
                g2.setStroke(new BasicStroke(1.8f));
            } else {
                g2.setColor(GamePalette.DUSK_NAVY);
                g2.setStroke(new BasicStroke(1.2f));
            }
            g2.drawRoundRect(x, y, w, h, arc, arc);

            g2.dispose();
        }
    }

    // --- DEFEAT PARTICLE FOR DEATH ANIMATION ---
    private static class DefeatParticle {
        double x, y, vx, vy;
        int size;
        Color color;
        float alpha = 1.0f;

        DefeatParticle(double x, double y, double vx, double vy, int size, Color color) {
            this.x = x;
            this.y = y;
            this.vx = vx;
            this.vy = vy;
            this.size = size;
            this.color = color;
        }

        void update() {
            x += vx;
            y += vy;
            vy += 0.35; // gravity
            alpha = Math.max(0f, alpha - 0.055f);
        }

        void render(Graphics2D g2) {
            if (alpha <= 0.01f) return;
            Graphics2D gp = (Graphics2D) g2.create();
            gp.setComposite(AlphaComposite.getInstance(AlphaComposite.SRC_OVER, alpha));
            gp.setColor(color);
            gp.fillOval((int) x, (int) y, size, size);
            gp.dispose();
        }
    }

    // --- CUSTOM ENEMY BOX SPRITE PANEL (Dizon, Alviola, Rola) ---
    private static class EnemyBoxPanel extends JPanel {
        protected boolean isHurt = false;
        protected boolean isAttacking = false;
        protected int shakeOffset = 0;
        protected int lungeOffset = 0; // horizontal lunge towards Juan (negative x)
        protected int stageNum = 1;
        protected String enemyName = "DIZON";
        protected Enemy currentEnemyObj;

        // Animated transition states (Death, Disappearance, Slide-in Entrance from Right)
        protected boolean isDying = false;
        protected float deathAlpha = 1.0f;
        protected float deathScale = 1.0f;
        protected int deathDropOffset = 0;
        protected boolean isEntering = false;
        protected int entranceOffset = 0;
        protected final java.util.List<DefeatParticle> particles = new java.util.ArrayList<>();

        EnemyBoxPanel() {
            setOpaque(false);
            setPreferredSize(new Dimension(140, 110));
        }

        void setEnemy(Enemy enemy) {
            this.currentEnemyObj = enemy;
            if (enemy != null) {
                this.stageNum = enemy.getStage();
                this.enemyName = enemy.getName();
            }
            this.isHurt = false;
            this.isAttacking = false;
            this.shakeOffset = 0;
            this.lungeOffset = 0;
            this.deathAlpha = 1.0f;
            this.deathScale = 1.0f;
            this.deathDropOffset = 0;
            this.entranceOffset = 0;
            this.isDying = false;
            this.isEntering = false;
            this.particles.clear();
            repaint();
        }

        void setEnemy(int stage, String name) {
            this.stageNum = stage;
            this.enemyName = (name != null) ? name.toUpperCase() : "DIZON";
            this.currentEnemyObj = EnemyFactory.createEnemy(stage, BattlePanel.DifficultySetting.SAKTO);
            this.isHurt = false;
            this.isAttacking = false;
            this.shakeOffset = 0;
            this.lungeOffset = 0;
            this.deathAlpha = 1.0f;
            this.deathScale = 1.0f;
            this.deathDropOffset = 0;
            this.entranceOffset = 0;
            this.isDying = false;
            this.isEntering = false;
            this.particles.clear();
            repaint();
        }

        void setHurt(boolean hurt) {
            this.isHurt = hurt;
            repaint();
        }

        void triggerHit() {
            isHurt = true;
            shakeOffset = 6;
            repaint();
            new javax.swing.Timer(180, e -> {
                isHurt = false;
                shakeOffset = 0;
                repaint();
                ((javax.swing.Timer) e.getSource()).stop();
            }).start();
        }

        // Enemy dies, shudders, collapses, bursts defeat particles, and disappears
        void triggerDeath(Runnable onDisappeared) {
            this.isDying = true;
            this.isHurt = true;
            this.isAttacking = false;
            this.lungeOffset = 0;
            this.deathAlpha = 1.0f;
            this.deathScale = 1.0f;
            this.deathDropOffset = 0;
            this.particles.clear();

            final int totalFrames = 22; // ~650ms at 30ms per frame
            final int[] frame = {0};

            javax.swing.Timer deathTimer = new javax.swing.Timer(30, null);
            deathTimer.addActionListener(e -> {
                frame[0]++;
                if (frame[0] <= 6) {
                    // Phase 1: Violent shudder and hurt flash
                    shakeOffset = (frame[0] % 2 == 0) ? 8 : -8;
                    isHurt = true;
                    // On frame 5, spawn defeat burst particles
                    if (frame[0] == 5) {
                        int cx = getWidth() / 2;
                        int cy = getHeight() / 2;
                        Color[] pColors = {
                            GamePalette.HONEY_GOLD,
                            GamePalette.PUMPKIN_ORANGE,
                            Color.WHITE,
                            GamePalette.SPICY_SWEETCORN
                        };
                        for (int i = 0; i < 20; i++) {
                            double angle = (2 * Math.PI / 20) * i;
                            double speed = 3.2 + (i % 3) * 1.6;
                            particles.add(new DefeatParticle(
                                cx, cy,
                                Math.cos(angle) * speed,
                                Math.sin(angle) * speed - 2.5,
                                6 + (i % 3) * 2,
                                pColors[i % pColors.length]
                            ));
                        }
                    }
                } else {
                    // Phase 2: Fade, shrink and sink into ground
                    shakeOffset = 0;
                    isHurt = false;
                    float progress = (float) (frame[0] - 6) / (totalFrames - 6);
                    deathAlpha = Math.max(0f, 1.0f - progress);
                    deathScale = Math.max(0.15f, 1.0f - 0.75f * progress);
                    deathDropOffset = (int) (24 * progress);
                }

                // Update particles
                for (int i = particles.size() - 1; i >= 0; i--) {
                    DefeatParticle p = particles.get(i);
                    p.update();
                    if (p.alpha <= 0.01f) {
                        particles.remove(i);
                    }
                }

                if (getParent() != null) getParent().repaint();
                repaint();

                if (frame[0] >= totalFrames) {
                    deathTimer.stop();
                    deathAlpha = 0f;
                    isDying = false;
                    shakeOffset = 0;
                    repaint();
                    if (getParent() != null) getParent().repaint();
                    if (onDisappeared != null) {
                        onDisappeared.run();
                    }
                }
            });
            deathTimer.start();
        }

        // New enemy shows up from the right side of the screen and smoothly settles into position
        void triggerEntrance(Enemy newEnemy, Runnable onArrived) {
            setEnemy(newEnemy);
            this.deathAlpha = 1.0f;
            this.deathScale = 1.0f;
            this.deathDropOffset = 0;
            this.isDying = false;
            this.particles.clear();
            this.isEntering = true;

            final int startOffset = 320; // Enters from offscreen right
            this.entranceOffset = startOffset;

            final int totalFrames = 22; // ~550ms
            final int[] frame = {0};

            javax.swing.Timer enterTimer = new javax.swing.Timer(25, null);
            enterTimer.addActionListener(e -> {
                frame[0]++;
                float t = (float) frame[0] / totalFrames;
                if (t >= 1.0f) {
                    entranceOffset = 0;
                    isEntering = false;
                    enterTimer.stop();
                    // Subtle arrival impact thump
                    shakeOffset = 3;
                    new javax.swing.Timer(80, evt -> {
                        shakeOffset = 0;
                        repaint();
                        if (getParent() != null) getParent().repaint();
                        ((javax.swing.Timer) evt.getSource()).stop();
                    }).start();

                    if (getParent() != null) getParent().repaint();
                    repaint();
                    if (onArrived != null) {
                        onArrived.run();
                    }
                } else {
                    // Ease out cubic
                    float ease = (float) (1.0 - Math.pow(1.0 - t, 3));
                    entranceOffset = (int) (startOffset * (1.0f - ease));
                    if (getParent() != null) getParent().repaint();
                    repaint();
                }
            });
            enterTimer.start();
        }

        void triggerAttack(Runnable onImpact, Runnable onComplete) {
            this.isAttacking = true;
            this.isHurt = false;
            this.shakeOffset = 0;
            this.lungeOffset = 0;

            final int[] frame = {0};
            // Trajectory: Pull back right (+4, +10, +14), strike left towards Juan (-14, -30, -45 apex impact!), Recoil (-25, -10, 0)
            final int[] steps = {4, 10, 14, -14, -30, -45, -45, -25, -10, -3, 0};

            javax.swing.Timer attackTimer = new javax.swing.Timer(32, null);
            attackTimer.addActionListener(e -> {
                frame[0]++;
                if (frame[0] < steps.length) {
                    lungeOffset = steps[frame[0]];
                    // Frame 5 is peak lunge impact!
                    if (frame[0] == 5) {
                        if (onImpact != null) {
                            onImpact.run();
                        }
                    }
                    repaint();
                } else {
                    lungeOffset = 0;
                    isAttacking = false;
                    repaint();
                    attackTimer.stop();
                    if (onComplete != null) {
                        onComplete.run();
                    }
                }
            });
            attackTimer.start();
        }

        @Override
        protected void paintComponent(Graphics g) {
            super.paintComponent(g);
            Graphics2D g2 = (Graphics2D) g.create();
            g2.setRenderingHint(RenderingHints.KEY_ANTIALIASING, RenderingHints.VALUE_ANTIALIAS_ON);
            // Allow drawing outside component bounds during entrance so it slides from offscreen right!
            g2.setClip(null);

            // Render defeat particles if any
            for (DefeatParticle p : particles) {
                p.render(g2);
            }

            // If completely dead / disappeared, do not render body
            if (deathAlpha <= 0.001f) {
                g2.dispose();
                return;
            }

            int panelH = getHeight();
            int panelW = getWidth();
            int baseW = 110;
            int baseH = 85;
            float eScale = 1.0f;
            if (panelH > 100) {
                eScale = Math.min((float) panelW / 140f, (float) panelH / 100f);
                if (eScale < 1.0f) eScale = 1.0f;
                if (eScale > 2.2f) eScale = 2.2f;
            }
            int w = (int) Math.round(baseW * eScale);
            int h = (int) Math.round(baseH * eScale);
            int x = (panelW - w) / 2 + shakeOffset + lungeOffset + entranceOffset;
            int y = (panelH - h) / 2 + (stageNum == 3 ? 10 : 0) + deathDropOffset;

            // Box shadow (fades with deathAlpha)
            if (deathAlpha > 0.1f) {
                int shadowAlpha = (int) (35 * deathAlpha);
                g2.setColor(new Color(0, 0, 0, Math.min(255, Math.max(0, shadowAlpha))));
                g2.fillOval(x + 5, y + h - 6, w - 10, 14);
            }

            // Motion wind trails when lunging forward towards Juan
            if (isAttacking && lungeOffset < -5) {
                g2.setColor(GamePalette.SKY_BLUE);
                g2.setStroke(new BasicStroke(2.5f));
                g2.drawLine(x + w + 4, y + 22, x + w + 26, y + 22);
                g2.drawLine(x + w + 6, y + 42, x + w + 32, y + 42);
                g2.drawLine(x + w + 3, y + 62, x + w + 24, y + 62);
            }

            // Motion trails when rushing in from the right edge
            if (isEntering && entranceOffset > 15) {
                g2.setColor(new Color(110, 210, 255, Math.min(220, entranceOffset)));
                g2.setStroke(new BasicStroke(2.8f));
                g2.drawLine(x + w + 6, y + 22, x + w + 36, y + 22);
                g2.drawLine(x + w + 12, y + 44, x + w + 50, y + 44);
                g2.drawLine(x + w + 8, y + 66, x + w + 38, y + 66);
            }

            // Apply death alpha & scale transformations
            Graphics2D gEnemy = (Graphics2D) g2.create();
            if (deathAlpha < 1.0f || deathScale < 0.999f) {
                gEnemy.setComposite(AlphaComposite.getInstance(AlphaComposite.SRC_OVER, deathAlpha));
                int cx = x + w / 2;
                int cy = y + h / 2;
                gEnemy.translate(cx, cy);
                gEnemy.scale(deathScale, deathScale);
                gEnemy.translate(-cx, -cy);
            }

            // PATTERN 2: FACTORY METHOD - Polymorphic sprite rendering via Enemy product
            if (currentEnemyObj != null) {
                currentEnemyObj.render(gEnemy, x, y, w, h, isHurt, lungeOffset);
            } else if (stageNum == 2) {
                new AlviolaEnemy().render(gEnemy, x, y, w, h, isHurt, lungeOffset);
            } else if (stageNum == 3) {
                new RolaBossEnemy().render(gEnemy, x, y, w, h, isHurt, lungeOffset);
            } else {
                new DizonEnemy().render(gEnemy, x, y, w, h, isHurt, lungeOffset);
            }

            gEnemy.dispose();
            g2.dispose();
        }
    }

        public void triggerHit() {
            this.isHurt = true;
            this.hitFlashAlpha = 0.75f;
            this.shakeOffset = -8;

            final int[] frame = {0};
            final int[] offsets = {-8, 8, -6, 5, -3, 0};
            javax.swing.Timer shakeTimer = new javax.swing.Timer(45, null);
            shakeTimer.addActionListener(e -> {
                frame[0]++;
                if (frame[0] < offsets.length) {
                    shakeOffset = offsets[frame[0]];
                    hitFlashAlpha = Math.max(0f, hitFlashAlpha - 0.15f);
                    repaint();
                } else {
                    shakeOffset = 0;
                    isHurt = false;
                    hitFlashAlpha = 0f;
                    repaint();
                    shakeTimer.stop();
                }
            });
            shakeTimer.start();
        }

        public void triggerAttack(Runnable onImpact, Runnable onComplete) {
            this.isAttacking = true;
            this.isHurt = false;
            this.shakeOffset = 0;
            this.lungeOffset = 0;
            this.currentAnimState = HeroAnimState.PRE_ATTACK;
            repaint();

            // Frame animation steps:
            // 0..2: PRE_ATTACK (wind-up crouch, subtle pullback)
            // 3..7: ATTACK (sword strike forward thrust, apex at frame 5)
            // 8..10: IDLE (recovering back to 0)
            final int[] frame = {0};
            final int[] offsets = {-3, -6, -4, 15, 30, 42, 38, 26, 14, 5, 0};
            final HeroAnimState[] states = {
                HeroAnimState.PRE_ATTACK,
                HeroAnimState.PRE_ATTACK,
                HeroAnimState.PRE_ATTACK,
                HeroAnimState.ATTACK,
                HeroAnimState.ATTACK,
                HeroAnimState.ATTACK,
                HeroAnimState.ATTACK,
                HeroAnimState.ATTACK,
                HeroAnimState.IDLE,
                HeroAnimState.IDLE,
                HeroAnimState.IDLE
            };

            javax.swing.Timer attackTimer = new javax.swing.Timer(36, null);
            attackTimer.addActionListener(e -> {
                frame[0]++;
                if (frame[0] < offsets.length) {
                    lungeOffset = offsets[frame[0]];
                    currentAnimState = states[frame[0]];
                    if (frame[0] == 5) { // Apex of strike
                        if (onImpact != null) onImpact.run();
                    }
                    repaint();
                } else {
                    lungeOffset = 0;
                    currentAnimState = HeroAnimState.IDLE;
                    isAttacking = false;
                    repaint();
                    attackTimer.stop();
                    if (onComplete != null) onComplete.run();
                }
            });
            attackTimer.start();
        }

        @Override
        protected void paintComponent(Graphics g) {
            super.paintComponent(g);
            Graphics2D g2 = (Graphics2D) g.create();
            g2.setRenderingHint(RenderingHints.KEY_ANTIALIASING, RenderingHints.VALUE_ANTIALIAS_ON);
            g2.setRenderingHint(RenderingHints.KEY_INTERPOLATION, RenderingHints.VALUE_INTERPOLATION_BILINEAR);
            g2.setClip(null); // Allow sword strike without boundary clipping

            int panelH = getHeight();
            int h = panelH > 0 ? (int) Math.round(panelH * 0.94) : 155;
            if (h < 110) h = 110;
            int w = (int) Math.round(h * (300.0 / 500.0));

            // Anchor boots firmly on arena floor baseline
            int targetGroundY = (panelH > 0) ? panelH - 14 : 140;
            int y = targetGroundY - (int) Math.round(h * (440.0 / 500.0));
            int x = (getWidth() - w) / 2 + shakeOffset + lungeOffset;

            // Ground shadow directly under boots
            int shadowW = (int) Math.round(w * 0.7);
            g2.setColor(new Color(0, 0, 0, 60));
            g2.fillOval(x + (w - shadowW) / 2, targetGroundY - 6, shadowW, 12);

            // Motion wind/slash trails when lunging forward
            if (isAttacking && lungeOffset > 10) {
                g2.setColor(new Color(255, 220, 110, Math.min(220, lungeOffset * 5)));
                g2.setStroke(new BasicStroke(2.2f));
                g2.drawLine(x - 24, y + h / 2 - 12, x - 4, y + h / 2 - 12);
                g2.drawLine(x - 36, y + h / 2 + 6, x - 6, y + h / 2 + 6);
                g2.drawLine(x - 20, y + h / 2 + 22, x - 4, y + h / 2 + 22);
            }

            Image spriteToDraw;
            switch (currentAnimState) {
                case PRE_ATTACK:
                    spriteToDraw = (preAttackSprite != null) ? preAttackSprite : idleSprite;
                    break;
                case ATTACK:
                    spriteToDraw = (attackSprite != null) ? attackSprite : idleSprite;
                    break;
                case IDLE:
                default:
                    spriteToDraw = idleSprite;
                    break;
            }

            if (spriteToDraw != null) {
                g2.drawImage(spriteToDraw, x, y, w, h, null);
                if (isHurt && hitFlashAlpha > 0f) {
                    g2.setComposite(AlphaComposite.getInstance(AlphaComposite.SRC_ATOP, hitFlashAlpha));
                    g2.setColor(GamePalette.CRIMSON_RED);
                    g2.fillRect(x, y, w, h);
                }
            } else {
                g2.setFont(new Font("Arial", Font.BOLD, 18));
                g2.setColor(isHurt ? GamePalette.CRIMSON_RED : GamePalette.CREAM);
                g2.drawString("👤 JUAN", x + 5, y + h / 2);
            }

            if (isHurt) {
                g2.setFont(new Font("Arial", Font.BOLD, 22));
                g2.drawString("💥", x + w / 2, y + 20);
            }

            g2.dispose();
        }
    }

            g2.setRenderingHint(RenderingHints.KEY_INTERPOLATION, RenderingHints.VALUE_INTERPOLATION_BILINEAR);

            int w = getWidth() - 4;
            int h = getHeight() - 4;
            int x = 2;
            int y = 2;

            char ch = Character.toUpperCase(getLetter());
            BufferedImage tileImg = getLetterTileImage(ch);

            if (isSelected) {
                // Golden aura for selected tile
                g2.setColor(new Color(245, 180, 50, 175));
                g2.setStroke(new BasicStroke(3.5f));
                g2.drawRoundRect(x - 1, y - 1, w + 2, h + 2, 10, 10);
            } else if (isHovered) {
                // Sky blue hover aura
                g2.setColor(new Color(90, 190, 255, 160));
                g2.setStroke(new BasicStroke(2.5f));
                g2.drawRoundRect(x - 1, y - 1, w + 2, h + 2, 10, 10);
            }

            if (tileImg != null) {
                // Draw high-resolution alphabet tile sprite centered as a square
                int size = Math.min(w, h);
                int drawX = x + (w - size) / 2;
                int drawY = y + (h - size) / 2;
                g2.drawImage(tileImg, drawX, drawY, size, size, null);

                if (isSelected) {
                    Graphics2D gTint = (Graphics2D) g2.create();
                    gTint.setComposite(AlphaComposite.getInstance(AlphaComposite.SRC_ATOP, 0.16f));
                    gTint.setColor(new Color(255, 200, 50));
                    gTint.fillRoundRect(drawX, drawY, size, size, 8, 8);
                    gTint.dispose();

                    g2.setColor(new Color(240, 150, 30));
                    g2.setStroke(new BasicStroke(2.2f));
                    g2.drawRoundRect(drawX, drawY, size, size, 8, 8);
                } else if (isHovered) {
                    g2.setColor(new Color(110, 200, 255));
                    g2.setStroke(new BasicStroke(1.8f));
                    g2.drawRoundRect(drawX, drawY, size, size, 8, 8);
                }
            } else {
                // Fallback procedural rendering if sprite is unavailable
                Color bg = isSelected ? GamePalette.HONEY_GOLD : (isHovered ? Color.WHITE : GamePalette.CREAM);
                g2.setColor(bg);
                g2.fillRoundRect(x, y, w, h, 10, 10);

                g2.setColor(isSelected ? GamePalette.PUMPKIN_ORANGE : (isHovered ? GamePalette.SKY_BLUE : GamePalette.DUSK_NAVY));
                g2.setStroke(new BasicStroke(isSelected ? 2.5f : (isHovered ? 2f : 1.2f)));
                g2.drawRoundRect(x, y, w, h, 10, 10);

                // Main Letter (Centered)
                g2.setFont(new Font("Arial", Font.BOLD, 19));
                g2.setColor(GamePalette.MIDNIGHT_NAVY);
                String str = String.valueOf(ch);
                FontMetrics fm = g2.getFontMetrics();
                int lx = x + (w - fm.stringWidth(str)) / 2;
                int ly = y + (h + fm.getAscent() - fm.getDescent()) / 2 - 1;
                g2.drawString(str, lx, ly);

                // Points Value (Bottom Right Subscript with high-contrast badge for procedural fallback)
                g2.setFont(new Font("Arial", Font.BOLD, 9));
                String ptStr = String.valueOf(getPoints());
                FontMetrics pfm = g2.getFontMetrics();
                int pw = pfm.stringWidth(ptStr);
                int px = x + w - pw - 4;
                int py = y + h - 4;

                g2.setColor(new Color(255, 255, 255, 190));
                g2.fillRoundRect(px - 2, py - pfm.getAscent() + 1, pw + 4, pfm.getAscent() + 2, 4, 4);
                g2.setColor(new Color(210, 195, 175));
                g2.setStroke(new BasicStroke(1.0f));
                g2.drawRoundRect(px - 2, py - pfm.getAscent() + 1, pw + 4, pfm.getAscent() + 2, 4, 4);

                g2.setColor(isSelected ? GamePalette.PUMPKIN_ORANGE : GamePalette.MIDNIGHT_NAVY);
                g2.drawString(ptStr, px, py);
            }

            // PATTERN 4: DECORATOR - Render decorator badge (Gold star / Healing heart)
            if (tileComponent != null) {
                tileComponent.renderDecoratorBadge(g2, x, y, w, h);
            }

            g2.dispose();
        }
    }

/**
 * Handles all CSV reading and writing for JUAN Word.
 * All CSV files live in the "data/" directory at the project root.
 *
 * Writable progress files:
 *  - players.csv
 *  - unlocks.csv
 *  - settings.csv
 *  - highscores.csv
 *  - bestiary_progress.csv
 *
 * Read-only game content files:
 *  - stages.csv
 *  - enemies.csv
 *  - bestiary.csv
 *  - letter_values.csv
 *  - cebuano_dictionary_all_entries.csv
 */
public class DataManager {

    public static final Path DATA_DIR = Paths.get("data");

    // File paths
    public static final Path PLAYERS = DATA_DIR.resolve("players.csv");
    public static final Path UNLOCKS = DATA_DIR.resolve("unlocks.csv");
    public static final Path SETTINGS = DATA_DIR.resolve("settings.csv");
    public static final Path HIGHSCORES = DATA_DIR.resolve("highscores.csv");
    public static final Path BESTIARY_PROGRESS = DATA_DIR.resolve("bestiary_progress.csv");

    public static final Path STAGES = DATA_DIR.resolve("stages.csv");
    public static final Path ENEMIES = DATA_DIR.resolve("enemies.csv");
    public static final Path BESTIARY = DATA_DIR.resolve("bestiary.csv");
    public static final Path LETTER_VALUES = DATA_DIR.resolve("letter_values.csv");
    public static final Path DICTIONARY = DATA_DIR.resolve("cebuano_dictionary_all_entries.csv");

    // Standard CSV headers
    private static final String PLAYERS_HEADER = "playerId,playerName,currentStage,currentRound,score,experience,coins,currentHealth,maxHealth,lastSaved,savedTiles,savedDifficulty,savedEnemyHealth,savedMap,unlockedSayonStage,unlockedSaktoStage,unlockedLisodStage,unlockedMapIndex";
    private static final String UNLOCKS_HEADER = "playerId,stage1Unlocked,stage2Unlocked,stage3Unlocked,stage4Unlocked,tikbalangDefeated,kapreDefeated,manananggalDefeated,aswangDefeated";
    private static final String SETTINGS_HEADER = "playerId,musicOn,sfxOn,volume";
    private static final String HIGHSCORES_HEADER = "playerName,score,stageReached,date";
    private static final String BESTIARY_PROGRESS_HEADER = "playerId,creatureId,discovered,timesDefeated";

    // In-memory cache for valid dictionary words (O(1) lookup)
    private Set<String> dictionaryWords = null;

    // SINGLETON PATTERN NI SYA SIR GUDO
    private static volatile DataManager instance;

    /**
     * Retrieves the canonical Singleton instance of DataManager.
     * Uses double-checked locking for thread safety and performance.
     */
    public static DataManager getInstance() {
        if (instance == null) {
            synchronized (DataManager.class) {
                if (instance == null) {
                    instance = new DataManager();
                }
            }
        }
        return instance;
    }

    public DataManager() {
        ensureFile(PLAYERS, PLAYERS_HEADER);
        ensureFile(UNLOCKS, UNLOCKS_HEADER);
        ensureFile(SETTINGS, SETTINGS_HEADER);
        ensureFile(HIGHSCORES, HIGHSCORES_HEADER);
        ensureFile(BESTIARY_PROGRESS, BESTIARY_PROGRESS_HEADER);
    }

    private void ensureFile(Path file, String header) {
        try {
            Files.createDirectories(DATA_DIR);
            if (!Files.exists(file)) {
                Files.write(file, Collections.singletonList(header), StandardCharsets.UTF_8);
            }
        } catch (IOException e) {
            System.err.println("Cannot create data file: " + file.toAbsolutePath() + " - " + e.getMessage());
        }
    }

    private List<String> readLines(Path file) {
        if (!Files.exists(file)) return new ArrayList<>();
        try {
            return new ArrayList<>(Files.readAllLines(file, StandardCharsets.UTF_8));
        } catch (IOException e) {
            System.err.println("Cannot read: " + file.toAbsolutePath() + " - " + e.getMessage());
            return new ArrayList<>();
        }
    }

    private void writeLines(Path file, List<String> lines) {
        try {
            Files.write(file, lines, StandardCharsets.UTF_8);
        } catch (IOException e) {
            System.err.println("Cannot write: " + file.toAbsolutePath() + " - " + e.getMessage());
        }
    }

    // ==========================================
    // 1. PLAYERS (Saved Game)
    // ==========================================

    public void savePlayer(Player p) {
        p.lastSaved = LocalDateTime.now().format(DateTimeFormatter.ofPattern("yyyy-MM-dd HH:mm"));
        List<String> lines = readLines(PLAYERS);
        if (lines.isEmpty() || !lines.get(0).startsWith("playerId,")) {
            lines.add(0, PLAYERS_HEADER);
        }
        boolean found = false;
        for (int i = 1; i < lines.size(); i++) {
            if (lines.get(i).startsWith(p.playerId + ",")) {
                lines.set(i, p.toCsv());
                found = true;
                break;
            }
        }
        if (!found) lines.add(p.toCsv());
        writeLines(PLAYERS, lines);
    }

    public Player loadPlayer(String playerId) {
        for (Player p : getAllPlayers()) {
            if (p.playerId.equals(playerId)) return p;
        }
        return null;
    }

    public List<Player> getAllPlayers() {
        List<Player> list = new ArrayList<>();
        List<String> lines = readLines(PLAYERS);
        for (int i = 1; i < lines.size(); i++) {
            if (lines.get(i).trim().isEmpty()) continue;
            Player p = Player.fromCsv(lines.get(i));
            if (p != null) list.add(p);
        }
        return list;
    }

    public Player loadLatestPlayer() {
        Player latest = null;
        for (Player p : getAllPlayers()) {
            if (latest == null || p.lastSaved.compareTo(latest.lastSaved) >= 0) {
                latest = p;
            }
        }
        return latest;
    }

    public String nextPlayerId() {
        int max = 0;
        for (Player p : getAllPlayers()) {
            try {
                max = Math.max(max, Integer.parseInt(p.playerId.substring(1)));
            } catch (Exception ignored) { }
        }
        return String.format("P%03d", max + 1);
    }

    public void deletePlayer(String playerId) {
        if (playerId == null || playerId.trim().isEmpty()) return;
        removePlayerLines(PLAYERS, playerId);
        removePlayerLines(UNLOCKS, playerId);
        removePlayerLines(SETTINGS, playerId);
        removePlayerLines(BESTIARY_PROGRESS, playerId);
    }

    private void removePlayerLines(Path file, String playerId) {
        List<String> lines = readLines(file);
        if (lines.isEmpty()) return;
        List<String> updated = new ArrayList<>();
        String header = lines.get(0);
        if (file.equals(PLAYERS) && !header.startsWith("playerId,")) {
            header = PLAYERS_HEADER;
        }
        updated.add(header); // Always keep header row intact
        for (int i = 1; i < lines.size(); i++) {
            String line = lines.get(i).trim();
            if (line.isEmpty()) continue;
            if (line.startsWith(playerId + ",")) continue;
            // Filter out any broken fragments
            if (file.equals(PLAYERS) && !line.startsWith("P")) continue;
            updated.add(line);
        }
        writeLines(file, updated);
    }

    public List<String> getTopScores(int limit) {
        List<String[]> rows = new ArrayList<>();
        List<String> lines = readLines(HIGHSCORES);
        for (int i = 1; i < lines.size(); i++) {
            String[] f = lines.get(i).split(",", -1);
            if (f.length >= 4) rows.add(f);
        }
        rows.sort((a, b) -> Integer.compare(parse(b[1]), parse(a[1])));
        List<String> out = new ArrayList<>();
        for (int i = 0; i < Math.min(limit, rows.size()); i++) {
            String[] r = rows.get(i);
            out.add((i + 1) + ". " + r[0] + " - " + r[1] + " pts (Stage " + r[2] + ", " + r[3] + ")");
        }
        return out;
    }

        public int stageId;
        public String name;
        public String location;
        public String difficulty;
        public int regularRounds;
        public String bossId;
    }

    public List<Stage> loadStages() {
        List<Stage> list = new ArrayList<>();
        List<String> lines = readLines(STAGES);
        for (int i = 1; i < lines.size(); i++) {
            String[] f = lines.get(i).split(",", -1);
            if (f.length >= 6) {
                Stage s = new Stage();
                s.stageId = parse(f[0]);
                s.name = f[1].trim();
                s.location = f[2].trim();
                s.difficulty = f[3].trim();
                s.regularRounds = parse(f[4]);
                s.bossId = f[5].trim();
                list.add(s);
            }
        }
        return list;
    }

    public static class Enemy {
        public String enemyId;
        public String name;
        public int stage;
        public String type;
        public int maxHealth;
        public int attack;
        public String ability;
    }

    public List<Enemy> loadEnemies() {
        List<Enemy> list = new ArrayList<>();
        List<String> lines = readLines(ENEMIES);
        for (int i = 1; i < lines.size(); i++) {
            String[] f = lines.get(i).split(",", -1);
            if (f.length >= 7) {
                Enemy e = new Enemy();
                e.enemyId = f[0].trim();
                e.name = f[1].trim();
                e.stage = parse(f[2]);
                e.type = f[3].trim();
                e.maxHealth = parse(f[4]);
                e.attack = parse(f[5]);
                e.ability = f[6].trim();
                list.add(e);
            }
        }
        return list;
    }

    public static class BestiaryEntry {
        public String creatureId;
        public String name;
        public int stage;
        public String description;
    }

    public List<BestiaryEntry> loadBestiary() {
        List<BestiaryEntry> list = new ArrayList<>();
        List<String> lines = readLines(BESTIARY);
        for (int i = 1; i < lines.size(); i++) {
            String[] f = lines.get(i).split(",", -1);
            if (f.length >= 4) {
                BestiaryEntry b = new BestiaryEntry();
                b.creatureId = f[0].trim();
                b.name = f[1].trim();
                b.stage = parse(f[2]);
                b.description = f[3].trim();
                list.add(b);
            }
        }
        return list;
    }

    public Map<Character, Integer> loadLetterPoints() {
        Map<Character, Integer> map = new HashMap<>();
        for (String line : readLines(LETTER_VALUES)) {
            String[] f = line.split(",", -1);
            if (f.length >= 2 && !f[0].trim().equalsIgnoreCase("letter")) {
                String letter = f[0].trim().toLowerCase();
                if (!letter.isEmpty()) {
                    map.put(letter.charAt(0), parse(f[1]));
                }
            }
        }
        return map;
    }

    public Map<Character, Integer> loadLetterWeights() {
        Map<Character, Integer> map = new HashMap<>();
        for (String line : readLines(LETTER_VALUES)) {
            String[] f = line.split(",", -1);
            if (f.length >= 3 && !f[0].trim().equalsIgnoreCase("letter")) {
                String letter = f[0].trim().toLowerCase();
                if (!letter.isEmpty()) {
                    map.put(letter.charAt(0), parse(f[2]));
                }
            }
        }
        return map;
    }

    // ==========================================
    // 7. DICTIONARY (Word Validation)
    // ==========================================

    /**
     * Loads the 16,000+ word Cebuano dictionary into memory for O(1) word validation.
     */
    public synchronized Set<String> getDictionaryWords() {
        if (dictionaryWords == null) {
            dictionaryWords = new HashSet<>();
            List<String> lines = readLines(DICTIONARY);
            for (int i = 1; i < lines.size(); i++) {
                String line = lines.get(i);
                if (line.isEmpty()) continue;
                String[] f = line.split(",", -1);
                if (f.length >= 2) {
                    String word = f[1].trim().toLowerCase();
                    if (!word.isEmpty()) {
                        dictionaryWords.add(word);
                        // Also index without hyphens (e.g. "ab-ab" -> "abab")
                        if (word.contains("-")) {
                            dictionaryWords.add(word.replace("-", ""));
                        }
                    }
                }
            }
        }
        return dictionaryWords;
    }

    /**
     * Checks if a word exists in the Cebuano dictionary.
     */
    public boolean isValidWord(String word) {
        if (word == null || word.trim().isEmpty()) return false;
        String clean = word.trim().toLowerCase();
        return getDictionaryWords().contains(clean) || getDictionaryWords().contains(clean.replace("-", ""));
    }

    private static int parse(String s) {
        try { return Integer.parseInt(s.trim()); } catch (Exception e) { return 0; }
    }
}

/**
 * PATTERN 2: FACTORY METHOD - Abstract Product
 * Base class for all combat opponents in JuanWord.
 */
public abstract class Enemy {
    protected final String name;
    protected final String title;
    protected final String subtitle;
    protected final int stage;
    protected final int baseHp;
    protected final int baseAtk;

    protected int maxHp;
    protected int curHp;
    protected int atk;

    public Enemy(String name, String title, String subtitle, int stage, int baseHp, int baseAtk) {
        this.name = name;
        this.title = title;
        this.subtitle = subtitle;
        this.stage = stage;
        this.baseHp = baseHp;
        this.baseAtk = baseAtk;
        this.maxHp = baseHp;
        this.curHp = baseHp;
        this.atk = baseAtk;
    }

    public String getName() { return name; }
    public String getTitle() { return title; }
    public String getSubtitle() { return subtitle; }
    public int getStage() { return stage; }
    public int getBaseHp() { return baseHp; }
    public int getBaseAtk() { return baseAtk; }
    public int getMaxHp() { return maxHp; }
    public int getCurHp() { return curHp; }
    public int getAtk() { return atk; }

    public void setCurHp(int hp) { this.curHp = Math.max(0, Math.min(maxHp, hp)); }
    public void setMaxHp(int maxHp) { this.maxHp = maxHp; }
    public void setAtk(int atk) { this.atk = atk; }

    public void takeDamage(int dmg) {
        this.curHp = Math.max(0, this.curHp - dmg);
    }

    public void scaleStats(BattlePanel.DifficultySetting difficulty) {
        if (difficulty == null) difficulty = BattlePanel.DifficultySetting.SAKTO;
        switch (difficulty) {
            case SAYON:
                this.maxHp = Math.max(30, (int) Math.round(baseHp * 0.75));
                this.atk = Math.max(4, (int) Math.round(baseAtk * 0.70));
                break;
            case LISOD:
                this.maxHp = (int) Math.round(baseHp * 1.30);
                this.atk = (int) Math.round(baseAtk * 1.40);
                break;
            case SAKTO:
            default:
                this.maxHp = baseHp;
                this.atk = baseAtk;
                break;
        }
        this.curHp = this.maxHp;
    }

    /**
     * Renders this enemy's unique visual box sprite into the arena.
     */
    public abstract void render(Graphics2D g2, int x, int y, int w, int h, boolean isHurt, int lungeOffset);
}

/**
 * Clean, solid, flat game button that uses the official GamePalette
 * and renders consistently across all platforms (bypassing OS LookAndFeel overrides).
 */
public class FlatGameButton extends JButton {
    private Color normalBg;
    private Color hoverBg;
    private Color normalFg;
    private Color borderColor;
    private int cornerRadius = 14;
    private boolean isHovered = false;

    public FlatGameButton(String text, Color bg, Color fg, Color border) {
        super(text);
        this.normalBg = bg;
        this.hoverBg = (bg != null) ? bg.brighter() : null;
        this.normalFg = fg;
        this.borderColor = border;

        setFont(new Font("Arial", Font.BOLD, 14));
        setFocusPainted(false);
        setContentAreaFilled(false);
        setBorderPainted(false);
        setOpaque(false);

        addMouseListener(new MouseAdapter() {
            @Override
            public void mouseEntered(MouseEvent e) {
                if (isEnabled()) {
                    isHovered = true;
                    repaint();
                }
            }

            @Override
            public void mouseExited(MouseEvent e) {
                isHovered = false;
                repaint();
            }
        });
    }

    public void setColors(Color bg, Color fg, Color border) {
        this.normalBg = bg;
        this.hoverBg = (bg != null) ? bg.brighter() : null;
        this.normalFg = fg;
        this.borderColor = border;
        repaint();
    }

    public void setBorderColor(Color border) {
        this.borderColor = border;
        repaint();
    }

    public void setCornerRadius(int r) {
        this.cornerRadius = r;
        repaint();
    }

    @Override
    protected void paintComponent(Graphics g) {
        Graphics2D g2 = (Graphics2D) g.create();
        g2.setRenderingHint(RenderingHints.KEY_ANTIALIASING, RenderingHints.VALUE_ANTIALIAS_ON);

        int w = getWidth() - 4;
        int h = getHeight() - 4;
        int x = 2;
        int y = 2;

        if (isEnabled()) {
            // Drop shadow
            g2.setColor(new Color(0, 0, 0, 35));
            g2.fillRoundRect(x + 1, y + 2, w, h, cornerRadius, cornerRadius);

            // Background
            g2.setColor(isHovered ? hoverBg : normalBg);
            g2.fillRoundRect(x, y, w, h, cornerRadius, cornerRadius);

            // Border
            if (borderColor != null) {
                g2.setColor(isHovered ? borderColor.brighter() : borderColor);
                g2.setStroke(new BasicStroke(isHovered ? 2.0f : 1.2f));
                g2.drawRoundRect(x, y, w, h, cornerRadius, cornerRadius);
            }

            // Foreground text
            g2.setFont(getFont());
            g2.setColor(normalFg);
        } else {
            // Disabled appearance
            g2.setColor(GamePalette.DUSK_NAVY);
            g2.fillRoundRect(x, y, w, h, cornerRadius, cornerRadius);

            g2.setColor(GamePalette.DEEP_INDIGO);
            g2.setStroke(new BasicStroke(1.0f));
            g2.drawRoundRect(x, y, w, h, cornerRadius, cornerRadius);

            g2.setFont(getFont());
            g2.setColor(new Color(GamePalette.SKY.getRed(), GamePalette.SKY.getGreen(), GamePalette.SKY.getBlue(), 100));
        }

        String text = getText();
        if (text != null && !text.isEmpty()) {
            FontMetrics fm = g2.getFontMetrics();
            int tx = x + (w - fm.stringWidth(text)) / 2;
            int ty = y + (h + fm.getAscent() - fm.getDescent()) / 2;
            g2.drawString(text, tx, ty);
        }

        g2.dispose();
    }
}

/**
 * High-performance graphical asset button for JuanWord.
 * Renders custom artist-crafted PNG button sprites with:
 * - High-DPI bilinear interpolation
 * - Tactile 2px press-down animation on click
 * - Smooth hover brightness enhancement
 * - High-visibility 38% translucent disabled state
 */
public class ImageAssetButton extends JButton {
    private Image buttonImage;
    private String currentImagePath;
    private boolean isHovered = false;
    private boolean isPressed = false;
    private int targetWidth = 192;
    private int targetHeight = 48;

    public ImageAssetButton(String imagePath) {
        this(imagePath, 192, 48);
    }

    public ImageAssetButton(String imagePath, int width, int height) {
        this.targetWidth = width;
        this.targetHeight = height;
        setImage(imagePath);

        Dimension size = new Dimension(width, height);
        setPreferredSize(size);
        setMinimumSize(size);
        setMaximumSize(size);
        setSize(size);

        setContentAreaFilled(false);
        setBorderPainted(false);
        setFocusPainted(false);
        setOpaque(false);

        addMouseListener(new MouseAdapter() {
            @Override
            public void mouseEntered(MouseEvent e) {
                if (isEnabled()) {
                    isHovered = true;
                    repaint();
                }
            }

            @Override
            public void mouseExited(MouseEvent e) {
                isHovered = false;
                isPressed = false;
                repaint();
            }

            @Override
            public void mousePressed(MouseEvent e) {
                if (isEnabled() && SwingUtilities.isLeftMouseButton(e)) {
                    isPressed = true;
                    repaint();
                }
            }

            @Override
            public void mouseReleased(MouseEvent e) {
                if (isPressed) {
                    isPressed = false;
                    repaint();
                }
            }
        });
    }

    public void setTargetSize(int width, int height) {
        if (this.targetWidth == width && this.targetHeight == height) return;
        this.targetWidth = width;
        this.targetHeight = height;
        Dimension size = new Dimension(width, height);
        setPreferredSize(size);
        setMinimumSize(size);
        setMaximumSize(size);
        setSize(size);
        revalidate();
        repaint();
    }

    public int getTargetWidth() {
        return targetWidth;
    }

    public int getTargetHeight() {
        return targetHeight;
    }

    public void setImage(String imagePath) {
        if (imagePath == null) return;
        this.currentImagePath = imagePath;
        this.buttonImage = loadGameImage(imagePath);
        repaint();
    }

    public String getCurrentImagePath() {
        return currentImagePath;
    }

    private Image loadGameImage(String path) {
        // 1. Direct file lookup
        File f = new File(path);
        if (f.exists()) {
            return new ImageIcon(f.getAbsolutePath()).getImage();
        }
        // 2. Lookup in src/juanword
        File fSrc = new File("src/juanword", path);
        if (fSrc.exists()) {
            return new ImageIcon(fSrc.getAbsolutePath()).getImage();
        }
        // 3. ClassLoader resource lookup (for JAR distribution)
        try {
            java.net.URL url = getClass().getResource("/juanword/" + path);
            if (url == null) url = getClass().getResource("/" + path);
            if (url == null) url = getClass().getResource(path);
            if (url != null) {
                return new ImageIcon(url).getImage();
            }
        } catch (Exception ignored) {}

        // 4. Fallback
        return new ImageIcon(path).getImage();
    }

    @Override
    protected void paintComponent(Graphics g) {
        Graphics2D g2 = (Graphics2D) g.create();
        g2.setRenderingHint(RenderingHints.KEY_ANTIALIASING, RenderingHints.VALUE_ANTIALIAS_ON);
        g2.setRenderingHint(RenderingHints.KEY_INTERPOLATION, RenderingHints.VALUE_INTERPOLATION_BILINEAR);

        int w = getWidth();
        int h = getHeight();

        int drawW = targetWidth > 0 ? Math.min(w, targetWidth) : w;
        int drawH = targetHeight > 0 ? Math.min(h, targetHeight) : h;
        if (drawW <= 0) drawW = w;
        if (drawH <= 0) drawH = h;

        int drawX = (w - drawW) / 2;
        int drawY = (h - drawH) / 2;

        if (isPressed && isEnabled()) {
            // Tactile Physical Button Press: Shift down by 2px on click
            drawY += 2;
        }

        if (!isEnabled()) {
            // Disabled State: Smoothly dimmed with 38% opacity for clean game UI look
            g2.setComposite(AlphaComposite.getInstance(AlphaComposite.SRC_OVER, 0.38f));
            if (buttonImage != null) {
                g2.drawImage(buttonImage, drawX, drawY, drawW, drawH, this);
            }
        } else if (isHovered) {
            // Hover State: Draw button sprite + subtle golden-white highlight overlay
            if (buttonImage != null) {
                g2.drawImage(buttonImage, drawX, drawY, drawW, drawH, this);
                g2.setComposite(AlphaComposite.getInstance(AlphaComposite.SRC_ATOP, 0.14f));
                g2.setColor(Color.WHITE);
                g2.fillRoundRect(drawX, drawY, drawW, drawH, 28, 28);
            }
        } else {
            // Normal Default State
            if (buttonImage != null) {
                g2.drawImage(buttonImage, drawX, drawY, drawW, drawH, this);
            }
        }

        g2.dispose();
    }

    // --- START SCREEN ---
    public JPanel createStartScreen() {
        // Load menu screen layers: bg.png (sky), mtapo.png (Mount Apo), indigframe.png (indigenous border frame)
        Image skyBg = loadGameImage("bg.png");
        Image mtApo = loadGameImage("mtapo.png");
        Image indigFrame = loadGameImage("indigframe.png");
        String fallbackPath = new java.io.File("TITLE_PAGE.png").exists() ? "TITLE_PAGE.png" : "TITILE_PAGE.png";
        Image fallbackBg = loadGameImage(fallbackPath);
        Image titleImage = loadGameImage("Title.png");

        // Declare containers early so doLayout can dynamically scale them
        final JPanel topBar = new JPanel(new BorderLayout());
        final JPanel bottomContainer = new JPanel(new FlowLayout(FlowLayout.CENTER, 0, 0));
        final JPanel buttonPanel = new JPanel(new GridLayout(4, 1, 0, 6));
        buttonPanel.setOpaque(false);

        JPanel backgroundPanel = new JPanel() {
            private BufferedImage cachedComposite = null;
            private int cachedW = -1;
            private int cachedH = -1;

            @Override
            public void doLayout() {
                int w = getWidth();
                int h = getHeight();
                if (w > 0 && h > 0) {
                    float S = Math.min((float) w / 960f, (float) h / 540f);
                    float btnScale = Math.max(1.0f, Math.min(S, 2.2f));
                    int btnW = Math.round(192 * btnScale);
                    int btnH = Math.round(48 * btnScale);
                    int strut = Math.max(4, Math.round(6 * btnScale));
                    int totalH = 4 * btnH + 3 * strut;

                    if (continueButton != null) continueButton.setTargetSize(btnW, btnH);
                    if (startButton != null) startButton.setTargetSize(btnW, btnH);
                    if (settingButton != null) settingButton.setTargetSize(btnW, btnH);
                    if (exitButton != null) exitButton.setTargetSize(btnW, btnH);

                    if (buttonPanel != null) {
                        if (buttonPanel.getLayout() instanceof GridLayout) {
                            ((GridLayout) buttonPanel.getLayout()).setVgap(strut);
                        }
                        buttonPanel.setPreferredSize(new Dimension(btnW, totalH));
                        buttonPanel.setMinimumSize(new Dimension(btnW, totalH));
                        buttonPanel.setMaximumSize(new Dimension(btnW, totalH));
                        buttonPanel.setSize(btnW, totalH);
                        buttonPanel.invalidate();
                    }

                    if (playerCardPanel != null) {
                        int cardW = Math.round(230 * btnScale);
                        int cardH = Math.round(48 * btnScale);
                        playerCardPanel.setPreferredSize(new Dimension(cardW, cardH));
                        playerCardPanel.setMaximumSize(new Dimension(cardW + 10, cardH + 4));
                        if (playerNameLabel != null) playerNameLabel.setFont(new Font("Arial", Font.BOLD, (int) Math.round(13 * btnScale)));
                        if (playerInfoLabel != null) playerInfoLabel.setFont(new Font("Arial", Font.PLAIN, (int) Math.round(11 * btnScale)));
                    }

                    int bPad = Math.max(16, (int) (20 * S));
                    bottomContainer.setBorder(BorderFactory.createEmptyBorder(0, 0, bPad, 0));
                    bottomContainer.invalidate();

                    int tPad = Math.max(12, (int) (18 * S));
                    int lPad = Math.max(16, (int) (24 * S));
                    topBar.setBorder(BorderFactory.createEmptyBorder(tPad, lPad, 0, 0));
                    topBar.invalidate();
                }
                super.doLayout();
                if (bottomContainer != null) {
                    bottomContainer.doLayout();
                }
                if (buttonPanel != null) {
                    buttonPanel.doLayout();
                }
            }

            @Override
            protected void paintComponent(Graphics g) {
                super.paintComponent(g);
                int w = getWidth();
                int h = getHeight();
                if (w <= 0 || h <= 0) return;

                // Rebuild cached background composite if size changes (render-to-texture for 60 FPS)
                if (cachedComposite == null || cachedW != w || cachedH != h) {
                    cachedW = w;
                    cachedH = h;
                    cachedComposite = new BufferedImage(w, h, BufferedImage.TYPE_INT_ARGB);
                    Graphics2D gComp = cachedComposite.createGraphics();
                    gComp.setRenderingHint(RenderingHints.KEY_INTERPOLATION, RenderingHints.VALUE_INTERPOLATION_BILINEAR);
                    gComp.setRenderingHint(RenderingHints.KEY_ANTIALIASING, RenderingHints.VALUE_ANTIALIAS_ON);

                    // 1. Base Layer: Sky & Atmospheric Background (bg.png)
                    if (skyBg != null) {
                        gComp.drawImage(skyBg, 0, 0, w, h, null);
                    } else if (fallbackBg != null) {
                        gComp.drawImage(fallbackBg, 0, 0, w, h, null);
                    }

                    // 2. Middleground Layer: Mount Apo & Foothills (mtapo.png)
                    if (mtApo != null) {
                        gComp.drawImage(mtApo, 0, 0, w, h, null);
                    }

                    // 3. Border Layer: Authentic Indigenous Decorative Frame (indigframe.png)
                    if (indigFrame != null) {
                        gComp.drawImage(indigFrame, 0, 0, w, h, null);
                    }

                    gComp.dispose();
                }

                // Blit the composited background in 1 pass
                g.drawImage(cachedComposite, 0, 0, this);

                // 4. Foreground Layer: Prominent Game Title Logo (Title.png)
                if (titleImage != null) {
                    Graphics2D g2 = (Graphics2D) g.create();
                    g2.setRenderingHint(RenderingHints.KEY_INTERPOLATION, RenderingHints.VALUE_INTERPOLATION_BILINEAR);

                    // Resize title card and place it centered at the top (prominently larger & responsive)
                    int maxTitleW = (int) (w * 0.55);
                    int maxTitleH = (int) (h * 0.38);
                    int titleW = Math.min(maxTitleW, maxTitleH * 1163 / 586);
                    int titleH = titleW * 586 / 1163;
                    int titleX = (w - titleW) / 2;
                    int titleY = Math.max(16, (int) (h * 0.035));

                    g2.drawImage(titleImage, titleX, titleY, titleW, titleH, this);
                    g2.dispose();
                }
            }
        };
        backgroundPanel.setLayout(new BorderLayout());

        // Top bar for Player Card (top-left, cleanly inset inside indigenous frame)
        topBar.setOpaque(false);
        topBar.setBorder(BorderFactory.createEmptyBorder(18, 24, 0, 0));

        playerCardPanel = createPlayerCardPanel();
        topBar.add(playerCardPanel, BorderLayout.WEST);
        backgroundPanel.add(topBar, BorderLayout.NORTH);

        // Buttons from Juan Word Assets: CONTINUE, START / NEW GAME, SETTINGS, EXIT
        continueButton = new ImageAssetButton("continuebtn.png", 192, 48);
        styleImageButton(continueButton);
        continueButton.addActionListener(e -> handleContinueGame());

        startButton = new ImageAssetButton("startbtn.png", 192, 48);
        styleImageButton(startButton);
        startButton.addActionListener(e -> handleStartGame());

        settingButton = new ImageAssetButton("settingbtn.png", 192, 48);
        styleImageButton(settingButton);
        settingButton.addActionListener(e -> showSettingsDialog());

        exitButton = new ImageAssetButton("exitbtn.png", 192, 48);
        styleImageButton(exitButton);
        exitButton.addActionListener(e -> {
            saveGameOnExit();
            System.exit(0);
        });

        // Stack buttons vertically
        buttonPanel.removeAll();
        buttonPanel.add(continueButton);
        buttonPanel.add(startButton);
        buttonPanel.add(settingButton);
        buttonPanel.add(exitButton);

        // Center bottom buttonPanel horizontally
        bottomContainer.setOpaque(false);
        bottomContainer.setBorder(BorderFactory.createEmptyBorder(0, 0, 20, 0));
        bottomContainer.add(buttonPanel);

        backgroundPanel.add(bottomContainer, BorderLayout.SOUTH);
        updateContinueButton();
        return backgroundPanel;
    }

    // Creates the content panel for San Pedro Cathedral stages dialog with dynamic per-difficulty boss & stage unlock tracking
    public JPanel createSanPedroStagesContent(Player player, Runnable onClose, java.util.function.BiConsumer<Integer, String> onStartStage) {
        JPanel contentPanel = new JPanel(new BorderLayout(12, 12));
        contentPanel.setBackground(GamePalette.DEEP_INDIGO);

        // Header panel
        JPanel topPanel = new JPanel(new GridLayout(2, 1, 0, 4));
        topPanel.setOpaque(false);
        topPanel.setBorder(BorderFactory.createEmptyBorder(15, 20, 0, 20));

        JLabel titleLabel = new JLabel("SAN PEDRO CATHEDRAL", SwingConstants.CENTER);
        titleLabel.setFont(new Font("Arial", Font.BOLD, 20));
        titleLabel.setForeground(GamePalette.CREAM);

        JLabel subLabel = new JLabel("Pilia ang Yugto (Stage) ug Kalisod (Difficulty)", SwingConstants.CENTER);
        subLabel.setFont(new Font("Arial", Font.PLAIN, 13));
        subLabel.setForeground(GamePalette.SKY);

        topPanel.add(titleLabel);
        topPanel.add(subLabel);
        contentPanel.add(topPanel, BorderLayout.NORTH);

        // Center: Difficulty selection row + 3 Stage Cards
        JPanel centerPanel = new JPanel(new BorderLayout(0, 10));
        centerPanel.setOpaque(false);
        centerPanel.setBorder(BorderFactory.createEmptyBorder(0, 25, 10, 25));

        // Difficulty Selector Row
        JPanel diffRow = new JPanel(new FlowLayout(FlowLayout.CENTER, 10, 5));
        diffRow.setOpaque(false);
        JLabel diffLbl = new JLabel("Kalisod:");
        diffLbl.setFont(new Font("Arial", Font.BOLD, 13));
        diffLbl.setForeground(GamePalette.CREAM);
        diffRow.add(diffLbl);

        String[] diffs = {"Sayon", "Sakto", "Lisod"};
        Color[] diffColors = {GamePalette.FOREST_GREEN, GamePalette.PUMPKIN_ORANGE, GamePalette.CRIMSON_RED};

        String initDiff = (player != null && player.savedDifficulty != null && !player.savedDifficulty.trim().isEmpty())
                ? player.savedDifficulty.trim() : "Sakto";
        if ("easy".equalsIgnoreCase(initDiff)) initDiff = "Sayon";
        else if ("hard".equalsIgnoreCase(initDiff)) initDiff = "Lisod";
        else if (!"Sayon".equalsIgnoreCase(initDiff) && !"Lisod".equalsIgnoreCase(initDiff)) initDiff = "Sakto";

        final String[] selectedDiff = {initDiff};

        // 3 Stage Cards container
        JPanel stagesList = new JPanel(new GridLayout(3, 1, 0, 10));
        stagesList.setOpaque(false);

        String[][] stageData = {
            {"1", "DIZON", "The Box (Ang Kahoy nga Kahon)", "#D36F02"},
            {"2", "ALVIOLA", "The Steel Box (Ang Puthaw nga Kahon)", "#4DA6C5"},
            {"3", "ROLA", "The Boss Box (Ang Hari nga Kahon)", "#C62828"}
        };

        // Dynamically rebuild/refresh the 3 stage cards for the currently selected difficulty
        Runnable refreshStageCards = () -> {
            stagesList.removeAll();
            int maxUnlockedForDiff = (player != null) ? player.getMaxUnlockedStage(selectedDiff[0]) : 1;

            for (int s = 0; s < 3; s++) {
                int stageNum = s + 1;
                String enemyName = stageData[s][1];
                String stageDesc = stageData[s][2];
                String accentHex = stageData[s][3];
                boolean isStageUnlocked = (maxUnlockedForDiff >= stageNum);

                JPanel stageCard = new JPanel(new BorderLayout(10, 0));
                stageCard.setBackground(isStageUnlocked ? GamePalette.DUSK_NAVY : new Color(14, 15, 42));
                stageCard.setBorder(BorderFactory.createCompoundBorder(
                    BorderFactory.createLineBorder(isStageUnlocked ? Color.decode(accentHex) : new Color(28, 30, 60), 2, true),
                    BorderFactory.createEmptyBorder(8, 12, 8, 12)
                ));

                // Left: Stage info
                JPanel infoPanel = new JPanel(new GridLayout(2, 1, 0, 2));
                infoPanel.setOpaque(false);

                String statusTag = isStageUnlocked ? (maxUnlockedForDiff > stageNum ? "✔ Nahuman" : "⭐ Andam") : "🔒 Naka-lock";
                Color titleCol = isStageUnlocked ? GamePalette.CREAM : new Color(120, 125, 155);

                JLabel sTitle = new JLabel("<html><b>STAGE " + stageNum + ": " + enemyName + "</b> <font size='2' color='" + (isStageUnlocked ? "#F5B820" : "#D36F02") + "'>[" + statusTag + "]</font></html>");
                sTitle.setFont(new Font("Arial", Font.BOLD, 14));
                sTitle.setForeground(titleCol);

                JLabel sDesc = new JLabel(stageDesc);
                sDesc.setFont(new Font("Arial", Font.PLAIN, 12));
                sDesc.setForeground(isStageUnlocked ? GamePalette.SKY : new Color(90, 95, 125));

                infoPanel.add(sTitle);
                infoPanel.add(sDesc);
                stageCard.add(infoPanel, BorderLayout.CENTER);

                // Right: Play button
                FlatGameButton playBtn;
                if (isStageUnlocked) {
                    playBtn = new FlatGameButton("Sukla ▶", Color.decode(accentHex), GamePalette.CREAM, GamePalette.DUSK_NAVY);
                    playBtn.setPreferredSize(new Dimension(100, 34));
                    playBtn.setCornerRadius(8);
                    if (customCursor != null) playBtn.setCursor(customCursor);
                    playBtn.addActionListener(e -> {
                        if (onStartStage != null) {
                            onStartStage.accept(stageNum, selectedDiff[0]);
                        }
                    });
                } else {
                    playBtn = new FlatGameButton("🔒 Lock", new Color(30, 32, 60), new Color(110, 115, 145), new Color(20, 22, 45));
                    playBtn.setPreferredSize(new Dimension(100, 34));
                    playBtn.setCornerRadius(8);
                    playBtn.setEnabled(false);
                }
                stageCard.add(playBtn, BorderLayout.EAST);

                stagesList.add(stageCard);
            }
            stagesList.revalidate();
            stagesList.repaint();
        };

        FlatGameButton[] diffBtns = new FlatGameButton[diffs.length];
        for (int d = 0; d < diffs.length; d++) {
            final String dName = diffs[d];
            final Color dColor = diffColors[d];
            diffBtns[d] = new FlatGameButton(dName, dColor, GamePalette.CREAM, GamePalette.DUSK_NAVY);
            diffBtns[d].setPreferredSize(new Dimension(85, 30));
            diffBtns[d].setCornerRadius(8);
            if (customCursor != null) diffBtns[d].setCursor(customCursor);

            final int idx = d;
            diffBtns[d].addActionListener(e -> {
                selectedDiff[0] = dName;
                for (int j = 0; j < diffBtns.length; j++) {
                    if (j == idx) {
                        diffBtns[j].setText("• " + diffs[j] + " •");
                        diffBtns[j].setBorderColor(GamePalette.CREAM);
                    } else {
                        diffBtns[j].setText(diffs[j]);
                        diffBtns[j].setBorderColor(GamePalette.DUSK_NAVY);
                    }
                    diffBtns[j].repaint();
                }
                refreshStageCards.run(); // Instantly update Stage 1, 2, and 3 cards!
            });
            diffRow.add(diffBtns[d]);
        }

        // Set initial selected visual state
        for (int d = 0; d < diffs.length; d++) {
            if (diffs[d].equalsIgnoreCase(selectedDiff[0])) {
                diffBtns[d].setText("• " + diffs[d] + " •");
                diffBtns[d].setBorderColor(GamePalette.CREAM);
            } else {
                diffBtns[d].setText(diffs[d]);
                diffBtns[d].setBorderColor(GamePalette.DUSK_NAVY);
            }
        }

        centerPanel.add(diffRow, BorderLayout.NORTH);

        // Initial populate of stage cards
        refreshStageCards.run();

        centerPanel.add(stagesList, BorderLayout.CENTER);
        contentPanel.add(centerPanel, BorderLayout.CENTER);

        // Bottom: Cancel / Back button
        JPanel bottomBar = new JPanel(new FlowLayout(FlowLayout.CENTER));
        bottomBar.setOpaque(false);
        bottomBar.setBorder(BorderFactory.createEmptyBorder(0, 0, 12, 0));

        FlatGameButton closeBtn = new FlatGameButton("Balik sa Mapa", GamePalette.DUSK_NAVY, GamePalette.CREAM, GamePalette.SKY_BLUE);
        closeBtn.setPreferredSize(new Dimension(140, 34));
        closeBtn.setCornerRadius(8);
        if (customCursor != null) closeBtn.setCursor(customCursor);
        closeBtn.addActionListener(e -> {
            if (onClose != null) onClose.run();
        });
        bottomBar.add(closeBtn);

        contentPanel.add(bottomBar, BorderLayout.SOUTH);
        return contentPanel;
    }

    public void showSanPedroStagesDialog() {
        if (currentPlayer == null) {
            currentPlayer = dataManager.loadLatestPlayer();
        }

        JDialog dialog = new JDialog(this, "San Pedro Cathedral - Pilia ang Yugto", true);

        dialog.setLocationRelativeTo(this);
        if (customCursor != null) {
            dialog.setCursor(customCursor);
        }

        JPanel content = createSanPedroStagesContent(
            currentPlayer,
            dialog::dispose,
            (stageNum, diffName) -> {
                dialog.dispose();

                    currentPlayer.currentStage = stageNum;
                    currentPlayer.currentRound = 1;
                    currentPlayer.savedDifficulty = diffName;
                    currentPlayer.lastSaved = LocalDateTime.now().format(DateTimeFormatter.ofPattern("yyyy-MM-dd HH:mm"));
                    dataManager.savePlayer(currentPlayer);
                    updatePlayerCard();
                    updateContinueButton();
                }
                startBattle("San Pedro Cathedral", stageNum, diffName);
            }
        );

        dialog.setContentPane(content);
        dialog.setVisible(true);
    }

    private void updatePlayerCard() {
        if (playerNameLabel == null || playerInfoLabel == null) return;

            playerNameLabel.setText(currentPlayer.playerName);
            playerInfoLabel.setText("Stage " + currentPlayer.currentStage + " • HP " + currentPlayer.currentHealth + "/" + currentPlayer.maxHealth + " • " + currentPlayer.coins + "c ▾");

    // Check if an active game run exists for the player
    public boolean hasActiveGame(Player p) {
        if (p == null) return false;
        // An active game is any run where a stage/round was started (currentRound >= 1), progress was made, or saved battle exists
        return p.currentRound > 0 || p.score > 0 || p.currentStage > 1 || (p.savedTiles != null && !p.savedTiles.isEmpty());
    }

    // Refresh Continue button state (enabled vs grayed-out)
    private void updateContinueButton() {
        if (continueButton == null) return;
        boolean hasAnyActive = false;
        List<Player> all = dataManager.getAllPlayers();
        for (Player p : all) {
            if (hasActiveGame(p)) {
                hasAnyActive = true;
                break;
            }
        }
        continueButton.setEnabled(hasAnyActive);
        if (hasAnyActive) {
            if (currentPlayer != null && hasActiveGame(currentPlayer)) {
                continueButton.setToolTipText("Ipadayon ang dula alang kang " + currentPlayer.playerName + " (Stage " + currentPlayer.currentStage + ", HP " + currentPlayer.currentHealth + ")");
            } else {
                continueButton.setToolTipText("Ipadayon ang aktibo nga dula (Continue active game run)");
            }
            if (customCursor != null) {
                continueButton.setCursor(customCursor);
            }
            if (startButton != null) {
                startButton.setImage("newgamebtn.png");
            }
        } else {
            continueButton.setToolTipText("Walay aktibo nga dula (No active game in progress)");
            continueButton.setCursor(Cursor.getDefaultCursor());
            if (startButton != null) {
                startButton.setImage("startbtn.png");
            }
        }
        continueButton.repaint();
    }

    // Handler for START button: if active game exists, ask confirmation before resetting
    private void handleStartGame() {
        if (hasActiveGame(currentPlayer)) {
            String msg = "Naa kay kasamtangang aktibo nga dula alang kang " + currentPlayer.playerName + ":\n"
                    + "  • Stage " + currentPlayer.currentStage + ", Round " + currentPlayer.currentRound + "\n"
                    + "  • HP: " + currentPlayer.currentHealth + "/" + currentPlayer.maxHealth + "\n"
                    + "  • Score: " + currentPlayer.score + ", Coins: " + currentPlayer.coins + "\n\n"
                    + "Kung magsugod ka ug bag-ong dula, MA-RESET kini balik sa Stage 1 Round 1!\n"
                    + "Sigurado ka nga gusto nimo i-reset ug magsugod ug bag-o?\n\n"
                    + "(Starting a new game will reset your active run back to Stage 1, Round 1.)";

            int choice = JOptionPane.showConfirmDialog(
                this,
                msg,
                "Kumpirmaha ang Pag-reset (Reset Active Game?)",
                JOptionPane.YES_NO_OPTION,
                JOptionPane.WARNING_MESSAGE
            );

            if (choice != JOptionPane.YES_OPTION) {
                return; // User cancelled, do not reset and do not switch screen
            }

            // User confirmed: reset run back to fresh inactive state (currentRound = 0)
            currentPlayer.currentStage = 1;
            currentPlayer.currentRound = 0;
            currentPlayer.score = 0;
            currentPlayer.experience = 0;
            currentPlayer.coins = 0;
            currentPlayer.currentHealth = currentPlayer.maxHealth;
            currentPlayer.savedTiles = "";
            currentPlayer.savedEnemyHealth = -1;
            currentPlayer.lastSaved = LocalDateTime.now().format(DateTimeFormatter.ofPattern("yyyy-MM-dd HH:mm"));
            dataManager.savePlayer(currentPlayer);
            updatePlayerCard();
            updateContinueButton();
        }

        // Proceed to map select with entrance animation
        if (mapSelectPanel != null) {
            mapSelectPanel.startEntranceAnimation();
        }
        cardLayout.show(mainContainer, "MAP_SELECT");
    }

    // Handler for CONTINUE button: resume active game directly or choose between multiple active save files
    private void handleContinueGame() {
        List<Player> allPlayers = dataManager.getAllPlayers();
        List<Player> activePlayers = new ArrayList<>();
        for (Player p : allPlayers) {
            if (hasActiveGame(p)) {
                activePlayers.add(p);
            }
        }

        if (activePlayers.isEmpty()) {
            JOptionPane.showMessageDialog(this, 
                "Walay aktibo nga dula sa bisan unsang save file.\nI-klik ang START aron makasugod ug bag-ong dula!", 
                "Walay Aktibo nga Dula", 
                JOptionPane.INFORMATION_MESSAGE);
            return;
        }

        // If multiple save files have active runs, show modal to let the user select which one to continue!
        if (activePlayers.size() > 1) {
            showContinueSaveFileDialog(activePlayers);
            return;
        }

        // Exactly one save file has an active run: resume directly!
        resumeSelectedPlayerGame(activePlayers.get(0));
    }

    private void resumeSelectedPlayerGame(Player player) {
        if (player == null) return;
        this.currentPlayer = player;
        updatePlayerCard();
        updateContinueButton();
        battlePanel.resumeBattle(currentPlayer);
        cardLayout.show(mainContainer, "BATTLE");
    }

        battlePanel.setupBattle(currentPlayer, mapName, stageNum, diffName);
        cardLayout.show(mainContainer, "BATTLE");
    }

    public void returnToMapSelect() {
        if (battlePanel != null) {
            battlePanel.saveGameState();
            Player bp = battlePanel.getCurrentPlayer();
            if (bp != null) {
                currentPlayer = bp;
            }
        }
        if (currentPlayer != null) {
            Player fresh = dataManager.loadPlayer(currentPlayer.playerId);
            if (fresh != null) {
                currentPlayer = fresh;
            }
        }
        if (mapSelectPanel != null) {
            mapSelectPanel.startEntranceAnimation();
        }
        cardLayout.show(mainContainer, "MAP_SELECT");
        updatePlayerCard();
        updateContinueButton();
    }

        if (battlePanel != null) {
            battlePanel.saveGameState();

            dataManager.savePlayer(currentPlayer);
        }
    }

    // Helper method to load images safely from root or classpath
    public Image loadGameImage(String filename) {
        try {
            File f = new File(filename);
            if (f.exists()) {
                return ImageIO.read(f);
            }
            java.net.URL url = getClass().getResource("/juanword/" + filename);
            if (url != null) {
                return ImageIO.read(url);
            }
            url = getClass().getResource("/" + filename);
            if (url != null) {
                return ImageIO.read(url);
            }
        } catch (Exception ignored) {}
        return null;
    }

    private class MapTileButton extends JButton {
        private boolean isHovered = false;
        private final String title;
        private final String subtitle;
        private final boolean isUnlocked;

        public MapTileButton(String title, String subtitle, boolean isUnlocked) {
            this.title = title;

            this.isUnlocked = isUnlocked;
            updateContent();
            setFocusPainted(false);
            setContentAreaFilled(false);
            setBorderPainted(false);
            if (isUnlocked && customCursor != null) {
                setCursor(customCursor);

            if (isUnlocked) {
                addMouseListener(new java.awt.event.MouseAdapter() {
                    @Override
                    public void mouseEntered(java.awt.event.MouseEvent e) {
                        isHovered = true;
                        updateContent();
                        repaint();
                    }

                    @Override
                    public void mouseExited(java.awt.event.MouseEvent e) {
                        isHovered = false;
                        updateContent();
                        repaint();
                    }
                });
            }
        }

        // Backward compatibility constructor
        public MapTileButton(String title, String subtitle) {
            this(title, subtitle, true);
        }

        private void updateContent() {
            if (!isUnlocked) {
                setText("<html><center><font size='4' color='#6C7293'><b>" + title + "</b></font><br>"
                        + "<font size='2' color='#4A4E6B'>" + subtitle + "</font><br>"
                        + "<font size='2' color='#D36F02'>🔒 <b>Naka-lock</b></font></center></html>");
            } else if (isHovered) {
                setText("<html><center><font size='5' color='#FFF8E9'><b>" + title + "</b></font><br>"
                        + "<font size='3' color='#9EE6FE'>" + subtitle + "</font><br>"
                        + "<font size='2' color='#F5B820'>⭐ <b>3 ka Yugto (Dizon, Alviola, Rola)</b></font></center></html>");
            } else {
                setText("<html><center><font size='4' color='#FFF8E9'><b>" + title + "</b></font><br>"
                        + "<font size='2' color='#9EE6FE'>" + subtitle + "</font><br>"
                        + "<font size='2' color='#387940'>✔ <b>Nabuksan (3 ka Yugto)</b></font></center></html>");
            }
        }

        @Override
        protected void paintComponent(Graphics g) {
            Graphics2D g2 = (Graphics2D) g.create();
            g2.setRenderingHint(RenderingHints.KEY_ANTIALIASING, RenderingHints.VALUE_ANTIALIAS_ON);

            if (!isUnlocked) {
                // Grayed-out locked map card: dark desaturated navy
                int pad = 10;
                int x = pad;
                int y = pad;
                int w = getWidth() - (pad * 2);
                int h = getHeight() - (pad * 2);

                g2.setColor(new Color(11, 11, 32));
                g2.fillRoundRect(x, y, w, h, 16, 16);

                g2.setColor(new Color(25, 27, 56));
                g2.setStroke(new BasicStroke(1.5f));
                g2.drawRoundRect(x, y, w, h, 16, 16);
            } else {
                // Normal: 10px margin. Hovered: expands outward to 2px margin.
                int pad = isHovered ? 2 : 10;
                int x = pad;
                int y = pad;
                int w = getWidth() - (pad * 2);
                int h = getHeight() - (pad * 2);

                // Background card: Deep Indigo normal, Dusk Navy hovered
                g2.setColor(isHovered ? GamePalette.DUSK_NAVY : GamePalette.DEEP_INDIGO);
                g2.fillRoundRect(x, y, w, h, 16, 16);

                // Border: Cerulean hovered, Dusk Navy normal

                    g2.setColor(GamePalette.CERULEAN);
                    g2.setStroke(new BasicStroke(2.5f));
                } else {
                    g2.setColor(GamePalette.DUSK_NAVY);
                    g2.setStroke(new BasicStroke(1.5f));
                }
                g2.drawRoundRect(x, y, w, h, 16, 16);
            }

            g2.dispose();
            super.paintComponent(g);
        }
    }

/**
 * Interactive Map Selection Screen for JuanWord.
 *
 * Game Architecture & Staging:
 * 1. Back Layer: Sky background (bg.png) + Towering Mount Apo (mtapo.png - Map 4, Extreme)
 * 2. Foreground Landmarks (Left to Right):
 *    - Map 1 (Left): Blue Building (bluebuilding.png - Predefined Difficulty: Easy / Sayon)
 *    - Map 2 (Center): Davao Museum (durianmuseum.png - Predefined Difficulty: Medium / Sakto)
 *    - Map 3 (Right): San Pedro Cathedral (sanpedro.png - Predefined Difficulty: Hard / Lisod)
 * 3. Silhouette Rendering: Locked maps are rendered as dark obsidian silhouettes with a lock emblem.
 * 4. Predefined Difficulty: Clicking an unlocked map launches directly into battle with that map's
 *    fixed difficulty (Easy, Medium, Hard, Extreme) without requiring manual difficulty selection.
 * 5. Interactive Animation & Hover:
 *    - Entrance: Landmarks rise smoothly from the bottom of the screen.
 *    - Hover: Expands hovered map (+7% anchored at base) with a custom luminous HUD badge.
 */
public class MapSelectPanel extends JPanel {

    private final JuanWord app;

    // Background Layers
    private BufferedImage skyBg;
    private BufferedImage mtApo;
    private BufferedImage indigFrame;

    // Foreground Landmark Sprites (Cropped tight to visual artwork)
    private BufferedImage blueBuildingImg;
    private BufferedImage durianMuseumImg;
    private BufferedImage sanPedroImg;

    // Silhouettes for locked maps
    private BufferedImage silBlueBuilding;
    private BufferedImage silDurianMuseum;
    private BufferedImage silSanPedro;
    private BufferedImage silMtApo;

    // Landmarks Configuration
    // 0: Blue Building (Easy), 1: Davao Museum (Medium), 2: San Pedro (Hard), 3: Mt Apo (Extreme)
    public static final int LANDMARK_BLUE_BUILDING = 0;
    public static final int LANDMARK_DURIAN_MUSEUM = 1;
    public static final int LANDMARK_SAN_PEDRO = 2;
    public static final int LANDMARK_MT_APO = 3;
    public static final int LANDMARK_COUNT = 4;

    // Animation & Hover States
    private float animProgress = 1.0f; // 0.0 -> 1.0
    private Timer entranceTimer;
    private Timer hoverTimer;

    // Level Selection Expand Transition
    private boolean isExpanding = false;
    private int expandingLandmark = -1;
    private float expandProgress = 0.0f; // 0.0 -> 1.0
    private Timer expandTimer;

    private int hoveredLandmark = -1;
    private final float[] currentScale = new float[]{1.0f, 1.0f, 1.0f, 1.0f};
    private final float[] targetScale = new float[]{1.0f, 1.0f, 1.0f, 1.0f};

    // Landmark Layout Rectangles (Computed during paint)
    private final Rectangle[] landmarkBounds = new Rectangle[LANDMARK_COUNT];

    // Navigation Bounds
    private final Rectangle backButtonBounds = new Rectangle(28, 22, 120, 36);
    private final Rectangle settingsButtonBounds = new Rectangle(0, 22, 38, 38);
    private final Rectangle fullscreenButtonBounds = new Rectangle(0, 22, 38, 38);
    private BufferedImage settingsIcon;

    public MapSelectPanel(JuanWord app) {
        this.app = app;
        setLayout(new BorderLayout());
        setOpaque(true);
        setBackground(GamePalette.DEEP_INDIGO);

        loadAssets();
        initInteractions();

        for (int i = 0; i < LANDMARK_COUNT; i++) {
            landmarkBounds[i] = new Rectangle();
        }
    }

    private void loadAssets() {
        skyBg = toBufferedImage(app.loadGameImage("bg.png"));
        mtApo = toBufferedImage(app.loadGameImage("mtapo.png"));
        indigFrame = toBufferedImage(app.loadGameImage("indigframe.png"));
        settingsIcon = toBufferedImage(app.loadGameImage("Cog Setting.png"));

        // Blue Building (Map 1: 1024x512 raw, cropped [0,0 to 648,512])
        BufferedImage bbRaw = toBufferedImage(app.loadGameImage("bluebuilding.png"));
        if (bbRaw == null) {
            bbRaw = toBufferedImage(app.loadGameImage("Blue Building.png"));
        }
        if (bbRaw != null) {
            int cropW = Math.min(648, bbRaw.getWidth());
            int cropH = Math.min(512, bbRaw.getHeight());
            blueBuildingImg = bbRaw.getSubimage(0, 0, cropW, cropH);
        }

        // Davao Museum (Map 2: raw 1080x1920, cropped [0, 300, 1080, 1620])
        BufferedImage dmRaw = toBufferedImage(app.loadGameImage("durianmuseum.png"));
        if (dmRaw != null) {
            int subY = Math.min(300, dmRaw.getHeight() - 1);
            int subH = Math.min(1620, dmRaw.getHeight() - subY);
            durianMuseumImg = dmRaw.getSubimage(0, subY, dmRaw.getWidth(), subH);
        }

        // San Pedro Cathedral (Map 3: raw 1612x1066, cropped [0, 201, 1612, 865])
        BufferedImage spRaw = toBufferedImage(app.loadGameImage("sanpedro.png"));
        if (spRaw != null) {
            int subY = Math.min(201, spRaw.getHeight() - 1);
            int subH = Math.min(865, spRaw.getHeight() - subY);
            sanPedroImg = spRaw.getSubimage(0, subY, spRaw.getWidth(), subH);
        }

        // Generate high-quality silhouettes for locked representation
        Color silColor = new Color(14, 18, 28, 252);
        Color apoSilColor = new Color(18, 24, 36, 245);
        if (blueBuildingImg != null) silBlueBuilding = createSilhouette(blueBuildingImg, silColor);
        if (durianMuseumImg != null) silDurianMuseum = createSilhouette(durianMuseumImg, silColor);
        if (sanPedroImg != null) silSanPedro = createSilhouette(sanPedroImg, silColor);
        if (mtApo != null) silMtApo = createSilhouette(mtApo, apoSilColor);
    }

    private BufferedImage toBufferedImage(Image img) {
        if (img == null) return null;
        if (img instanceof BufferedImage) return (BufferedImage) img;
        int w = img.getWidth(null);
        int h = img.getHeight(null);
        if (w <= 0 || h <= 0) return null;
        BufferedImage bImg = new BufferedImage(w, h, BufferedImage.TYPE_INT_ARGB);
        Graphics2D g2 = bImg.createGraphics();
        g2.drawImage(img, 0, 0, null);
        g2.dispose();
        return bImg;
    }

    private BufferedImage createSilhouette(BufferedImage src, Color color) {
        if (src == null) return null;
        BufferedImage sil = new BufferedImage(src.getWidth(), src.getHeight(), BufferedImage.TYPE_INT_ARGB);
        Graphics2D g = sil.createGraphics();
        g.drawImage(src, 0, 0, null);
        g.setComposite(AlphaComposite.SrcIn);
        g.setColor(color);
        g.fillRect(0, 0, src.getWidth(), src.getHeight());
        g.dispose();
        return sil;
    }

    public void startEntranceAnimation() {
        if (entranceTimer != null && entranceTimer.isRunning()) {
            entranceTimer.stop();
        }
        animProgress = 0.0f;
        // 60 FPS animation timer (approx 600ms duration)
        final long startTime = System.currentTimeMillis();
        final long duration = 650;

        entranceTimer = new Timer(16, e -> {
            long elapsed = System.currentTimeMillis() - startTime;
            if (elapsed >= duration) {
                animProgress = 1.0f;
                entranceTimer.stop();
            } else {
                animProgress = (float) elapsed / duration;
            }
            repaint();
        });
        entranceTimer.start();
    }

    private void initInteractions() {
        // Continuous smooth hover interpolation timer (60 FPS)
        hoverTimer = new Timer(16, e -> {
            boolean needRepaint = false;
            for (int i = 0; i < LANDMARK_COUNT; i++) {
                float target = (i == hoveredLandmark) ? 1.07f : 1.0f;
                targetScale[i] = target;
                if (Math.abs(currentScale[i] - target) > 0.002f) {
                    currentScale[i] += (target - currentScale[i]) * 0.25f;
                    needRepaint = true;
                } else {
                    currentScale[i] = target;
                }
            }
            if (needRepaint) {
                repaint();
            }
        });
        hoverTimer.start();

        addMouseMotionListener(new MouseMotionAdapter() {
            @Override
            public void mouseMoved(MouseEvent e) {
                if (isExpanding) return;
                Point p = e.getPoint();

                // Check Navigation controls first
                if (backButtonBounds.contains(p) || settingsButtonBounds.contains(p) || fullscreenButtonBounds.contains(p)) {
                    if (hoveredLandmark != -1) {
                        hoveredLandmark = -1;
                        repaint();
                    }
                    setCursor(Cursor.getPredefinedCursor(Cursor.HAND_CURSOR));
                    return;
                }

                int prevHover = hoveredLandmark;
                int newHover = hitTestLandmark(p.x, p.y);

                if (newHover != prevHover) {
                    hoveredLandmark = newHover;
                    if (hoveredLandmark >= 0) {
                        setCursor(Cursor.getPredefinedCursor(Cursor.HAND_CURSOR));
                    } else {
                        setCursor(Cursor.getDefaultCursor());
                    }
                    repaint();
                }
            }
        });

        addMouseListener(new MouseAdapter() {
            @Override
            public void mouseClicked(MouseEvent e) {
                if (isExpanding) return;
                Point p = e.getPoint();

                // Back Button clicked
                if (backButtonBounds.contains(p)) {
                    app.showStartScreen();
                    return;
                }

                // Settings Button clicked
                if (settingsButtonBounds.contains(p)) {
                    app.showSettingsDialog();
                    return;
                }

                // Fullscreen Button clicked
                if (fullscreenButtonBounds.contains(p)) {
                    app.toggleFullScreen();
                    return;
                }

                // Landmark clicked
                int clicked = hitTestLandmark(p.x, p.y);
                if (clicked >= 0) {
                    handleLandmarkClick(clicked);
                }
            }

            @Override
            public void mouseExited(MouseEvent e) {
                if (hoveredLandmark != -1) {
                    hoveredLandmark = -1;
                    repaint();
                }
                setCursor(Cursor.getDefaultCursor());
            }
        });
    }

    /**
     * Hit tests all 4 landmarks with priority given to foreground buildings,
     * followed by Mount Apo in the background.
     */
    private int hitTestLandmark(int mouseX, int mouseY) {
        int w = getWidth();
        int h = getHeight();

        // 1. Check Foreground Buildings: Blue Building (0), Davao Museum (1), San Pedro (2)
        // Check in visual order of overlap (Davao Museum is central foreground)
        int[] checkOrder = {LANDMARK_DURIAN_MUSEUM, LANDMARK_BLUE_BUILDING, LANDMARK_SAN_PEDRO};
        for (int idx : checkOrder) {
            Rectangle r = landmarkBounds[idx];
            if (r != null && r.contains(mouseX, mouseY)) {
                BufferedImage sprite = getSpriteForLandmark(idx);
                if (sprite != null) {
                    // Check alpha pixel
                    double relX = (double) (mouseX - r.x) / r.width;
                    double relY = (double) (mouseY - r.y) / r.height;
                    int imgX = (int) (relX * sprite.getWidth());
                    int imgY = (int) (relY * sprite.getHeight());
                    if (imgX >= 0 && imgX < sprite.getWidth() && imgY >= 0 && imgY < sprite.getHeight()) {
                        int pixel = sprite.getRGB(imgX, imgY);
                        int alpha = (pixel >> 24) & 0xff;
                        if (alpha > 30) {
                            return idx;
                        }
                    }
                }
            }
        }

        // 2. Check Background Landmark: Mount Apo (3)
        // If not hitting any foreground building, test mountain visible region (y < h * 0.72)
        if (mouseY > 60 && mouseY < h * 0.72) {
            if (mtApo != null) {
                int apoImgX = (int) ((double) mouseX / w * mtApo.getWidth());
                int apoImgY = (int) ((double) mouseY / h * mtApo.getHeight());
                if (apoImgX >= 0 && apoImgX < mtApo.getWidth() && apoImgY >= 0 && apoImgY < mtApo.getHeight()) {
                    int pixel = mtApo.getRGB(apoImgX, apoImgY);
                    int alpha = (pixel >> 24) & 0xff;
                    if (alpha > 40) {
                        return LANDMARK_MT_APO;
                    }
                }
            }
        }

        return -1;
    }

    private BufferedImage getSpriteForLandmark(int idx) {
        switch (idx) {
            case LANDMARK_BLUE_BUILDING: return blueBuildingImg;
            case LANDMARK_DURIAN_MUSEUM: return durianMuseumImg;
            case LANDMARK_SAN_PEDRO: return sanPedroImg;
            case LANDMARK_MT_APO: return mtApo;
            default: return null;
        }
    }

    private void handleLandmarkClick(int landmarkIdx) {
        Player player = app.getCurrentPlayer();
        int maxUnlocked = (player != null) ? player.unlockedMapIndex : 1;

        int requiredMapIdx = landmarkIdx + 1; // 1: Blue Building, 2: Davao Museum, 3: San Pedro, 4: Mt Apo
        boolean isUnlocked = (maxUnlocked >= requiredMapIdx);

        String mapName;
        String predefinedDiff;
        switch (landmarkIdx) {
            case LANDMARK_BLUE_BUILDING:
                mapName = "Blue Building";
                predefinedDiff = "Sayon"; // Easy
                break;
            case LANDMARK_DURIAN_MUSEUM:
                mapName = "Davao Museum";
                predefinedDiff = "Sakto"; // Medium
                break;
            case LANDMARK_SAN_PEDRO:
                mapName = "San Pedro Cathedral";
                predefinedDiff = "Lisod"; // Hard
                break;
            case LANDMARK_MT_APO:
                mapName = "Mt. Apo";
                predefinedDiff = "Extreme"; // Extreme
                break;
            default:
                return;
        }

        if (isUnlocked) {
            // Trigger expand transition, which smoothly scales up the landmark and then starts battle!
            triggerExpandTransition(landmarkIdx, mapName, predefinedDiff);
        } else {
            // Show Dabawenyo themed lock notification
            String prevMapName = (landmarkIdx == LANDMARK_DURIAN_MUSEUM) ? "Blue Building (Easy)"
                               : (landmarkIdx == LANDMARK_SAN_PEDRO) ? "Davao Museum (Medium)"
                               : "San Pedro Cathedral (Hard)";
            String lockMessage = "🔒 NAKA-LOCK PA KINI NGA MAPA!\n\n"
                    + "Kinahanglan nimo pildihon una ang \"" + prevMapName + "\"\n"
                    + "aron maablihan ang bag-ong mapa ug ang predefined difficulty niini (" + predefinedDiff + ").\n\n"
                    + "Padayon sa pagkat-on ug pagduwa, Juan!";

            JOptionPane.showMessageDialog(
                    this,
                    lockMessage,
                    "Mapa Naka-Lock (" + mapName + ")",
                    JOptionPane.WARNING_MESSAGE
            );
        }
    }

    public boolean isExpanding() {

        return expandProgress;

        return expandingLandmark;

    public void setExpandProgressForTesting(int landmarkIdx, float progress) {
        this.isExpanding = (progress < 1.0f);
        this.expandingLandmark = landmarkIdx;
        this.expandProgress = progress;
        repaint();
    }

    public void triggerExpandTransition(int landmarkIdx, String mapName, String predefinedDiff) {
        if (isExpanding) return;
        isExpanding = true;
        expandingLandmark = landmarkIdx;
        expandProgress = 0.0f;

        // Reset hover and cursor
        hoveredLandmark = -1;
        setCursor(Cursor.getDefaultCursor());

        final int durationMs = 320;
        final int frameDelay = 16; // ~60fps
        final long startTime = System.currentTimeMillis();

        if (expandTimer != null && expandTimer.isRunning()) {
            expandTimer.stop();
        }

        expandTimer = new Timer(frameDelay, e -> {
            long elapsed = System.currentTimeMillis() - startTime;
            float rawT = Math.min(1.0f, (float) elapsed / durationMs);
            expandProgress = rawT;
            repaint();

            if (rawT >= 1.0f) {
                ((Timer) e.getSource()).stop();
                isExpanding = false;
                expandProgress = 0.0f;
                expandingLandmark = -1;
                app.startBattle(mapName, 1, predefinedDiff);
            }
        });
        expandTimer.start();
    }

    @Override
    protected void paintComponent(Graphics g) {
        super.paintComponent(g);
        Graphics2D g2 = (Graphics2D) g.create();
        g2.setRenderingHint(RenderingHints.KEY_ANTIALIASING, RenderingHints.VALUE_ANTIALIAS_ON);
        g2.setRenderingHint(RenderingHints.KEY_INTERPOLATION, RenderingHints.VALUE_INTERPOLATION_BILINEAR);

        int w = getWidth();
        int h = getHeight();
        if (w <= 0 || h <= 0) {
            g2.dispose();
            return;
        }

        // 1. Sky Background (Layer 1: bg.png)
        if (skyBg != null) {
            g2.drawImage(skyBg, 0, 0, w, h, null);
        }

        // Get player progress for lock/unlock states
        Player player = app.getCurrentPlayer();
        int maxUnlocked = (player != null) ? player.unlockedMapIndex : 1;

        // Entrance animation ease-out cubic
        float ease = (float) (1.0 - Math.pow(1.0 - animProgress, 3));
        float maxRiseOffset = h * 0.72f;
        int animOffset = (int) (maxRiseOffset * (1.0f - ease));

        float S = Math.min((float) w / 960f, (float) h / 540f);
        int stageW = Math.min(w, (int) (h * 16.0 / 9.0));
        int stageOffset = (w - stageW) / 2;

        // Compute base layout coordinates for all 4 landmarks
        // Map 1: Blue Building
        boolean isBbUnlocked = (maxUnlocked >= 1);
        BufferedImage bbSprite = isBbUnlocked ? blueBuildingImg : (silBlueBuilding != null ? silBlueBuilding : blueBuildingImg);
        int bbH = (int) (h * 0.44);
        int bbW = (bbSprite != null) ? (int) (bbH * (bbSprite.getWidth() / (double) bbSprite.getHeight())) : (int) (stageW * 0.28);
        int bbX = stageOffset + (int) (stageW * 0.025);
        int bbY = h - bbH - (int) (18 * S) + animOffset;
        landmarkBounds[LANDMARK_BLUE_BUILDING].setBounds(bbX, bbY, bbW, bbH);

        // Map 2: Davao Museum
        boolean isDmUnlocked = (maxUnlocked >= 2);
        BufferedImage dmSprite = isDmUnlocked ? durianMuseumImg : (silDurianMuseum != null ? silDurianMuseum : durianMuseumImg);
        int dmW = (int) (stageW * 0.32);
        int dmH = (int) (h * 0.46);
        int dmX = stageOffset + (stageW - dmW) / 2 - (int) (10 * S);
        int dmY = h - dmH - (int) (18 * S) + animOffset;
        landmarkBounds[LANDMARK_DURIAN_MUSEUM].setBounds(dmX, dmY, dmW, dmH);

        // Map 3: San Pedro Cathedral
        boolean isSpUnlocked = (maxUnlocked >= 3);
        BufferedImage spSprite = isSpUnlocked ? sanPedroImg : (silSanPedro != null ? silSanPedro : sanPedroImg);
        int spH = (int) (h * 0.48);
        int spW = (spSprite != null) ? (int) (spH * (spSprite.getWidth() / (double) spSprite.getHeight())) : (int) (stageW * 0.30);
        int spX = stageOffset + stageW - spW + (int) (50 * S);
        int spY = h - spH - (int) (18 * S) + animOffset;
        landmarkBounds[LANDMARK_SAN_PEDRO].setBounds(spX, spY, spW, spH);

        // Map 4: Mount Apo
        boolean isMtApoUnlocked = (maxUnlocked >= 4);
        BufferedImage apoSprite = isMtApoUnlocked ? mtApo : (silMtApo != null ? silMtApo : mtApo);

        if (!isExpanding) {
            // Normal Map Selection Rendering
            // 2. Mount Apo at the Back Layer
            if (apoSprite != null) {
                float apoScale = currentScale[LANDMARK_MT_APO];
                if (apoScale > 1.001f) {
                    int sw = (int) (w * apoScale);
                    int sh = (int) (h * apoScale);
                    g2.drawImage(apoSprite, (w - sw) / 2, (h - sh) / 2, sw, sh, null);
                } else {
                    g2.drawImage(apoSprite, 0, 0, w, h, null);
                }

                if (!isMtApoUnlocked) {
                    int apoLockSize = 34;
                    int apoLockX = w / 2 - apoLockSize / 2;
                    int apoLockY = (int) (h * 0.28);
                    drawLockEmblem(g2, apoLockX, apoLockY, apoLockSize);
                }
            }

            // 3. Foreground Buildings
            if (bbSprite != null) {
                drawBuilding(g2, bbSprite, bbX, bbY, bbW, bbH, currentScale[LANDMARK_BLUE_BUILDING], isBbUnlocked);
            }
            if (dmSprite != null) {
                drawBuilding(g2, dmSprite, dmX, dmY, dmW, dmH, currentScale[LANDMARK_DURIAN_MUSEUM], isDmUnlocked);
            }
            if (spSprite != null) {
                drawBuilding(g2, spSprite, spX, spY, spW, spH, currentScale[LANDMARK_SAN_PEDRO], isSpUnlocked);
            }

            // 4. Indigenous Frame (Top layer framing the viewport)
            if (indigFrame != null) {
                g2.drawImage(indigFrame, 0, 0, w, h, null);
            }

            // 5. Navigation & HUD Overlay
            drawNavigationAndHUD(g2, w, h, maxUnlocked);
        } else {
            // EXPAND TRANSITION ANIMATION:
            // The selected icon smoothly expands into the battle background position,
            // while non-selected landmarks and HUD fade away!
            float t = (float) (1.0 - Math.pow(1.0 - expandProgress, 3)); // ease-out cubic
            float fadeAlpha = Math.max(0.0f, 1.0f - expandProgress * 2.2f);

            // A. Draw non-selected landmarks fading out
            if (fadeAlpha > 0.005f) {
                Graphics2D gFade = (Graphics2D) g2.create();
                gFade.setComposite(AlphaComposite.getInstance(AlphaComposite.SRC_OVER, fadeAlpha));

                if (expandingLandmark != LANDMARK_MT_APO && apoSprite != null) {
                    gFade.drawImage(apoSprite, 0, 0, w, h, null);
                    if (!isMtApoUnlocked) {
                        drawLockEmblem(gFade, w / 2 - 17, (int) (h * 0.28), 34);
                    }
                }
                if (expandingLandmark != LANDMARK_BLUE_BUILDING && bbSprite != null) {
                    drawBuilding(gFade, bbSprite, bbX, bbY, bbW, bbH, 1.0f, isBbUnlocked);
                }
                if (expandingLandmark != LANDMARK_DURIAN_MUSEUM && dmSprite != null) {
                    drawBuilding(gFade, dmSprite, dmX, dmY, dmW, dmH, 1.0f, isDmUnlocked);
                }
                if (expandingLandmark != LANDMARK_SAN_PEDRO && spSprite != null) {
                    drawBuilding(gFade, spSprite, spX, spY, spW, spH, 1.0f, isSpUnlocked);
                }
                gFade.dispose();
            }

            // B. Draw expanding landmark scaling up and morphing into Battle background position
            BufferedImage activeSprite = getSpriteForLandmark(expandingLandmark);
            if (activeSprite != null) {
                int startX, startY, startW, startH;
                int targetX, targetY, targetW, targetH;

                if (expandingLandmark == LANDMARK_MT_APO) {
                    startX = 0; startY = 0; startW = w; startH = h;
                    targetW = (int) (w * 1.15f); targetH = (int) (h * 1.15f);
                    targetX = (w - targetW) / 2; targetY = (h - targetH) / 2;
                } else {
                    if (expandingLandmark == LANDMARK_BLUE_BUILDING) {
                        startX = bbX; startY = bbY; startW = bbW; startH = bbH;
                    } else if (expandingLandmark == LANDMARK_DURIAN_MUSEUM) {
                        startX = dmX; startY = dmY; startW = dmW; startH = dmH;
                    } else {
                        startX = spX; startY = spY; startW = spW; startH = spH;
                    }

                    int imgW = activeSprite.getWidth();
                    int imgH = activeSprite.getHeight();
                    double scale = Math.min((double) (w - 100) / imgW, (double) (h - 60) / imgH) * 1.08;
                    targetW = (int) Math.round(imgW * scale);
                    targetH = (int) Math.round(imgH * scale);
                    targetX = (w - targetW) / 2;
                    targetY = h - targetH - 22;
                }

                int curX = (int) (startX + (targetX - startX) * t);
                int curY = (int) (startY + (targetY - startY) * t);
                int curW = (int) (startW + (targetW - startW) * t);
                int curH = (int) (startH + (targetH - startH) * t);

                g2.drawImage(activeSprite, curX, curY, curW, curH, null);
            }

            // C. Indigenous Border Frame stays crisp on top of the landmark
            if (indigFrame != null) {
                g2.drawImage(indigFrame, 0, 0, w, h, null);
            }

            // D. Soft battle scrim fading in as transition completes
            if (expandProgress > 0.20f) {
                float scrimProgress = (expandProgress - 0.20f) / 0.80f;
                int alpha = (int) (scrimProgress * 95);
                g2.setColor(new Color(12, 16, 52, Math.min(95, alpha)));
                g2.fillRect(0, 0, w, h);
            }

            // E. Navigation & HUD fade out rapidly
            float hudAlpha = Math.max(0.0f, 1.0f - expandProgress * 2.8f);
            if (hudAlpha > 0.01f) {
                Graphics2D gHud = (Graphics2D) g2.create();
                gHud.setComposite(AlphaComposite.getInstance(AlphaComposite.SRC_OVER, hudAlpha));
                drawNavigationAndHUD(gHud, w, h, maxUnlocked);
                gHud.dispose();
            }
        }

        g2.dispose();
    }

    private void drawBuilding(Graphics2D g2, BufferedImage img, int x, int y, int w, int h, float scale, boolean unlocked) {
        int drawW = (int) (w * scale);
        int drawH = (int) (h * scale);
        int drawX = x - (drawW - w) / 2;
        int drawY = y - (drawH - h); // Grounded baseline

        if (scale > 1.002f) {
            // Luminous ambient glow on hover
            Graphics2D gGlow = (Graphics2D) g2.create();
            gGlow.setComposite(AlphaComposite.getInstance(AlphaComposite.SRC_OVER, unlocked ? 0.35f : 0.20f));
            gGlow.drawImage(img, drawX - 3, drawY - 3, drawW + 6, drawH + 6, null);
            gGlow.dispose();
        }

        g2.drawImage(img, drawX, drawY, drawW, drawH, null);

        // If locked, render the lock emblem badge centered over the silhouette
        if (!unlocked) {
            int lockSize = 36;
            int lockX = drawX + (drawW - lockSize) / 2;
            int lockY = drawY + (drawH - lockSize) / 2;
            drawLockEmblem(g2, lockX, lockY, lockSize);
        }
    }

    private void drawLockEmblem(Graphics2D g2, int lockX, int lockY, int lockSize) {
        Graphics2D gLock = (Graphics2D) g2.create();
        gLock.setRenderingHint(RenderingHints.KEY_ANTIALIASING, RenderingHints.VALUE_ANTIALIAS_ON);
        gLock.setRenderingHint(RenderingHints.KEY_TEXT_ANTIALIASING, RenderingHints.VALUE_TEXT_ANTIALIAS_ON);

        // Lock icon rendered cleanly over the silhouette without the yellow circular badge
        int fontSize = Math.max(16, (int) (lockSize * 0.78));
        gLock.setFont(new Font("Segoe UI Emoji", Font.PLAIN, fontSize));
        // Soft drop shadow for crisp depth
        gLock.setColor(new Color(0, 0, 0, 180));
        gLock.drawString("🔒", lockX + 3, lockY + fontSize + 1);
        gLock.setColor(Color.WHITE);
        gLock.drawString("🔒", lockX + 2, lockY + fontSize);
        gLock.dispose();
    }

    private void drawNavigationAndHUD(Graphics2D g2, int w, int h, int maxUnlocked) {
        float uiScale = Math.max(0.85f, Math.min(2.0f, Math.min((float) w / 960f, (float) h / 540f)));
        int stageW = Math.min(w, (int) (h * 16.0 / 9.0));
        int stageOffset = (w - stageW) / 2;
        int barY = Math.max(16, (int) (22 * uiScale));

        // --- BACK BUTTON ---
        int backW = (int) (120 * uiScale);
        int backH = (int) (36 * uiScale);
        backButtonBounds.setBounds(stageOffset + (int) (28 * uiScale), barY, backW, backH);
        boolean backHovered = backButtonBounds.contains(getMousePositionSafe());
        g2.setColor(backHovered ? new Color(42, 54, 88, 235) : new Color(24, 32, 54, 210));
        g2.fillRoundRect(backButtonBounds.x, backButtonBounds.y, backButtonBounds.width, backButtonBounds.height, (int) (12 * uiScale), (int) (12 * uiScale));
        g2.setColor(backHovered ? new Color(255, 215, 100) : new Color(242, 190, 75));
        g2.setStroke(new BasicStroke(1.5f * uiScale));
        g2.drawRoundRect(backButtonBounds.x, backButtonBounds.y, backButtonBounds.width, backButtonBounds.height, (int) (12 * uiScale), (int) (12 * uiScale));
        g2.setFont(new Font("Arial", Font.BOLD, Math.max(11, (int) (13 * uiScale))));
        g2.setColor(Color.WHITE);
        g2.drawString("← Balik (Back)", backButtonBounds.x + (int) (10 * uiScale), backButtonBounds.y + (int) (23 * uiScale));

        // --- SETTINGS BUTTON (Right) ---
        int setSize = (int) (38 * uiScale);
        int setX = stageOffset + stageW - setSize - (int) (28 * uiScale);
        settingsButtonBounds.setBounds(setX, barY, setSize, setSize);
        boolean setHovered = settingsButtonBounds.contains(getMousePositionSafe());
        g2.setColor(setHovered ? new Color(42, 54, 88, 235) : new Color(24, 32, 54, 210));
        g2.fillRoundRect(setX, barY, setSize, setSize, (int) (12 * uiScale), (int) (12 * uiScale));
        g2.setColor(setHovered ? new Color(255, 215, 100) : new Color(242, 190, 75));
        g2.drawRoundRect(setX, barY, setSize, setSize, (int) (12 * uiScale), (int) (12 * uiScale));
        if (settingsIcon != null) {

            g2.drawImage(settingsIcon, setX + iconInset, barY + iconInset, iconSize, iconSize, null);
        } else {

            g2.drawString("⚙", setX + (int) (11 * uiScale), barY + (int) (25 * uiScale));
        }

        // --- FULLSCREEN TOGGLE BUTTON ---
        int fsX = setX - setSize - (int) (8 * uiScale);
        fullscreenButtonBounds.setBounds(fsX, barY, setSize, setSize);
        boolean fsHovered = fullscreenButtonBounds.contains(getMousePositionSafe());
        g2.setColor(fsHovered ? new Color(42, 54, 88, 235) : new Color(24, 32, 54, 210));
        g2.fillRoundRect(fsX, barY, setSize, setSize, (int) (12 * uiScale), (int) (12 * uiScale));
        g2.setColor(fsHovered ? new Color(255, 215, 100) : new Color(242, 190, 75));
        g2.drawRoundRect(fsX, barY, setSize, setSize, (int) (12 * uiScale), (int) (12 * uiScale));
        g2.setFont(new Font("Segoe UI Symbol", Font.BOLD, Math.max(13, (int) (16 * uiScale))));
        g2.setColor(Color.WHITE);
        String fsIcon = app.isFullScreen() ? "🗗" : "⛶";
        int fsTw = g2.getFontMetrics().stringWidth(fsIcon);
        g2.drawString(fsIcon, fsX + (setSize - fsTw) / 2, barY + (int) (25 * uiScale));

        // --- CENTER TITLE BADGE ---
        int titleW = (int) (290 * uiScale);
        int titleH = (int) (36 * uiScale);
        int titleX = (w - titleW) / 2;
        g2.setColor(new Color(24, 32, 54, 210));
        g2.fillRoundRect(titleX, barY, titleW, titleH, (int) (12 * uiScale), (int) (12 * uiScale));
        g2.setColor(new Color(242, 190, 75));
        g2.drawRoundRect(titleX, barY, titleW, titleH, (int) (12 * uiScale), (int) (12 * uiScale));
        g2.setFont(new Font("Arial", Font.BOLD, Math.max(12, (int) (14 * uiScale))));
        g2.setColor(new Color(254, 240, 200));
        String title = "PILIA ANG MAPA  •  MAP SELECT";
        int tw = g2.getFontMetrics().stringWidth(title);
        g2.drawString(title, (w - tw) / 2, barY + (int) (23 * uiScale));

        // --- DYNAMIC HOVER HUD BADGE (LEVEL BANNER) ---
        if (hoveredLandmark >= 0) {
            String mapName;
            String diffName;
            Color bColor;

            if (hoveredLandmark == LANDMARK_BLUE_BUILDING) {
                mapName = "BLUE BUILDING";
                diffName = "EASY";
                bColor = new Color(110, 230, 130);
            } else if (hoveredLandmark == LANDMARK_DURIAN_MUSEUM) {
                mapName = "DAVAO MUSEUM";
                diffName = "MEDIUM";
                bColor = new Color(245, 175, 75);
            } else if (hoveredLandmark == LANDMARK_SAN_PEDRO) {
                mapName = "SAN PEDRO CATHEDRAL";
                diffName = "HARD";
                bColor = new Color(255, 120, 90);
            } else {
                mapName = "MT. APO";
                diffName = "EXTREME";
                bColor = new Color(185, 120, 255);
            }

            int bw = (int) (290 * uiScale);
            int bh = (int) (46 * uiScale);
            int bx = (w - bw) / 2;
            int by = barY + titleH + (int) (8 * uiScale);

            g2.setColor(new Color(15, 20, 36, 235));
            g2.fillRoundRect(bx, by, bw, bh, (int) (14 * uiScale), (int) (14 * uiScale));
            g2.setColor(bColor);
            g2.setStroke(new BasicStroke(1.8f * uiScale));
            g2.drawRoundRect(bx, by, bw, bh, (int) (14 * uiScale), (int) (14 * uiScale));

            g2.setFont(new Font("Arial", Font.BOLD, Math.max(12, (int) (14 * uiScale))));
            g2.setColor(new Color(255, 248, 233));
            int nw = g2.getFontMetrics().stringWidth(mapName);
            g2.drawString(mapName, (w - nw) / 2, by + (int) (20 * uiScale));

            g2.setFont(new Font("Arial", Font.BOLD, Math.max(10, (int) (12 * uiScale))));
            g2.setColor(bColor);
            int dw = g2.getFontMetrics().stringWidth(diffName);
            g2.drawString(diffName, (w - dw) / 2, by + (int) (37 * uiScale));
        }
    }

    private Point getMousePositionSafe() {
        Point p = getMousePosition();
        return (p != null) ? p : new Point(-100, -100);
    }

package juanword;

/**
 * Model class for player save data.
 * Matches the schema of data/players.csv:
 * playerId,playerName,currentStage,currentRound,score,experience,coins,currentHealth,maxHealth,lastSaved,savedTiles,savedDifficulty,savedEnemyHealth,savedMap,unlockedSayonStage,unlockedSaktoStage,unlockedLisodStage
 */
public class Player {
    public String playerId;
    public String playerName;
    public int currentStage = 1;
    public int currentRound = 0; // 0 = no active game started yet; >= 1 = active run
    public int score = 0;
    public int experience = 0;
    public int coins = 0;
    public int currentHealth = 100;
    public int maxHealth = 100;
    public String lastSaved = "";

    // Saved in-progress battle state for CONTINUE
    public String savedTiles = "";
    public String savedDifficulty = "Sakto";
    public int savedEnemyHealth = -1;
    public String savedMap = "San Pedro Cathedral";

    // Per-difficulty unlocked stages (San Pedro Cathedral: Stage 1 = Dizon, Stage 2 = Alviola, Stage 3 = Rola Boss)
    public int unlockedSayonStage = 1;
    public int unlockedSaktoStage = 1;
    public int unlockedLisodStage = 1;

    // Ordered Map progression (1: Blue Building [Easy], 2: Davao Museum [Medium], 3: San Pedro [Hard], 4: Mt Apo [Extreme])
    public int unlockedMapIndex = 1;

    public Player(String playerId, String playerName) {
        this.playerId = playerId;
        this.playerName = playerName;
        this.currentStage = 1;
        this.currentRound = 0;
        this.unlockedSayonStage = 1;
        this.unlockedSaktoStage = 1;
        this.unlockedLisodStage = 1;
        this.unlockedMapIndex = 1;
    }

    /**
     * Returns the maximum stage unlocked for a specific difficulty.
     * Stage 1 (Dizon) is always unlocked (>= 1).
     * Stage 2 (Alviola) is unlocked if return >= 2.
     * Stage 3 (Rola the Boss) is unlocked if return >= 3.
     */
    public int getMaxUnlockedStage(String difficulty) {
        if (difficulty == null) return Math.max(1, currentStage);
        switch (difficulty.trim().toLowerCase()) {
            case "sayon":
            case "easy":
                return Math.max(1, unlockedSayonStage);
            case "sakto":
            case "medium":
            case "normal":
                return Math.max(1, unlockedSaktoStage);
            case "lisod":
            case "hard":
                return Math.max(1, unlockedLisodStage);
            default:
                return Math.max(1, currentStage);
        }
    }

    /**
     * Records a stage unlock for a specific difficulty.
     * Guaranteed never to decrease existing unlocked stage progress.
     */
    public void recordStageUnlock(String difficulty, int stage) {
        if (difficulty == null) return;
        switch (difficulty.trim().toLowerCase()) {
            case "sayon":
            case "easy":
                unlockedSayonStage = Math.max(unlockedSayonStage, stage);
                break;
            case "sakto":
            case "medium":
            case "normal":
                unlockedSaktoStage = Math.max(unlockedSaktoStage, stage);
                break;
            case "lisod":
            case "hard":
                unlockedLisodStage = Math.max(unlockedLisodStage, stage);
                break;
        }
        currentStage = Math.max(currentStage, Math.min(stage, 3));
    }

    /**
     * Convenience helper: returns true if the boss (Stage 3 Rola) is unlocked for the specified difficulty.
     */
    public boolean isBossUnlocked(String difficulty) {
        return getMaxUnlockedStage(difficulty) >= 3;
    }

    /** Returns true if the map at the given 1-based index is unlocked. */
    public boolean isMapUnlocked(int mapIdx) {
        return unlockedMapIndex >= mapIdx;
    }

    /** Returns true if the named map is unlocked (1: Blue Building, 2: Davao Museum, 3: San Pedro, 4: Mt Apo). */
    public boolean isMapUnlocked(String mapName) {
        if (mapName == null) return false;
        String m = mapName.trim().toLowerCase();
        if (m.contains("blue") || m.contains("building")) return unlockedMapIndex >= 1;
        if (m.contains("museum") || m.contains("durian")) return unlockedMapIndex >= 2;
        if (m.contains("san pedro") || m.contains("pedro")) return unlockedMapIndex >= 3;
        if (m.contains("apo")) return unlockedMapIndex >= 4;
        return false;
    }

    /** Unlocks up to the specified map index (never decreases). */
    public void unlockNextMap(int mapIndex) {
        this.unlockedMapIndex = Math.max(this.unlockedMapIndex, mapIndex);
    }

    /** Returns the predefined difficulty for a map: Blue Building = Sayon (Easy), Davao Museum = Sakto (Medium), San Pedro = Lisod (Hard), Mt Apo = Extreme. */
    public static String getPredefinedDifficultyForMap(String mapName) {
        if (mapName == null) return "Sayon";
        String m = mapName.trim().toLowerCase();
        if (m.contains("blue") || m.contains("building")) return "Sayon";
        if (m.contains("museum") || m.contains("durian")) return "Sakto";
        if (m.contains("san pedro") || m.contains("pedro")) return "Lisod";
        if (m.contains("apo")) return "Extreme";
        return "Sayon";
    }

    /** Converts this player into a single CSV line. */
    public String toCsv() {
        return String.join(",",
                playerId,
                playerName.replace(",", " "),
                String.valueOf(currentStage),
                String.valueOf(currentRound),
                String.valueOf(score),
                String.valueOf(experience),
                String.valueOf(coins),
                String.valueOf(currentHealth),
                String.valueOf(maxHealth),
                lastSaved,
                (savedTiles != null ? savedTiles : ""),
                (savedDifficulty != null ? savedDifficulty : "Sayon"),
                String.valueOf(savedEnemyHealth),
                (savedMap != null ? savedMap : "Blue Building"),
                String.valueOf(unlockedSayonStage),
                String.valueOf(unlockedSaktoStage),
                String.valueOf(unlockedLisodStage),
                String.valueOf(unlockedMapIndex)
        );
    }

    /** Parses one CSV line into a Player instance (returns null if invalid). */
    public static Player fromCsv(String line) {
        if (line == null || line.trim().isEmpty()) return null;
        String[] f = line.split(",", -1);
        if (f.length < 10) return null;
        try {
            Player p = new Player(f[0].trim(), f[1].trim());
            p.currentStage = Integer.parseInt(f[2].trim());
            p.currentRound = Integer.parseInt(f[3].trim());
            p.score = Integer.parseInt(f[4].trim());
            p.experience = Integer.parseInt(f[5].trim());
            p.coins = Integer.parseInt(f[6].trim());
            p.currentHealth = Integer.parseInt(f[7].trim());
            p.maxHealth = Integer.parseInt(f[8].trim());
            p.lastSaved = f[9].trim();
            if (f.length >= 11) {
                p.savedTiles = f[10].trim();
            }
            if (f.length >= 12) {
                p.savedDifficulty = f[11].trim();
            }
            if (f.length >= 13) {
                try { p.savedEnemyHealth = Integer.parseInt(f[12].trim()); } catch (Exception ignored) {}
            }
            if (f.length >= 14) {
                p.savedMap = f[13].trim();
            }
            if (f.length >= 15 && !f[14].trim().isEmpty()) {
                try { p.unlockedSayonStage = Integer.parseInt(f[14].trim()); } catch (Exception ignored) {}
            } else {
                p.unlockedSayonStage = Math.max(1, p.currentStage);
            }
            if (f.length >= 16 && !f[15].trim().isEmpty()) {
                try { p.unlockedSaktoStage = Integer.parseInt(f[15].trim()); } catch (Exception ignored) {}
            } else {
                p.unlockedSaktoStage = Math.max(1, p.currentStage);
            }
            if (f.length >= 17 && !f[16].trim().isEmpty()) {
                try { p.unlockedLisodStage = Integer.parseInt(f[16].trim()); } catch (Exception ignored) {}
            } else {
                p.unlockedLisodStage = Math.max(1, p.currentStage);
            }
            if (f.length >= 18 && !f[17].trim().isEmpty()) {
                try { p.unlockedMapIndex = Integer.parseInt(f[17].trim()); } catch (Exception ignored) {}
            } else {
                p.unlockedMapIndex = 1;
            }
            return p;
        } catch (NumberFormatException e) {
            return null;
        }
    }

    @Override
    public String toString() {
        return playerName + " (" + playerId + ") - Stage " + currentStage
                + ", Round " + currentRound + ", Score " + score
                + ", XP " + experience + ", Coins " + coins
                + ", HP " + currentHealth + "/" + maxHealth
                + ", Saved: " + lastSaved
                + ", Unlocks: [Sayon: " + unlockedSayonStage + ", Sakto: " + unlockedSaktoStage + ", Lisod: " + unlockedLisodStage + "]"
                + ", UnlockedMap: " + unlockedMapIndex
                + ", Tiles: " + (savedTiles.isEmpty() ? "None" : savedTiles);
    }
}

        // Spiked / Studded Corners: Spicy Sweetcorn

        g2.fillOval(x + 4, y + 4, 6, 6);

        g2.fillOval(x + w - 10, y + h - 10, 6, 6);
